 \begin{lemma}[Orientation Class of a Disk]\label{lem:filt-orientationClassOfDisk}
	Let $n\geq 1$ and let $(\Sigma,\partial\Sigma)$ be a compact pair such that $\Sigma\setminus\partial\Sigma$ is an oriented $C^1$-manifold of dimension $n$. Let $g:S^n\to\Sigma/\partial\Sigma$ be a homeomorphism with $g(S)=[\partial\Sigma]$ and let $U\subseteq S^n\setminus\{S\}$ be open such that $\restr{g}{U}$ is an orientation preserving $C^1$-diffeomorphism onto an open subset of $\Sigma\setminus\partial\Sigma$. Define $o_\Sigma\coloneq(g^{-1})^*(\beta_n)\in K^{-n}(\Sigma,\partial\Sigma)$. Let $V$ be an oriented $n$-dimensional vector space and $h:\Sigma/\partial\Sigma\to\overline V$ a pointed map such that for some $z\in V$ we have $h^{-1}(z)=\{y\}$ with $y\in g(U)$, $h$ is $C^1$ near $y$ and $\restr{\diff h}{y}$ is invertible. Then
	\begin{equation*}
		h^*(\beta_V)=\sign\det\left(\restr{\diff h}{y}\right)\cdot o_\Sigma\,.
	\end{equation*}
\end{lemma}

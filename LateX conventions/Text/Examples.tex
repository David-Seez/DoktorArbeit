	\section{Examples}
	We begoin with a minimal example that has non trivial higher differential:
	\begin{example}[The Sphere]
	If we take the three-sphere $S^3\subseteq \R^4$ and denote the north pole with $N$ and the south pole with $S$ we get the Morse chain complex as
	\begin{center}
		\begin{tikzcd}
		0\arrow[r]& \langle S \rangle_{\Z} \arrow[r] &0\arrow[r] & 0\arrow[r]&\langle N \rangle_{\Z} \arrow[r]& 0 
		\end{tikzcd}
	\end{center}Then for the second page we get the Morse homology together with the 0 maps and on the third page we have: 
	\begin{center}
		\begin{tikzcd}
			\langle S\rangle  & 0 & 0 &\arrow[lll, "d_3^0", bend left] \langle N\rangle
		\end{tikzcd}
	\end{center}
	Now we want to check what $d_3^0(S)$ is. For this we want to use its definition and hence we need to give the Morse filtration. For this we can use:
	\begin{equation*}
		N_{-1}\coloneq\emptyset,\quad N_{0,1,2}\coloneq H_- \text{ the southern hemisphere},\quad  N_3=S^3\,.
	\end{equation*}
	First we translate $E_r$ into the language where the definition was formulated. 
	We know that $EM_3^p$ is naturally isomorphic to $$E^{p,0}_3=\frac{\gamma^{-1}\big(\alpha^2D^{p,-2}\big)}{\beta\big(\ker(\restr{\alpha^2}{D^{p+3-1,-2}})\big)}$$
	Now for $r=3$ and $p=0$
	\begin{equation*}
		\alpha^2D^{2,-2}=\im (K^0(N_2)\to K^0(N_0))=\im(K^0(H_-)\to K^0(H_-))\cong Z
	\end{equation*} Hence $E^{0,0}_3={\gamma^{-1}(\alpha^2D^{2,-2})}^{0,0}\cap E^{0,0}_0= K^0(N_0,\emptyset)\cong \Z$. And similar we conclude that $E^{3,0}_3=K^3(N_3,N_2)\cong\Z$. 
	Now we analyze the map 
	\begin{equation*}
		d^{3,0}_3:  K^3(N_3,N_2) \to K^0(N_0,\emptyset) 
	\end{equation*} So far we where able to ignore all orientation, (which we formally need for the morse boundary operator). So lets fix the orientations of the unstable part by saying the unstable part of the north pole (i.e. the whole $S^3$) is the canonical orientation of $S^3\supset \R^4$, whilst the unstable part of the south pole (i.e. the south pole) has the orientation $-1$. 
	
	Now $d_r=\beta\circ \alpha^2\circ \gamma$ and $\gamma: K^3(N_3,N_2)\to K^0(N_3)$ maps our generator induced by $N$ to the standard generator of $K^3(S^3)$ induced by the choosen orientation, due to $N_2$ being a contractible subset. Now $\alpha^2:  K^3(N_3) \to  K^3(N_1)$ is the zero map, since its range is $0$ due to $N_1$ being homotopically equivalent to the point and $3$ being odd.Finally $\beta: K^3(N_1) \to K^4(N_0,\emptyset)$ is the connecting homomorphism that trivial here. Hence $d^3,0_3$ is the zero map.And thereby $EM_4$ which equalse $E_\infty$ is  
	\begin{center}
		\begin{tikzcd}
				\langle S\rangle  & 0 & 0 & \langle N\rangle
		\end{tikzcd}
	\end{center}Now how to we get information about our beloved K groups back? For this we remined ourself about the filtration 
	\begin{equation*}
		K^0(S^3)=F{-1}K^0(S^3)\supseteq F^0K^0(S^3)\supseteq F^1 K^0(S^3) \supseteq F^2K^0(S^3)\supseteq F^3K^0(S^3)=0
	\end{equation*}
	Now using this filtration we have the short exact sequence
	\begin{center}
		\begin{tikzcd}
			0 \arrow[r]&F^0K^0(S^3)\arrow[r] &K^0(S^3)\arrow[r]&E^{0,0}_4=\frac{F^{-1}K^0(S^3)}{F^{0}K^0(S^3)} \arrow[r]&0
		\end{tikzcd}
	\end{center}Hence, $K^0(S^3)$ is an extension of $F^0K^0(S^3)$ and $E^{0,0}_4\cong \Z$ 
	Furthermore, we have the exact sequence
	\begin{center}
		\begin{tikzcd}
			0 \arrow[r]
			&F^1K^0(S^3)\arrow[r] 
			&F^0K^0(S^3)\arrow[r]
			&E^{1,-1}_4=\frac{F^{0}K^0(S^3)}{F^{1}K^0(S^3)}=0 \arrow[r]
			&0
		\end{tikzcd}
	\end{center} Hence, $F^0K^0(S^3)\cong F^1K^0(S^3)$ and by the same argument we can continue this chain 
	\begin{equation*}
		F^0K^0(S^3)\cong F^1K^0(S^3)\cong F^2K^0(S^3)\cong F^3K^0(S^3)=0
	\end{equation*}
	Hence, the first short exact sequence gives is an isomorphism between $K^0(S^3)\cong \Z$.
	\end{example} 
	Now we want to calculate the K group of a non trivial example:
	\begin{example}[The Real Projective Space]
	Here, we define $\R\PP^n$ to be $\R^{n+1}\setminus 0$ modulo scalar multiples. Hence it is the space of lines in $\R^{n+1}$. Alternatively we can view it as the $n$-sphere where the operation from $\{\id,-\id\}$ is quotient out. From this description we get a metric, since $S^n\supseteq \R^{n+1}$ inherits the subspacae metric in which all operations from $\{\id,-\id\}$ are isometries. Hence, this metric projects down to $\R\PP^n$. Now let
	\begin{equation*}
		A=\diag(\lambda_0,\lambda_1,\dots,\lambda_n)
	\end{equation*}be a symmetric matrix with real $\lambda_i\in \R$. Then we define the map
	\begin{equation*}
		\Lambda:\R^{n+1}\setminus\{0\}\to \R; \Lambda(x)\coloneq \frac{\langle x,Ax\rangle}{\norm{x}^2}\,.
	\end{equation*}Notice that this function descends to a well defined function $\tilde{\Lambda}:\R\PP^n\to \R$. On the whole $\R^{n+1}$ we can define the function
	\begin{equation*}
		f:\R^{n+1}\to \R;~f(x)\coloneq \frac{1}{2}\langle x,Ax\rangle=\frac{1}{2}\sum_{i=0}^n\lambda_i x_i^2\,.
	\end{equation*} Now assume that $\phi_t(x)$ is the flow associated to $- \grad(f)$, meaning
	\begin{equation*}
		\frac{\diff \phi_t(x)}{\diff t}=-\grad_{\phi_t(x)}(f)=-Ax\,,
	\end{equation*} since 
	\begin{equation*}
		\frac{\partial f}{\partial x_i}(x)=\lambda_ix_i\,.
	\end{equation*}
	Now we have the lemma: 
	\begin{lemma}
		The flow lines of the negative gradient flow with respect to $\frac{1}{2}\tilde{\Lambda}$ on $\R\PP^n$ are the images under the quotient map 
		\begin{equation*}
			\pi: \R^{n+1}\setminus \{0\}\to \R\PP^n
		\end{equation*}of the flow lines of the linear system
		\begin{equation*}
			\frac{\diff \Phi_t(x)}{\diff t}=-A \Phi_t(x)
		\end{equation*}
	\end{lemma}
	\begin{proof}
	Let $x$ be a point on the flow line $\Phi_t(x)$. Then $-\grad(f)(x)=-Ax.$ Now we split this into a part that is orthogonal to $y$ and one that is paralell:
	\begin{equation*}
		-\grad(f)(x)=\underbrace{-\Lambda(x)x}_{\text{parallel to } x}~+~\underbrace{-Ax+\Lambda(x)x}_{\text{orthogonal to }x}\,.
	\end{equation*} The claim of orthogonality is just a calculation of the inner product $\langle -Ax+\Lambda(x)x,x\rangle=0$
	\end{proof} Now we radially project this onto the sphere via 
	\begin{equation*}
		\pi_r:\R^{n+1}\setminus\{0\}\to S^n; x\mapsto \frac{x}{\norm{x}}
	\end{equation*} the differential of this projection kills the parallel part and hence reads:
	\begin{equation*}
		\restr{\diff \pi_r}{x}(-\grad_x(f))=\frac{-Ax+\Lambda(x)x}{\norm{x}}
	\end{equation*} If we restrict $\restr{f}{S^n}$ we get that the gradient $-\grad_y(\restr{f}{S^n})=-Ay+\Lambda(y)y$. To see this, let $v\in T_yS^b\supseteq T_yR^{n+1}$ and remember that the inner produkt on $S^b$ is induced from the euclidean one. Then 
	\begin{align*}
		\langle \grad_y(\restr{f}{S^n}),y\rangle_{S^n}
		&=\diff \restr{f}{S^n}(v)=\diff f(v)=\langle \grad_y(f),v\rangle_{\R^{n+1}}\\
		&=\langle \mathrm{proj}_{T_yS^n}(\grad_y(f)),v\rangle_{\R^{n+1}}=\langle Ay-\Lambda(y)y,v \rangle_{\R^{n+1}}
	\end{align*} Here $\mathrm{proj}_{T_yS^n}$ denotes the projection onto the subspace ${T_yS^n}\subseteq T_y\R^{n+1}$. This however tells us that the image $\widetilde{\Phi_t(x)}$ of $\Phi_t(x)$ under the radial projection solves the equation
	\begin{equation*}
		\frac{\diff \widetilde{\Phi_t(y)}}{t}=-\grad_y(\restr{f}{S^n})(\widetilde{\Phi_t(y)})=-\frac{1}{2}\grad_y(\restr{\Lambda}{S^n})(\widetilde{\Phi_t(y)})\quad \text{ since } f=\frac{1}{2}\Lambda
	\end{equation*} Hence descending to the quotient $\R\PP^n$ yields that $\left[\widetilde{\Phi_{2t}(y)}\right]$ is a solution of the flow associeted to the negative gradient of $\tilde{\Lambda}$ in $\R\PP^n$. Now the other way round should be trivial: Take a solution $\psi_t(x)$ of $\tilde{\Lambda}$ in $\R\PP^n$. Then for fixed $t,x$ let $U$ be a neighbourhood of $\psi_t(x)$. This $U$ can be lifted to $S^n$ and the solution there solves the lifted version of the flow induced from $\tilde{\Lambda}$. And from here we can argue that the solution is part of a solution of the linear differential equation 
	\begin{equation*}
		\frac{\diff \phi_t(x)}{\diff t}=-\grad_{\phi_t(x)}(f)=-Ax\,.
	\end{equation*} This concludes the proof.
	Hence critical points in $\R\PP^n$ are eigenvalues of $A$. Hence, for the function to be Morse, we need the critical points to be isolated, meaning that all eigenspaces have dimension one. So we assume that the diagonal entries in $A$ are all different. For simplicity we assume that $\lambda_0<\lambda_1\cdots <\lambda_n$. Since $A$ is diagonal we know that $[e_i]$ are the critical points.
	\begin{lemma}
		Each critical point $[e_i]$ is non degenerate, of index $i$ and its unstable manifold is
		\begin{equation*}
			W([e_i]\to)\eqcolon U_i=\{[x_0:x_1,\dots x_i,0,\dots,0]\in \R\PP^n| x_i\neq 0\}\,,
		\end{equation*} whilst the stable manifold is
		\begin{equation*}
			W(\to [e_i])\eqcolon S_i=\{[0,\dots,0,x_i,x_{i+1},\dots,x_n]\in\R\PP^n|x_i\neq 0\}
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Let $B^n$ be the open unit ball in $\R^n$ and $V_i\coloneq \{[x_,\dots,x_n]\in\R\PP^n|x_i\neq 0\}$. Then we define the parametrization
		\begin{align*}
			\phi_i:B^n &\to V_i\\
			(y_1,\dots y_n)&\mapsto [y_1,\dots,y_i,\sqrt{1-\norm{y}_2^2},y_{i+1},\dots,y_n]\,.
		\end{align*} Now the image gives representatives of norm one and hence in this chart we calculate:
		\begin{align*}
			\tilde{\Lambda}\circ \phi_i(y_1,\dots,y_n)
			&= \sum_{j=1}^iy_i^2\lambda_{i-1}+\lambda_i({1-\norm{y}_2^2})+\sum_{j=i+1}^n y_j^2\lambda_j\\
			&= \sum_{j=1}^iy_i^2\lambda_{i-1}+\lambda_i(1-\sum_{j=1}^n y_j^2 )+\sum_{j=i+1}^n y_j^2\lambda_j \\
			&= \sum_{j=1}^iy_i^2(\lambda_{i-1}-\lambda_i)+\lambda_i+\sum_{j=i+1}^n y_j^2(\lambda_j-\lambda_i)
		\end{align*}
	\end{proof}
	\end{example}

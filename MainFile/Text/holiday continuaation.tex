\begin{definition}
	Let $N$ be a compact subset of a local flow $X\subseteq \Gamma$. Then a subset $K\subseteq N$ is said to be \textbf{positively invariant} in $N$ if 
	\begin{equation*}
		\gamma \in K , \quad t\geq0,\quad \phi_{[0,t]}(\gamma)\subseteq N \Rightarrow \phi_t(\gamma)\in K\, .
	\end{equation*}
\end{definition}
	
\begin{definition}
	Let $X\subseteq \Gamma$ be a local flow and let $S\subseteq X$ be an isolated invariant set. Then a pair $(N_1,N_0)$ of compact sets in $X$ is said to be an \textbf{index pair} for $S$ in $X$ is $N_0\subseteq N_1$0 and
	\begin{enumerate}
		\item $N_1\setminus N_0$ is a neighbourhood of $S$ in $X$ and $S=I(\cl(N_1\setminus N_0))$,
		\item $N_0$ is positively invariant in $N_1$, and
		\item if $\gamma\in N_1$ and $\phi_{[0,\infty)}(\gamma)\subseteq N_1$, then there exists a $t\geq0$ with $\phi_t([0,t])(\gamma)\subseteq N_1$ and $\phi_t(\gamma)\in N_0$.
	\end{enumerate}
\end{definition}

\section{A Morse-Theoretic Filtration}
\subsection{Conley index}
\begin{definition}[Compact Invariant Isolated Subset] \label{def:morseFiltr-compactInvariantIsolatedSubset}
	For a flow $\phi_t:M\to M$ on a locally compact metric space, we call a subset $S\subseteq M$ an \textbf{invariant subset} if and only if $\phi_t(S)=S$ for all $t\in \R$. For any subset $N\subseteq M$ we define the \textbf{maximal invariant subset}
	\begin{align*}
		I(N)&=\{x\in N \mid \phi_t(x)\in N \forall t\in \R\}\\
		&=\bigcap_{t\in \R} \phi_t(N).
	\end{align*} A \textbf{compact invariant subset} $S$ is called \textbf{isolated} if a compact neighbourhood $N$ exists such that $I(N)=S$.
\end{definition}
\begin{definition}[Index Pairs] \label{def:morseFiltr-indexPairs}
	Let $S$ be an isolated compact invariant subset. A topological pair $(N,L)$ of compact subsets of $M$ where $L\subseteq N$ is called an \textbf{index pair of }S, if it satisfies the following:
	\begin{enumerate}
		\item $S=I(\overline{N\setminus L})\subseteq \inti({N\setminus L})$.
		\item  $x\in L$ and $\phi_{[0,t]}(x)\subseteq N$ implies that  $\phi_{[0,t]}(x)\subseteq L$. We call $L$ \textbf{positively invariant in } $N$. Here $\phi_{[0,t]}(x)\coloneq\{ \phi_{\tilde{t}}(x)|\tilde{t}\in [0,t]\}$.
		\item For all $x\in N$ such that a $t$ exists, with $\phi_t(x)\not\in N$, there exists a $t'$ with $\phi_{[0,t']}(x)\subseteq N$ and $\phi_{t'}(x)\in L$.
	\end{enumerate}
	This definition captures how the flow lines leave the invariant space $S$. Due to the third property we call $L$ the \textbf{exit set}.
\end{definition}


\begin{definition}[Regular Index Pairs]\label{def:morseFiltr-regularIndexPairs}
	We call an index pair $(N,L)$ \textbf{regular} if and only if the inclusion $i: L \hookrightarrow N$ is a cofibration.
\end{definition}
\begin{theorem}[Homotopy Equivalence of Index Pairs]\label{thm:morseFiltr-homotopyEquivalenceOfIndexPairs}
	If $S$ is an isolated compact invariant subset and $(N,L)$ and $(\tilde{N},\tilde{L})$ are two index pairs of $S$, then $N/L$ and $\tilde{N}/\tilde{L}$ are homotopy equivalent as pointed spaces. This homotopy equivalence is canonical, and transitive up to homotopy. 
\end{theorem}
\begin{proof}
	For a proof of the homotopy equivalence we refer to \cite{banyaga2004lectures}, Theorem 7.15. The constructed homotopy equivalence is given as follows: Choose $T\geq 0$ such that the following implications hold for $t\geq T$:
\begin{align*}
\phi_{[-t,t]}(x) \subseteq N\setminus L \quad &\Rightarrow \quad x \in \tilde{N}\setminus \tilde{L}\,, \\
\phi_{[-t,t]}(x) \subseteq  \tilde{N}\setminus \tilde{L} \quad &\Rightarrow \quad x \in N\setminus L\,, 
\end{align*} where $\phi_{[a,b]}(x) \coloneq \{\phi_t(x)\mid t\in [a,b]\}$.
Then the map:
\begin{align*}
h:N/L \times [T,\infty) & \to \tilde{N}/\tilde{L} \\
([x],t) & \mapsto 
\begin{cases}
[\phi_{3t}(x)] & \text{ if } \phi_{[0,2t]}(x)\subseteq N\setminus L \text{ and } \phi_{[t,3t]}(x)\subseteq \tilde{N}\setminus \tilde{L}\\
[\tilde{L}] & \text{ otherwise,}
\end{cases}
\end{align*}
is continuous and for any fixed $t$ this gives our homotopy equivalence. Obviously, such a map is canonical and transitive up to homotopy.
\end{proof}

\begin{definition}[Connected Simple System]\label{def:morseFiltr-connectedSimpleSystem}
	A \textbf{connected simple system} consists of a set $I_0$ of pointed spaces along with a set $I_m$ of homotopy classes of maps between these spaces such that 
		\begin{enumerate}[(i)]
			\item $\hom(X,Y)\coloneq \{[f]\in [X,Y]\mid[f]\in I_m\}$ consists of exactly one element for each pair of spaces $(X,Y)\in I_0^2$, where $[X,Y]$ denotes the set of homotopy classes of maps between $X$ and $Y$.
			\item If $X,Y,Z\in I_0,[f]\in \hom(X,Y),$ and $[g]\in \hom(Y,Z)$, then $[g\circ f]\in \hom(X,Z)$.
			\item $\hom(X,X)=\{[\id_X]\}$ for all $X\in I_0$. 
		\end{enumerate}
\end{definition}
\begin{remark}\label{rem:morseFiltr-connectedSimpleSystemSalamon}
	This definition comes from Salamon in \cite{Salamon1985}.
\end{remark}


\begin{remark}\label{rem:morseFiltr-indexPairConnectedSimpleSystem} Let $S$ be an isolated compact invariant subset of $M$ with respect to the Morse flow $\phi$. Then the collections
\begin{align*}
	I_0&\coloneq \{N  / L\mid (N,L)\text{ is an index pair of }S\}\\
	I_m&\coloneq \{[f] \mid f \text{ is a homotopy equivalence from Theorem \ref{thm:morseFiltr-homotopyEquivalenceOfIndexPairs}}\}\,,
\end{align*} make up a connected simple system. We denote this by $\mathcal{IP}_S$. If $S=\Crit_k(f)$ for some $k\in \N$, we denote $\mathcal{IP}_{\Crit_k(f)}=\mathcal{IP}_k$.
\end{remark}


\begin{definition}[Morphisms of Connected Simple Systems]\label{def:morseFiltr-morphismsOfConnectedSimpleSystems}
A \textbf{morphism} $\Phi:I\to J$ between two connected simple systems $I=(I_0,I_m)$ and $J=(J_0,J_m)$ is a set of homotopy classes of maps between spaces in $I_0$ and $J_0$ such that
\begin{enumerate}[(i)]
	\item For every $X\in I_0$ and $Y\in J_0$ the set $\Phi(X,Y)\coloneq \{[\phi]\in [X,Y]\mid[\phi]\in \Phi\}$ consists of exactly one element. 
	\item If $X,\bar{X}\in I_0$ and $Y,\bar{Y}\in J_0$ and if $[\phi]\in \Phi(X,Y)$, $[f]\in \hom(\bar{X},X)$, $[g]\in \hom(Y,\bar{Y})$, then $[g\circ \phi\circ f]\in \Phi(\bar{X},\bar{Y})$
\end{enumerate}
\end{definition}
\begin{remark}\label{rem:morseFiltr-morphismInducedBySingleMap}
	By the second property of a morphism between connected simple systems $I$ and $J$, we can induce such a morphism by a single map $\phi:X\to Y$ where $X\in I_0$ and $Y\in J_0$. 
\end{remark}


\begin{definition}[Canonical Isomorphism Class]\label{def:morseFiltr-canonicalIsomorphismClass}
	A \textbf{canonical isomorphism class} of Abelian groups (or other algebraic structures) is a set of groups $R_0$ together with a set of group homomorphisms $R_m$ such that
	\begin{enumerate}[(i)]
		\item For $R,L \in R_0$ the set $\hom(R,L)\coloneq \{f:R\to L: f \in R_m\}$ consists of exactly one element. 
		\item If $R,L,S\in R_0$ and $f\in \hom(R,L)$ and $g\in \hom(L,S)$, then $g\circ f\in \hom(L,S)\,$.
		\item $\hom(R,R)=\{\id_R\}$
	\end{enumerate}
	A \textbf{morphism} between two canonical isomorphism classes of Abelian groups $\Psi: (R_0,R_m)\to (S_0,S_m)$ is a collection of homomorphisms such that 
	\begin{enumerate}[(i)]
		\item For any $R\in R_0$ and $S\in S_0$ the set $\Psi(R,S)\coloneq \{f:R\to S: f\in \Psi\}$ contains exactly one element.
		\item If $R,\bar{R}\in R_0$ and $S,\bar{S}\in S_0$, $f\in \hom(\bar{R},R)$, $g\in \hom(S,\bar{S})$ and $\psi\in \Psi(R,S)$, then $g\circ \psi\circ f\in \Psi(\bar{R},\bar{S})\,$.
	\end{enumerate}
\end{definition}
\begin{remark}
	Again the second property lets us induce a morphism between two canonical isommorphism classes of Abelian Groups $\Psi: (R_0,R_m)\to (S_0,S_m)$ by a single isomorphism 
	\begin{equation*}
		\psi: R\to S
	\end{equation*} for $R\in R_0$ and $S\in S_0$. We denote the induced morphism by $\psi_{\#}\coloneq \Psi$. 
\end{remark}

\begin{example}
	Any pointed topological space $X$ defines a simple connected system $(\{X\},{[\id_x]})$. Similar, any Abelian group $G$ defines a caninical isomorphism class of Abelian groups $(\{G\},\{\id_R\})$.
	Let $\phi :S \to L$ be a second homomorphism that induces a morphism of canonical isomorphism classes. It is then obvious that 
	\begin{equation*}
		(\phi\circ \psi)_{\#}=\phi_{\#}\circ \psi_{\#}\,.
	\end{equation*}
\end{example}

\begin{definition}\label{def:morseFiltr-systemCategories}
We denote the category of connected simple systems with their morphisms by $\CSS$ and similar we call the category of canonical isomorphism classes of Abelian groups by $\CIAB$, respectively, the category for rings by $\CIRing$.
\end{definition}

\begin{definition} \label{def:morseFiltr-postcomposedFunctor}
	Let $G$ be an Abelian group and $X$ be any set. We can \textbf{tag} $G$ with $X$ by defining the set $G\times \{X\}$ and endowing it with the algebraic information from $G$. (for example we define $(g,X)\cdot (g',X)=(g\cdot_Gg',X)$). Let $f:G\to L$ be a homomorphism. Then $f$ induces a homomorphism between the tagged $G\times \{X\}$ and $L\times \{Y\}$ by sending $(g,X)\mapsto (f(g),Y)$.
	Let $F:\TopP\to \Ab$ be a (contravariant) functor from pointed topological spaces to Abelian groups that factors over the homotopy category, i.e. is homotopy invariant. We first define the tagged functor $F^{tag}$ to map spaces to groups that are tagged with the spaces they come from:
	\begin{align*}
		F^{tag}: \TopP &\to \Ab\\
		X&\mapsto F(X)\times \{X\}\\
		\big(f:X\to Y\big)&\mapsto \big(F^{tag}(f):F(Y)\times \{Y\}\to F(X)\times\{X\}\big)
	\end{align*}
	For covariant functors the procedure is analogous.
	Now we can define the \textbf{postcomposed functor} $F_{\#}$ from the category of connected simple systems to the canonical isomorphism classes as follows:
	\begin{align*}
	F_{\#} : \CSS		&\to   \CIAB \\
	I_0 & \mapsto \{F^{tag}(X)\mid X \in I_0 \} \\
	I_m & \mapsto \{ F^{tag}(g) \mid [g]\in I_m \}\\
	\end{align*}
	Let $\Phi:(I_0,I_m)\to (J_0,J_m)$ be a morphism in $\CSS$. Then we define
	\begin{equation*}
		F_{\#}(\Phi): F_{\#}(J_0,J_m) \to  F_{\#}(I_0,I_m)\,,\quad 
		F_{\#}(\Phi) \coloneq \{F^{tag}(g)\mid [g]\in \Phi \}\,.
	\end{equation*}
This obviously becomes a new functor, where the tagging removed uniqueness problems for non-injective (in the literal sense not up to isomorphism) functors $F$. If $F(X)=F(Y)$ for different $X,Y\in I_0$ with $g:X\to Y\in I_m$, then without tagging there could be two different morphisms $F(X)\to F(X)$: the identity and $F(g)$. Since our functors are almost always injective, we will ignore the tagging from now on. 
\end{definition}
\begin{remark}
	Obviously, the procedure of postcomposing a functor works for any homotopy invariant functor from pointed topological spaces into any algebraic category and not just Abelian groups.
\end{remark}

\begin{remark}[Critical points as isolated compact invariant subsets] \label{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets}
	Let ${f:M\to \R}$ be a Morse function on a smooth Riemannian manifold $(M,g)$. Let $q\in \Crit_k(f)$. Around $q$ we have refined Morse charts $\phi:U \to \R^m\cong \R^{m-\lambda_q}\oplus \R^{\lambda_p}$ where  the splitting of $\R^m$ is taken such that $\diff \phi_q$ is expanding in $\R^{m-\lambda_q}$ and contracting in $\R^{\lambda_p}$. With this we can define the balls:
	\begin{align*}
		D^{s}_\epsilon &~\coloneq~\{v\in \R^{m-\lambda_q} \mid \norm{v}\leq \epsilon \}, \\
		D^{u}_\epsilon &~\coloneq~\{v\in \R^{\lambda_q\phantom{-m}} \mid \norm{v}\leq \epsilon \}. 
	\end{align*}
For epsilon small enough, they give rise to the index pair $N_q\coloneq\phi^{-1}( D^{s }_\epsilon\times  D^{u }_\epsilon)$ and $L_q\coloneq\phi^{-1}( D^{ s}_\epsilon\times \partial D^{u}_\epsilon)$. To see this, notice that in such a refined Morse chart $\phi$, the differential of the flow in the critical point splits the tangent space into $T^u_qM\times T^s_qM$. Finally, since the property of $\diff \phi_x$ being a contraction on $T^s_qM$ is an open condition and thereby there is an $\epsilon>0$ such that $\diff \phi_x$ is a contraction on $T^s_qM$ for all $x$ with $\norm{\phi^{-1}(x)}<\epsilon$. Thereby, the norm of $\phi_t(x)$ projected to $T^s_qM$ is strictly decreasing in $t$. This proves that $L_q$ is an exit set of $N_q$. Similarly, one sees that $L_q$ is positively invariant in $N_q$. 

This index pair is in fact regular, since $(N_q,L_q)\simeq  (D^{ s}_\epsilon\times  D^{u }_\epsilon, D^{s}_\epsilon\times \partial D^{u}_\epsilon) \simeq (D^{u}_{\epsilon},\partial D^{u}_{\epsilon} )$ and the latters regularity is easy to see by showing that it is an NDR pair. We will cal such an index pair a \textbf{standart index pair of $q$}.

Furthermore, the index pair $(N_q,L_q)$ is homotopically equivalent to $\overline{T^u_qM}$, which is clear, as $(D^{u}_{\epsilon},\partial D^{u}_{\epsilon} ) \simeq \overline{T^u_qM}$ by mapping $\R^{\lambda_q}\to T^u_qM$ via $\restr{\diff \phi^{-1}}{0}\circ R_0$, where $R_0:\R^m\to T_0\R^m$ is the linear identification from Derfinition \ref{def:vb-linearIdentification}. Finally, we just need to strech the ball to infinity. 


We know the K-group of such a pair to be:
	\begin{align} \label{eq: homology of index pair of critical point}
		K^i(N_q,L_q)\cong K^i(\overline{T^u_qM})\cong \begin{cases}
			\Z & \text{if } i-k\in 2\Z,\\
			0& \text{else. }
		\end{cases}
	\end{align}
	Notice that for $k=0$ we need to consider $\partial D^k=\emptyset$ to get an index pair. That's why we let $\partial$ denote the manifold boundary instead of the topological boundary. However, now we can identify for all $k$: \begin{equation} \label{ kritical points as direct sum of homology groups}
		C^{k}(M,\mathfrak{A},\tilde{K}^{l}(pt))=\bigoplus_{ \Crit_k(f)} \Z \cong \bigoplus_{q\in \Crit_k(f)} K^{-k}(N_q,L_q).
	\end{equation}
	Or using the language of connected simple systems, we have the identification 
	\begin{equation} 
		C^{k}(M,\mathfrak{A},\tilde{K}^{l}(pt))\cong \bigoplus_{q\in \Crit_k(f)} K_{\#}^{-k}(\mathcal{IP}_q)\,,
	\end{equation}where the isomorphism can be read as an isomorphism of canonical isomorphism classes with the left-hand side vierwd as an object of $\CIAB$ consists of only one group.
\end{remark}
\begin{remark}\label{rem:morseFiltr-indexPairForDifferentChart}
	For any chart $\chi$ that is centered at a critical point $q$ and $\epsilon>0$ we define the topological pair 
	\begin{equation*}
		(N_\chi,L_\chi)\coloneq (\chi^{-1}(D^s_{\epsilon}\times D^u_{\epsilon}), \chi^{-1}(D^s_{\epsilon}\times \partial D^u_{\epsilon}))\,.
	\end{equation*}The argument that shows that $(N_\psi,L_\psi)$ is a reglar index pair of $q$ only needed that $\restr{\diff \psi}{q}$ splitts. Hence if $\restr{\diff \chi}{q}=\restr{\diff \psi}{q}$, we can conclude, that there is an $\epsilon>0$ such that $(N_\chi,L_\chi)$ is also a regular index pair.
\end{remark}


\begin{remark}\label{rem:morseFiltr-morphismToCanonicalIntegers}
	Let $\underline{\Z}$ be the canonical isomorphism class of Abelian groups consisting only of the group $\Z$, and let $q$ be a critical point of index $\lambda_q$ in $M$. Then, Remark \ref{rem:morseFiltr-morphismInducedBySingleMap}, together with Remark \ref{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets} gives us the existence of a morphism for all $i\in \N$:
	\begin{equation*}
		\tilde{K}^{-\lambda_q-2i}(\mathcal{IP}_{\{q\}})\to \underline{\Z}\,.
	\end{equation*} Furthermore, this map is given by mapping a generator of $\tilde{K}^{-\lambda_q-2i}(N_q/L_q)$ to one, and hence uniquely determined, provided we have a canonical way to choose a generator of $\tilde{K}^{-\lambda_q-2i}(N_q/L_q)\cong \tilde{K}^{-\lambda_q-2i}(\overline{T^u_qM})$, where the isomorphism is natural and described in Remark \ref{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets}. 
	
	Hence, if $T^u_qM$ is oriented, Corollary \ref{cor:orient-vectorspaceOrientationInducesKtheoreticOrientation} gives us just what we need, making the map canonical. The generator is then given by pulling back the generator of $K^{-\ind_p}(\overline{T^u_qM})$ along the map of pairs:
	\begin{align*}
	 \kappa_q\coloneq  \restr{\diff \phi^{-1}}{0}\circ R_0\circ s\circ pr_u \circ \phi	\,:\,(\phi^{-1}( D^{s }_\epsilon\times  D^{u }_\epsilon),\phi^{-1}( D^{ s}_\epsilon\times \partial D^{u}_\epsilon))  \to (\overline{T^u_qM},\{\infty\})
	\end{align*} where $pr_u$ is the projection onto the unstable coordinates and $s$ is the stretching $x\mapsto x\cdot (\frac{1}{\epsilon}-\norm{x}-\frac{1}{\epsilon})$ with $\restr{\diff s}{0}=\frac{1}{\epsilon}\Idm$. We call such a generator the \textbf{standart generator at $q$}.
\end{remark}



	
\begin{example}[A Map of K-Groups]
	Let $f:M\to \R$ be a Morse-Smale function on a smooth compact Riemannian manifold $(M,g)$. Let $q\in \Crit_k(f)$ and $p\in \Crit_{k-1}(f)$. Assume for a moment we already have a regular index pair $(N_2,N_0)$ of the isolated compact subset $S=W(p\to q)\cup\{p,q\}$. 
	For a $c\in (f(p),f(q))$, we set $N_1=N_0\cup (N_2 \cap M^c)$, where $M^c$ is the sublevel set. Clearly now $(N_2,N_1)$ is a regular index pair of $q$ and $(N_1,N_0)$ is a regular index pair of $p$.
	
	Furthermore, we have the triple connecting homomorphism $\delta_{triple}^{p,q}: K^{-\ind_p}(N_1,N_0) \to K^{-\ind_p+1}(N_2,N_1)$, which we can compose with the Bott periodicity map $b\coloneq \beta \boxtimes (-)$. This lets us induce the map:
	\begin{equation*}
	 \Big(b\circ \delta_{triple}^{p,q}\Big)_{\#}:K^{-\ind_p}_{\#}(\mathcal{IP}_{\{p\}})\to K^{-\ind_q}_{\#}(\mathcal{IP}_{\{q\}})
    \end{equation*}
	Now the disjoint union of the index pairs $(N_q,L_q)$ for all $q\in \Crit_{k}(f)$ is an index pair of $\Crit_{k}(f)$ itself. The collapsing maps $c_p:\bigvee _{p\in \Crit_{k-1}}\big(N_p/L_p\big)\to \big(N_p /L_p\big)$ together with the inclusions $i_q:\big(N_q/L_q\big)\to \bigvee_{q \in \Crit_{k}}(N_q/L_q)$ let us define the map 
\begin{equation*}
	\Delta^{k-1} \coloneq \Big(\sum_{q\in\Crit_{k}}\sum_{p\in \Crit_{k-1}} c_p\circ \big(b\circ \delta_{triple}^{p,q}\circ i_q\big)\Big)_{\#}: K^{-(k-1)}_{\#}(\mathcal{IP}_{k-1}) \to K^{-k}_{\#}(\mathcal{IP}_{k} )\,.
\end{equation*} 

Due to the naturality of the triple connecting homomorphism, the inclusion and the collapsing, any list of topological triples $(X,Y,Z)$ such that $(X,Y)$ is a regular index pair of $\{q\}$, $(Y,Z)$ of $\{p\}$ and $(X,Z)$ of $W(q\to p)\cup \{q\}\cup \{p\}$ for all $q\in \Crit_k(f),p\in \Crit_{k-1}(f)$ gives rise to the same morphism of canonical isomorphism classes.
\end{example}

\begin{lemma}\label{lem:morseFilt-flowTimeAndProjectionToLevelSets}
	Let $\ind_q(f)\geq1$.
	\begin{enumerate}[(a)]
		\item For $x\in D_q\setminus \{q\}$ there is a unique $\tau(x)\geq 0$ with $f(\phi_{\tau(x)}(x))=c$. The resulting map $\tau:D_q\setminus \{q\}\to [0,\infty)$ is smooth, $\tau^{-1}(0)=S(q\to)$, $\tau(\phi_s(x))=\tau(x)-s$, $\restr{\diff \tau}{x}(-\grad_x(f))=-1$ and $\tau(x)\to \infty$ for $x\to q$. We set $\tau(q)=\infty$. 
		\item The map $\rho: D_q\setminus \{q\}\to S(q\to),~\rho(x)\coloneq \phi_{\tau(x)}(x)$ is a smooth retraction with $\rho\circ \phi_s=\rho$. In particular, $\restr{\diff \rho}{x}(\grad_x(f))=0$ and for $y\in S(q\to),~w\in T_y(S(q\to))$ and $s\geq 0$:
		 \begin{equation*}
			\diff \rho \big(\restr{\diff \phi_{-s}}{y}w\big)=w,\quad \diff \tau \big(\restr{\diff \phi_{-s}}{y}w\big)=0\,.
		\end{equation*}
		\item $S(q\to)$ is a closed submanifold of $W(q\to)$ of dimension $\ind_q-1$. There are homeomorphisms $g_q:S^{\ind_q} \to D_q /S(q\to) $ with $g_q(S)=[S(q\to)]$ and a constant $T_q\geq 0$ such that, with the open neighbourhood 
		\begin{equation*}
			W_q\coloneq \{x\in D_q \mid \tau(x)>T_q\}\ni q
		\end{equation*} of $q$, the map $g_q$ restricts to an orientation preserving diffeomorphism from $U_q\coloneq g_q^{-1}(W_q)$ onto $W_q$. 
	\end{enumerate}
\end{lemma}\todo{Bild}

\begin{proof}
(a) The function $f$ strictly decreases along flow lines, which approach critical points. As there is no critical point in $f^{-1}[c,b)$ our flow line needs to pass through $f^{-1}(c)$ giving us a $\tau(x)$, where uniqueness follows again from strict monotonicity. This uniqueness lets us deduce that $\tau(\phi_s(x))=\tau(x)-s$. 
The smoothness follows from the implicit function theorem applied to $G(x,t)\coloneq f(\phi_t(x))-c$, since $\partial_t G(x,t)=-\norm{\grad_{\phi_t(x)}(f)}^2\neq 0$. (compare Proposition \ref{prop:vb-ffDecreasesAlongGradientFlow} for the calculation) Differentiating $\tau(\phi_s(x))=\tau(x)-s$ in $s=0$ gives $\diff \tau (-\grad(f))=-1$. Finally assume that $x_n\to q$ whilst $\tau(x_n)\leq T$ for some $T>0$. Then we can pass to a subsequence such that $\tau(x_n)\to t$ for $t\leq T$. In this subsequence we then have that $\lim_{n\to \infty}f(\phi_{\tau(x_n)}(x_n))=f(\phi_t(q))=b$ contradicting $f\big(\phi_{\tau(x_n)}(x_n)\big)=c<b$ for all $n\in \N$. 

(b) By (a) we have that $\rho\circ \phi_s=\rho$, and differentiating it in $s$ gives us $\diff \rho (\grad(f))=0$. The last two identities are the differentials of $\rho\circ \phi_{-s}=\id$ and $\tau \circ \phi_{-s} =\mathrm{const}_s$ on $S(q\to)$.

(c) $\grad(f)$ is tangent to $W(q\to)$ (meaning that it is contained in the tangent bundle of $W(q\to)$ as a subbundle of $TM$) and non-zero away from $q$. Hence, $c$ is a regular value of $\restr{f}{W(q\to)}$, ($\diff \restr{f}{W(q\to)} (\grad(f))$ is nonzero outside of critical points) hence $S(q\to)$ is a submanifold of $W(q\to)$.

Let $\psi=(\psi_s,\psi_u)$ be a refined Morse chart around $q$ such that the unstable coordinates correspond to a positively oriented basis of $T^u_qM=T_qW(q\to)$. 
Here, $\psi_u$ denotes $\pi_u\circ \psi$ where $\pi_u$ projects to the unstable coordinates (and $\pi_s$ is defined analogously). 
By Lemma \ref{lem:morse-localUnStableManifoldTheorem} the unstable manifold is, in these coordinates the graph of a smooth function $g$ with $g(0)=0$, that is Lipschitz with Lipschitz constant less than one. (This $g$ does not denote the Riemannian metric, which doesnt appear by name in this proof.)
Hence, $\psi_u$ restricts to a diffeomorphism from a neighbourhood $O$ of $q$ in $W(q\to)$ onto an open ball around $0$ and $\norm{\psi(x)}\leq \sqrt{2}\norm{\psi_u(x)}$ on $O$ 
(The latter is deduced from the lipschitz constant of $g$  being less than or equal to one). Write $u(t)\coloneq \psi_u(\phi_t(x))$ for $x\in O$ (and $t$ small enough such that $\phi_t(x)\in O$) and denote pullback of the gradient via the chart $\psi$ by $X\coloneq -q_{\psi}\circ \grad(f)\circ \psi^{-1}$, where $q_\psi$ denotes the coordinate map on the tangent bundle induced by the frame $\partial_i$ (which comes from the morse chart. Compare Definition \ref{def:morse-jacobeanOfTheGradient}). 
By Lemma \ref{lem:morse-jabobianOfTheGradient} together with Remark \ref{rem:morse-jacobianOfTheFlowInRefinedMorsechart} we know that $\restr{\diff X}{0}=J(\grad(f))_{\psi}(0)$ is diagonal. Now we define 
\begin{equation*}
	Y:\R^{\ind_q}\supseteq B \to \R^{\ind_q}\,,\quad v\mapsto Y(v)\coloneq \pi_u \circ (X_{g(v),v})
\end{equation*} and Taylor-expand said function in the origin. By the chain rule and splitting of the tanget space at the origin we get that $Y(v)=\Lambda v+R(v)$ with $\Lambda=\diag(2\mu_1,\dots ,2\mu_{\ind_q})$ and $\norm{R(v)}=o(\norm{v})$.
 Now \begin{equation*}
	\frac{\diff}{\diff t} (u(t))=\pi_u \circ  q_\psi\circ \frac{\diff}{\diff t} \phi_t(x)=\pi_u \circ q_\psi \circ -\grad_{\phi_t(x)}(f)=Y(u(t))\,.
\end{equation*} Thus, with Cauchy-Schwarz we can calculate that for $v$ close enough to zero:

\begin{equation*}
	\frac{\diff}{\diff t}\frac12\norm{u(t)}^2=\big\langle u(t),\Lambda u(t)\big\rangle\underbrace{+\big\langle u(t),R(u(t))\big\rangle}_{\geq -\norm{u(t)}\cdot \underbrace{\norm{R(u(t))}}_{\leq \mu_{min}\norm{u(t)}}}\geq 2\mu_{\min}\norm{u(t)}^2-\mu_{\min}\norm{u(t)}^2>0\,.
\end{equation*} 
Now choose $r$ small enough such that the estimate holds and the ball $B_r(0)\subseteq \psi_u(O)$, and $f>c$ on $E_r\coloneq (\restr{\psi_u}{O})^{-1}(B_r(0))$. Then:
\begin{enumerate}[(i)]
	\item For $y\in \partial E_r$ and $t>0$ we have $\phi_{-t}(y)\in \inti(E_r)$, since $\psi_u(\phi_{-t}(y))$ strictly decreases in $t$, by the estimate above.
	\item Every flow line in  $D_q\setminus \{q\}$ meets $\partial E_r$ exactly once: It meets the boundary at least once, as it starts in $q\in \inti(E_r)$ and ends in $S(q\to)$ which is disjoint from $E_r$. If a flow line met the boundary at multiple times, there would be a time $t_0$ where the flow line reenters $E_r$, which forces the $t$ differential of said flow line in the chart $\psi_u$ to be less than or equal to $0$ (the norm needs to be non-increasing), which contradicts the estimate above.
\end{enumerate}

Now we come to the homeomorphism: By (ii), the map 
\begin{equation*}
	\Theta:\{(y,s)\mid y\in \partial E_r, 0\leq s \leq \tau(y)\}\to D_q\setminus \inti(E_r)\,,\quad \Theta(y,s)\coloneq \phi_s(y)
\end{equation*}  is a continuous bijection from a compact set into a Hausdorff space and hence a homeomorphism. Now we define 
\begin{align*}
	G:D_q\to D_1\subseteq \R^{\ind_q}\,, \quad 
	\begin{cases}
		G(x)\coloneq \frac{1}{2r}\psi_u(x) & \text{ for }x\in E_r,\\
		G(\phi_s(y))\coloneq \left(\frac12 + \frac{s}{2 \tau(y)}\right) \frac{\psi_u(y)}{r}&\text{ for } y\in \partial E_r,~ 0\leq s\leq \tau(y)\,.
	\end{cases}
\end{align*}
Both formulas define something continuous (even smooth on their domain) and agree on $\partial E_r$. Hence $G$ is continuous. Furthermore, $G$ is bijective and the domain is compact, whilst the image is Hausdorff, giving us a homeomorphism that maps $\partial D_q$ onto $\partial D_1$. Now let $\overline{\pi_S}:S^{\ind_q}\to \overline{\R^{\ind_q}}$ denote the function from Definition \ref{def:orient-fromRelaSpaceToTheSphere} and with this define 
\begin{equation*}
	g_q^{-1}\coloneq \overline{\pi_S^{-1}}(\frac{G}{1-\norm{G}}): D_q\big/S(q\to) \to S^{\ind_q}
	\quad \text{ where $\frac{c}{0}\coloneq \infty$. for $c\in \R$}
\end{equation*} 

Now only the statement about the orientation preservation of $g_q$ at a neighbourhood of $q$ is missing: Define $T_q\coloneq \max_{\partial E_r}\tau$. If $\tau(x)>T_q$ and $x\neq q$, let $y$ be the point, where the flow line through $x$ passes through the boundary $\partial E_r$. Then $x=\phi_{-t}(y)$ with $t=\tau(x)-\tau(y)>0$ and thereby $x\in \inti(E_r)$ by (i). Hence, $W_q\subseteq \inti (E_r)$ and in said interior of $E_r$ $G$ is a diffeomorphism that is orientation preserving at $q$ by the definition of $\psi_u$ and by Lemma \ref{lem:orient-computingSignInCharts} we know that the sign of the determinant of a diffeomorphism of oriented manifolds is computed by a smooth procedure, and thereby it is locally constant, making $G$ orientation preserving on the whole $\inti(E_r)$. The map $\overline{\pi_S^{-1}}$ is orientation preserving: This follows from the orientation of $S^{\ind_q}$ by Remark \ref{conv:orient-orientationOfTheSphere}. Furthermore, the stretching $v\mapsto \frac{v}{1-\norm{v}}$ is orientation preserving where $D_1$ is oriented as a subspace of $\R^{\ind_q}$(this can be shown by looking at the sign of the determinant of its Jacobian). Hence the whole composition $g_q^{-1}$ is orientation preserving and thereby also $g_q$ is.
\end{proof}

\begin{lemma}\label{lem:morseFiltr-orientationClassOfADisk}
	Let $n\geq 1$ and let $(\Sigma,\partial \Sigma)$ be a compact pair ($\partial \Sigma$ denotes a closed subspace of $\Sigma$, and not some kind of boundary. In applications of the Lemma, it will be a boundary) such that $(\Sigma \setminus \partial \Sigma)$ is an oriented $C^1$-manifold of dimension $n$. Let $g:S^n\to \Sigma \big/ \partial \Sigma$ be a homeomorphism with $g(S)=[\partial \Sigma]$ (where $S$ denotes the south pole), and let $U\subseteq S^n\setminus S$ be open such that $\restr{g}{U}$ is an orientation preserving $C^1$-diffeomorphism onto an open subset of $\Sigma \setminus \partial \Sigma$. Define
	\begin{equation*}
		o_{\Sigma}\coloneq (g^{-1})^*(\beta_n)\in\tilde{K}^{-n}(\Sigma /\partial \Sigma)= K^{-n}(\Sigma , \partial \Sigma)\,,
	\end{equation*} with $\beta_n$ being the canonical generator defined in Definition \ref{def:ktheo-higherBottGenerators}. Let $V$ be an oriented $n$-dimensional vector space with K-theoretic orientation $\beta_V$ and $h:\Sigma / \partial \Sigma \to \overline{V}$ be a pointed map such that for some $z\in V$ we have that $h^{-1}(z)=\{y\}$ with $y\in g(U)$, $h$ is $C^1$ near $y$ and $\restr{\diff h}{y}$ is invertible. Then
	\begin{equation*}
		h^*(\beta_V)=\sign\det\left(\restr{\diff h}{y}\right)\cdot o_{\Sigma}
	\end{equation*} 
\end{lemma}
\begin{proof}
	Let $\kappa_V:V\to \R^n$ be the coordinate map of a positively oriented basis, meaning that $\left(\overline{\pi_S^{-1}}\circ \overline{\kappa_V}\right)^*(\beta_n)=\beta_V$ (compare Corollary \ref{cor:orient-vectorspaceOrientationInducesKtheoreticOrientation}). Then $\xi\coloneq \overline{\pi_S^{-1}}\circ \overline{\kappa_V}\circ h \circ g:S^n\to S^n$ satisfies:
	\begin{enumerate}[(i)]
		\item $\xi^{-1}(\pi_S^{-1}\circ \kappa_V(z))=\{g^{-1}(y)\}$,
		\item $\xi$ is $C^1$ near $g^{-1}(y)\in U$, 
		\item by the chain rule, Lemma \ref{lem:orient-computingSignInCharts} and since $\pi_S^{-1}$, $\kappa_V$ and $g$ are orientation preserving we have:
		\begin{equation*}
			\sign \det(\restr{\diff \xi}{g^{-1}(y)})=\sign \det (\restr{\diff h}{y})\,.
		\end{equation*}
	\end{enumerate}
	Hence by the Single Regular Preimage Lemma \ref{lem:orient-singleRegularPreimage} applied to $\xi^*$ we can calculate
	\begin{equation*}
		g^*h^*(\beta_V)=\xi^*(\beta_n)=\sign \det \left(\restr{\diff h}{y}\right)\cdot \beta_n\,.
	\end{equation*} Finally we can apply $(g^{-1})^*$ from the left to finish the proof.
\end{proof}



\begin{remark}
	We can apply the Lemma above to the pair $(D_q,S(q\to))$ together with $g=g_q$, and $U=U_q$ from Lemma \ref{lem:morseFilt-flowTimeAndProjectionToLevelSets} (c). 
\end{remark}

\begin{lemma}[Oriented Tubular Embedding]\label{lem:filt-orientedTubularEmbedding}
	Let $B_p\coloneq W(\to p)\cap M^c$ and let $J:D_{\epsilon}(\mathcal{N})\to N_p$ be the tubular embedding from Lemma \ref{lem:morseFilt-npIsTubular}, where $\mathcal{N}\to B_p$ denotes the normal bundle. Now we can orient the normal bundle via the bundle isomorphism $l:\mathcal{N}\to Q^p$  where $Q^p$ denotes the quotient bundle from Definition \ref{def:morseFilt-setupCommutingDiff}(the map is given by including the normal bundle into the tangent bundle and projecting then onto the quotient bundle). Then $J$ can be chosen to be fibrewise orientation preserving, meaning that with the linear identification $R_0:\mathcal{N}_x\to T_0\mathcal{N}_x$ and the projection bundle map $pr: \restr{TM}{W(\to p)\cap M^c}\to Q^p$ we have that for all $x$ the map
	\begin{equation*}
		pr\circ \restr{\diff  J_x}{0}\circ R_0:\mathcal{N}_x\to Q_x^p
	\end{equation*} is orientation preserving.
\end{lemma}
\begin{proof}
	This follows directly from $pr\circ \restr{\diff  J_x}{0}\circ R_0:\mathcal{N}_x\to Q_x^p$ being a bundle isomorphism. Hence it either is orientation preserving in all fibers, or reversing in all fibers (due to the connected base space). If the latter happens, we redefine $J$ by composing it with a map $\rho$ that reflects, say the first coordinate: Here we choose a global orthonormal trivialisation of $\mathcal{N}$ (which exists since the base space $B_p$ is contractible) such that $\rho$ maps the domain of the tubular embedding isomorphically onto itself. 
\end{proof}
\begin{remark}
	For a fiber $V_j$ of the tubular neighbourhood from Lemma \ref{lem:morseFilt-npIsTubular} with center $x_j$, we can apply the Lemma \ref{lem:morseFiltr-orientationClassOfADisk} with $U=S^n\setminus \{S\}$ and $g_j$ given by the oriented tubular embedding composed with the inverse of $v\mapsto \pi^{-1}_S\left(\frac{v}{1-\norm{v}}\right)$. Then $g_j(N)=x_j$ for the north pole $N$ and $h_j\coloneq \overline{\kappa_V^{-1}}\circ \overline{\pi_S}\circ g_j^{-1}$ is a map where we can apply the Lemma \ref{lem:morseFiltr-orientationClassOfADisk} above with $h^{-1}(0)=\{x_j\}$ to conclude that $h^*_j(\beta_V)=o_{V_j}$.(It is easy to see that $h_j$ is locally a diffeomorphism and $h$ as invertable differential in the origin.)
\end{remark}

\begin{lemma}\label{lem:morseFilt-canonicalGeneratorsOnTransversalDisks}
	Let $q\in \Crit(f)$ with $n\coloneq \ind(q)\geq 1$, let $(N,L)$ be an index pair of $\{q\}$ and let $\Sigma \subseteq N$ with $\partial \Sigma \subseteq L$ be as in Lemma \ref{lem:morseFiltr-orientationClassOfADisk}. Assume 
	\begin{enumerate}[(i)] 
		\item $\Sigma \cap W(\to p)=\{x\}$ with $x\in g(U)$ and $T_x\Sigma \oplus T_xW(\to p)=T_xM$, ($\Sigma$ is transversal)
		\item for every $s\geq 0$ there is a neighbourhood $U_s$ of $x$ in $\Sigma$ with $\phi_{[0,s]}(U_s)\subseteq N\setminus L$.
	\end{enumerate}
	Orient $\Sigma\setminus \partial \Sigma$ such that the isomorphism $T_x\Sigma \to Q_x^p$ is orientation preserving. Then for the inclusion $i:(\Sigma,\partial \Sigma)\to (N,L)$
	\begin{equation*}
		i^*\left(\alpha_q^{(N,L)}\right)=o_{\Sigma}\,.
	\end{equation*}
\end{lemma}
\begin{remark}
	The two cases where we apply said lemma are: 
	\begin{enumerate}
		\item $\Sigma=D_q,~x=q,~(N,L)=(C,B)$ (from Lemma \ref{lem:morseFilt-flowTimeAndProjectionToLevelSets}) where $Q^q_q=T_qM/T^s_qM\cong T^u_qM$, so $o_{D_q}$ is taken with respect to the orientation of $T^u_qM$. 
		\item $\Sigma=V_j$ $x=x_j$ and $(N,L)=(B,A)$.
	\end{enumerate}
\end{remark}
\begin{proof}[Proof of Lemma \ref{lem:morseFilt-canonicalGeneratorsOnTransversalDisks}]
	We prove the Lemma in three steps. First we restriced the statement to the easier case, where $\Sigma\subseteq W(q\to)$. Then we define a new chart such that the stable manifold is not just a graph of a function, but a linear subspace. This makes sure that the projection onto the unstable coordinates collapses the stable part to a point which finally in step 3 generalizes the statement for arbitrary $\Sigma$:

	\textit{Step 1}. Let $\chi=(\chi_s,\chi_u)$ be a chart around $q$ such that 
	
	\begin{enumerate}
		\item $W(q\to)$ is the graph of function over the unstable coordinates,
		\item the box $(N_\chi,L_\chi)$ defined as 
	\begin{equation*}
		(\chi^{-1}(D^s_{\epsilon}\times D^u_{\epsilon}), \chi^{-1}(D^s_{\epsilon}\times \partial D^u_{\epsilon}))
	\end{equation*} is a regular index pair of $\{q\}$.
		\item $\epsilon$ is small enough such that $W(q\to)\cap N_\chi$ is the graph of the function from above.
		\item $\diff \chi_q=\diff \psi_q$ where $\psi$ is a refined Morse chart.
	\end{enumerate}
	 The existence of such a chart follows from the local unstable manifold theorem \ref{lem:morse-localUnStableManifoldTheorem}.Let  $\kappa_{\chi}$ be the map defined as 
	 \begin{equation*}
		\kappa_{\chi} \coloneq  \restr{\diff \chi^{-1}}{0}\circ R_0\circ s\circ pr_u\circ \chi
	 \end{equation*} similar to the construction of $\kappa_q$ in Remark \ref{rem:morseFiltr-morphismToCanonicalIntegers}. Let $\Sigma$ be a disk contained in $W(q\to)$ and hence $\Sigma \cap W(\to q)=q$. Near $q$ and for large $t>0$ the homotopy equivalence $c:(N,L) \to (N_\chi,L_\chi)$ is given by $y\mapsto[\phi_{3t}(y)]$(Compare Theorem \ref{thm:morseFiltr-homotopyEquivalenceOfIndexPairs} and notice that $q$ is in the interior of both $(N\setminus L)$ and $(N_\chi\setminus L_\chi$)). Define $i:(\Sigma,\partial \Sigma) \to (N,L)$ to be the inclusion. 
	 Then consider the function 
	 \begin{equation*}
	 h\coloneq \underbrace{\restr{\diff \chi^{-1}}{0}\circ R_0\circ s\circ pr_u\circ \chi}_{\kappa_\chi}\circ c\circ i: (\Sigma /\partial \Sigma)\to (\overline{T^u_qM})\,
	 \end{equation*} If $h(y)=0$, then $y$ cannot be $[\partial \Sigma]$ as that point end at ionfinity. 
	 By definition of $\kappa_{\chi}$ we know that $\chi\circ c\circ i(y)\in D^s_\epsilon \times \{0\}$. 
	 And since $y\in \Sigma\subseteq W(q\to)$, and $c$ restricts to a map from $W(q\to)$ to $W(q\to)$ (restrictied to the right domain, and image), we know that $\chi\circ c \circ i(y)\in \chi(W(q\to))\cap D^s_\epsilon \times \{0\}$. 
	 Finally, we assumed that $W(q\to)$ was the graph of the unstable coordinates in $\chi$ and hence, there is only one $x\in D^s_\epsilon \times \{0\}\cap \chi(W(q\to))$, which is $(0,0)$. Hence $\phi_{3t}(y)=q$ which is a fixpoint of the flow and thereby, $y=q$ and $0$ has a unique preimage under $h$. Furthermore, $\restr{\diff h}{q}$ is an endomorphism on $T^u_qM$. We can calculate its determinant by calculating the jacobean determinants of $c\circ i$ and $\kappa_\chi$ in $q$ (resp. in the origin) in a refined Morse chart seperately: 
	 \begin{enumerate}
		\item The function $\restr{\diff (c\circ i)}{q}$ (restricted in its image to $T^u_qM$) has positive determinant which follows from the flow in a refined morse chart in a critical point restricted to the unstable space has a differential that is diagonal with positive entries. This was calculated in Remark \ref{rem:morse-jacobianOfTheFlowInRefinedMorsechart}.
		\item The function $\kappa_\chi$ can be differentiated in $q$:
		\begin{equation*}
			\diff \kappa_{\chi}=\diff \Big(\restr{\diff \chi^{-1}}{0}\circ R_0\circ s\circ pr_u\circ \chi \Big)\,
		\end{equation*}
		 where $\iota\coloneq\restr{\diff\chi^{-1}}{0}\circ R_0:\{0\}\times\R^{n}\to T^u_qM$ is a linear isomorphism (since $\restr{\diff\chi}{q}$ maps $T^u_qM$ onto the unstable coordinate plane) and $s$ is the radial stretching with $\restr{\diff s}{0}=\lambda \id$, where $\lambda>0$. On $T^u_qM$ the map $\mathrm{pr}_u\circ\restr{\diff\chi}{q}$ is the inverse of $\iota$, so, identifying $T_0T^u_qM$ with $T^u_qM$,
\begin{equation*}
	\restr{\diff h}{q}=\lambda\,\restr{\diff\phi_{3t}}{q}\Big|_{T^u_qM}\in\operatorname{End}(T^u_qM)\,.
\end{equation*}
Its determinant is again positive, as the flow restricted to the unstable tangent space of a critical point is diagonal with positive entries.
	 \end{enumerate}
Hence, $\restr{\diff h}{q}$ has positive determinant and Lemma \ref{lem:morseFiltr-orientationClassOfADisk} gives us the equality
\begin{equation*}
	i^*\circ c^*\circ \kappa_{\chi}^*(\beta_{T^u_qM})=h^*(\beta_{T^u_qM})=o_{\Sigma}\,.
\end{equation*} This concludes the proof for the special case, where $\Sigma$ is contained in $(q\to)$.

So far we have seen that for any chart $\chi$ that satisfies the properties from Step 1 there is the equality above. If we choose $\chi$ to be a refined morse chart, then $\kappa_{\chi}^*(\beta_{T^u_qM})=\alpha_q^{(N_\chi,L_chi)}$ is a canonical generator and a member of $(\alpha_q)_{\#}$.

\textit{Step 2} We know by the local unstable manifold theorem, that for our refined morse chart $\psi$ there exists a smooth $g_s$ such that $\psi \big(W(\to q)\big)$ is the graph of $g_s$ with $g_s(0)=0$ and $\restr{\diff g_s }{0}=0$. The chart $\tilde{\psi}\coloneq (\psi_s,\psi_u-g_s\circ \psi_s)$ satisfies $\restr{\diff \tilde{\psi}}{q}=\restr{\diff \psi}{q}$, $W(\to q)=\tilde{\psi}^{-1}(\R^{m-n}\times \{0\})$ (locally), $W(q\to)$ is still the graph of a function over the unstable coordinates and by Remark \ref{rem:morseFiltr-indexPairForDifferentChart} ,for small $\epsilon$, its box $(N_{\tilde{\psi}},L_{\tilde{\psi}})$ is an index pair. We claim now, that $\kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})$ is a member of the canonical generator $(\alpha_q)_{\#}$.
Let $E\coloneq W(q\to)\cap N_\psi$, be the unstable disk in the box of $\psi$. In the chart $\psi$, $E$ is the graph of a map 
\begin{equation*}
	e:D^u_\epsilon\to D^s_\epsilon 
\end{equation*} \todo{Bild einfügen}  Now we define the deformation homotopy for $t\in[0,1]$
\begin{equation*}
	H_t(l,x)\coloneq \big((1-t)l+te(x),x\big)
\end{equation*} that shows that $(E,\partial E)$ is a strong deformation retract of $(N_\psi,L_\psi)$ onto and hence induces an isomorphism $K^{-n}(N_\psi,L_\psi)\to K^{-n}(E,\partial E)$.
Now we apply Step 1 to the two charts $\chi=\psi$ and $\chi=\tilde{\psi}$ with $\Sigma=E$, and $(N,L)=(N_\psi,L_\psi)$. This tells us that pulling back the canonical generator along both paths in the diagram 
\begin{center}
\begin{tikzcd}
                                      &                                                           & {(N_{\tilde{\psi}},L_{\tilde{\psi}})} \arrow[rd, "\kappa_{\tilde{\psi}}"] &                                  \\
{(E,\partial E)} \arrow[r, "i", hook] & {(N_\psi,L_\psi)} \arrow[ru, "c_1"] \arrow[rd, "c_2\simeq \id"] &                                                                           & {(\overline{T^u_qM},\{\infty\})} \\
                                      &                                                           & {(N_\psi,L_\psi)} \arrow[ru, "\kappa_{\psi}=\kappa_q"']                   &                                 
\end{tikzcd}
\end{center} induce the same $o_E$. Now $c_2\simeq \id$ and the inclusion $i$ induces an isomorphism by the deformation above. Hence, 
\begin{equation*}
	c_1^*\circ \kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})=\kappa_q^*(\beta_{T^u_qM})=\alpha_q^{(N_\psi,L_\psi)}\,.
\end{equation*}

But the canonical generators are compatible with the maps $c$ and $c_1^*$ is an isomorphism, we know that $\alpha_q^{(N_{\tilde{\psi}},L_{\tilde{\psi}})}$ is the unique element with $c_1^*(\alpha_q^{(N_{\tilde{\psi}},L_{\tilde{\psi}})})=\alpha_q^{(N_{{\psi}},L_{{\psi}})}$. Thereby 
\begin{equation*}
	\kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})=\alpha_q^{(N_{\tilde{\psi}},L_{\tilde{\psi}})}\,.
\end{equation*}

\textit{Step 3} Now we assume $\Sigma$ to just satisfy the Lemmas conditions, use again $\tilde{\psi}$ and define the map on the quotients
 \begin{equation*}
	h\coloneq \underbrace{\restr{\diff \tilde{\psi}^{-1}}{0}\circ R_0\circ s\circ pr_u\circ \tilde{\psi}}_{\kappa_{\tilde{\psi}}}\circ c\circ i
 \end{equation*} with $c:(N,L)\to (N_{\tilde{\psi}},L_{\tilde{\psi}})$ being the canonical homotopy equivalence as in Step 1. If $h(y)=0$, then similar to before, we argue, that $h(y)=0$ means that $\tilde{\psi}\circ c\circ i(y)\in D^s_\epsilon\times \{0\}$ which in turn means that $c(y)=\phi_{3t}(y)\in \tilde{\psi}^{-1}(D^s_\epsilon\times\{0\})$. By definition of $\tilde{\psi}$ we made sure that $W(\to q)\supseteq \tilde{\psi}^{-1}(D^s_\epsilon\times\{0\})$. 
 Thereby, our $y\in \Sigma\cap W(\to q)=\{x\}$ by requirements on $\Sigma$. By (ii) and since $\phi_{[t,3t]}(x)$ are close to $q$ for large $t$ our map $h$ equals $h=\kappa_{\tilde{\psi}}\circ \phi_{3t}$ near $x$. 
 Due to the projection $pr_u$, $\kappa_{\tilde{\psi}}$ is constant on $W(\to q)$ (for a small enough box), and the flow $\phi_{3t}$ preserves $W(\to q)$. Thereby, the differential factors through the kernel of $\diff( pr_u \circ \tilde{\psi})_{\phi_{3t}(x)}=T_{\phi_{3t}(x)}W(\to q)$. And since $\Sigma$ intersects $W(\to q)$ transversally  we factor through the composition
 \begin{equation*}
	\restr{\diff h}{x}:T_x\Sigma \, \xrightarrow{\cong}\, Q^q_x \, \xrightarrow{[\diff \phi_{3t}]} \, Q^q_{\phi_{3t}(x)} \, \xrightarrow{[\diff \kappa_{\tilde{\psi}}]}  \, T^u_qM \,.
 \end{equation*} Now the first isomorphism is orientation preserving, by the choice of orientation on $\Sigma$. 
 The sign of $\det([\diff \phi_s])$ (as a map between oriented vector spaces) is continuous in $s$ (as it is calculated by the sign of the determinant of a map that depends continuously on $s$), where the bases are induced by a positive global frame (existence follows from contractibility of the base space) and hence they depend continuous on the base point. For $s=0$ the sign is positive and thereby, the second map is orientation preserving.
  Finally, the sign of $[\diff \kappa_{\tilde{\psi}}]_y$ (as an isomorphism between oriented vector spaces for all $y$) is also continuous in $y$ (on the connected local stable disk) and equals $1$ at $y=q$, as 
 \begin{equation*}
	[\diff \kappa_{\tilde{\psi}}]_q:Q^q_q\cong T^u_qM \to T^u_qM
 \end{equation*} is given by the identity together with the differential of the stretching of the origin. Thereby, the sign of $\det \restr{\diff h}{x}=1$. Lemma \ref{lem:morseFiltr-orientationClassOfADisk} gives us 
 \begin{equation*}
	o_\Sigma=h^*(\beta_{T^u_qM})=i^*\circ c^* \circ \kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})
 \end{equation*}
 Then, Step 2 gives us that $\kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})=\alpha_q^{(N_{\tilde{psi}},L_{\tilde{psi}})}\in (\alpha_q)_{\#}$. Thereby $c^* \circ \kappa_{\tilde{\psi}}^*(\beta_{T^u_qM})=\alpha_q^{(N,L)}$ by definition. This finally concludes the proof.

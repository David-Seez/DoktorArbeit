\subsection{Conley index}
\begin{definition}[Compact Invariant Isolated Subset] \label{def:morseFiltr-compactInvariantIsolatedSubset}
	For a flow $\phi_t:M\to M$ on a locally compact metric space, we call a subset $S\subseteq M$ an \textbf{invariant subset} if and only if $\phi_t(S)=S$ for all $t\in \R$. For any subset $N\subseteq M$ we define the \textbf{maximal invariant subset}
	\begin{align*}
		I(N)&=\{x\in N \mid \phi_t(x)\in N \forall t\in \R\}\\
		&=\bigcap_{t\in \R} \phi_t(N).
	\end{align*} A \textbf{compact invariant subset} $S$ is called \textbf{isolated} if a compact neighbourhood $N$ exists such that $I(N)=S$.
\end{definition}
\begin{definition}[Index Pairs] \label{def:morseFiltr-indexPairs}
	Let $S$ be an isolated compact invariant subset. A topological pair $(N,L)$ of compact subsets of $M$ where $L\subseteq N$ is called an \textbf{index pair of }S, if it satisfies the following:
	\begin{enumerate}
		\item $S=I(\overline{N\setminus L})\subseteq \inti({N\setminus L})$.
		\item  $x\in L$ and $\phi_{[0,t]}(x)\subseteq N$ implies that  $\phi_{[0,t]}(x)\subseteq L$. We call $L$ \textbf{positively invariant in } $N$. Here $\phi_{[0,t]}(x)\coloneq\{ \phi_{\tilde{t}}(x)|\tilde{t}\in [0,t]\}$.
		\item For all $x\in N$ such that a $t$ exists, with $\phi_t(x)\not\in N$, there exists a $t'$ with $\phi_{[0,t']}(x)\subseteq N$ and $\phi_{t'}(x)\in L$.
	\end{enumerate}
	This definition captures how the flow lines leave the invariant space $S$. Due to the third property we call $L$ the \textbf{exit set}.
\end{definition}
\begin{remark}\label{rem:morseFiltr-existenceOfIndexPairs}
	For any isolated compact invariant subset $S$ in a locally compact metric space $M$ with a flow $\phi: M\to M$ there exists an index pair $(N_1,N_0)$ of $S$. 
\end{remark}
\begin{proof}
	This is Theorem 4.3 in \cite{Salamon1985}.
\end{proof}

\begin{theorem}[Homotopy Equivalence of Index Pairs]\label{thm:morseFiltr-homotopyEquivalenceOfIndexPairs}
	If $S$ is an isolated compact invariant subset and $(N,L)$ and $(\tilde{N},\tilde{L})$ are two index pairs of $S$, then $N/L$ and $\tilde{N}/\tilde{L}$ are homotopy equivalent as pointed spaces. This homotopy equivalence is canonical, and transitive up to homotopy. 
\end{theorem}
\begin{proof}
	For a proof of the homotopy equivalence we refer to \cite{banyaga2004lectures}, Theorem 7.15. The constructed homotopy equivalence is given as follows: Choose $T\geq 0$ such that the following implications hold for $t\geq T$:
\begin{align*}
\phi_{[-t,t]}(x) \subseteq N\setminus L \quad &\Rightarrow \quad x \in \tilde{N}\setminus \tilde{L}\,, \\
\phi_{[-t,t]}(x) \subseteq  \tilde{N}\setminus \tilde{L} \quad &\Rightarrow \quad x \in N\setminus L\,, 
\end{align*} where $\phi_{[a,b]}(x) \coloneq \{\phi_t(x)\mid t\in [a,b]\}$.
Then the map:
\begin{align*}
h:N/L \times [T,\infty) & \to \tilde{N}/\tilde{L} \\
([x],t) & \mapsto 
\begin{cases}
[\phi_{3t}(x)] & \text{ if } \phi_{[0,2t]}(x)\subseteq N\setminus L \text{ and } \phi_{[t,3t]}(x)\subseteq \tilde{N}\setminus \tilde{L}\\
[\tilde{L}] & \text{ otherwise,}
\end{cases}
\end{align*}
is continuous and for any fixed $t$ this gives our homotopy equivalence. Obviously, such a map is canonical and transitive up to homotopy.
\end{proof}

\begin{definition}[Connected Simple System]\label{def:morseFiltr-connectedSimpleSystem}
	A \textbf{connected simple system} consists of a set $I_0$ of pointed spaces along with a set $I_m$ of homotopy classes of maps between these spaces such that 
		\begin{enumerate}[(i)]
			\item $\hom(X,Y)\coloneq \{[f]\in [X,Y]\mid[f]\in I_m\}$ consists of exactly one element for each pair of spaces $(X,Y)\in I_0^2$, where $[X,Y]$ denotes the set of homotopy classes of maps between $X$ and $Y$.
			\item If $X,Y,Z\in I_0,[f]\in \hom(X,Y),$ and $[g]\in \hom(Y,Z)$, then $[g\circ f]\in \hom(X,Z)$.
			\item $\hom(X,X)=\{[\id_X]\}$ for all $X\in I_0$. 
		\end{enumerate}
\end{definition}
\begin{remark}\label{rem:morseFiltr-connectedSimpleSystemSalamon}
	This definition comes from Salamon in \cite{Salamon1985}.
\end{remark}


\begin{remark}\label{rem:morseFiltr-indexPairConnectedSimpleSystem} Let $S$ be an isolated compact invariant subset of $M$ with respect to the Morse flow $\phi$. Then the collections
\begin{align*}
	I_0&\coloneq \{N  / L\mid (N,L)\text{ is an index pair of }S\}\\
	I_m&\coloneq \{[f] \mid f \text{ is a homotopy equivalence from Theorem \ref{thm:morseFiltr-homotopyEquivalenceOfIndexPairs}}\}\,,
\end{align*} make up a connected simple system. We denote this by $\mathcal{IP}_S$. If $S=\Crit_k(f)$ for some $k\in \N$, we denote $\mathcal{IP}_{\Crit_k(f)}=\mathcal{IP}_k$.
\end{remark}


\begin{definition}[Morphisms of Connected Simple Systems]\label{def:morseFiltr-morphismsOfConnectedSimpleSystems}
A \textbf{morphism} $\Phi:I\to J$ between two connected simple systems $I=(I_0,I_m)$ and $J=(J_0,J_m)$ is a set of homotopy classes of maps between spaces in $I_0$ and $J_0$ such that
\begin{enumerate}[(i)]
	\item For every $X\in I_0$ and $Y\in J_0$ the set $\Phi(X,Y)\coloneq \{[\phi]\in [X,Y]\mid[\phi]\in \Phi\}$ consists of exactly one element. 
	\item If $X,\bar{X}\in I_0$ and $Y,\bar{Y}\in J_0$ and if $[\phi]\in \Phi(X,Y)$, $[f]\in \hom(\bar{X},X)$, $[g]\in \hom(Y,\bar{Y})$, then $[g\circ \phi\circ f]\in \Phi(\bar{X},\bar{Y})$
\end{enumerate}
\end{definition}
\begin{remark}\label{rem:morseFiltr-morphismInducedBySingleMap}
	By the second property of a morphism between connected simple systems $I$ and $J$, we can induce such a morphism by a single map $\phi:X\to Y$ where $X\in I_0$ and $Y\in J_0$. 
\end{remark}


\begin{definition}[Canonical Isomorphism Class]\label{def:morseFiltr-canonicalIsomorphismClass}
	A \textbf{canonical isomorphism class} of Abelian groups (or other algebraic structures) is a set of groups $R_0$ together with a set of group homomorphisms $R_m$ such that
	\begin{enumerate}[(i)]
		\item For $R,L \in R_0$ the set $\hom(R,L)\coloneq \{f:R\to L: f \in R_m\}$ consists of exactly one element. 
		\item If $R,L,S\in R_0$ and $f\in \hom(R,L)$ and $g\in \hom(L,S)$, then $g\circ f\in \hom(L,S)\,$.
		\item $\hom(R,R)=\{\id_R\}$
	\end{enumerate}
	A \textbf{morphism} between two canonical isomorphism classes of Abelian groups $\Psi: (R_0,R_m)\to (S_0,S_m)$ is a collection of homomorphisms such that 
	\begin{enumerate}[(i)]
		\item For any $R\in R_0$ and $S\in S_0$ the set $\Psi(R,S)\coloneq \{f:R\to S: f\in \Psi\}$ contains exactly one element.
		\item If $R,\bar{R}\in R_0$ and $S,\bar{S}\in S_0$, $f\in \hom(\bar{R},R)$, $g\in \hom(S,\bar{S})$ and $\psi\in \Psi(R,S)$, then $g\circ \psi\circ f\in \Psi(\bar{R},\bar{S})\,$.
	\end{enumerate}
\end{definition}
\begin{remark}
	Again the second property lets us induce a morphism between two canonical isommorphism classes of Abelian Groups $\Psi: (R_0,R_m)\to (S_0,S_m)$ by a single isomorphism 
	\begin{equation*}
		\psi: R\to S
	\end{equation*} for $R\in R_0$ and $S\in S_0$. We denote the induced morphism by $\psi_{\#}\coloneq \Psi$. 
\end{remark}

\begin{example}
	Any pointed topological space $X$ defines a simple connected system $(\{X\},{[\id_x]})$. Similar, any Abelian group $G$ defines a caninical isomorphism class of Abelian groups $(\{G\},\{\id_R\})$.
	Let $\phi :S \to L$ be a second homomorphism that induces a morphism of canonical isomorphism classes. It is then obvious that 
	\begin{equation*}
		(\phi\circ \psi)_{\#}=\phi_{\#}\circ \psi_{\#}\,.
	\end{equation*}
\end{example}

\begin{definition}\label{def:morseFiltr-systemCategories}
We denote the category of connected simple systems with their morphisms by $\CSS$ and similar we call the category of canonical isomorphism classes of Abelian groups by $\CIAB$, respectively, the category for rings by $\CIRing$.
\end{definition}
\begin{definition}
	Let $a\in A$ be an element of an abelien group $A$, which is part of a canonical isomorphism class $A\in A_{\#}$. Then we a collection of elements 
	\begin{equation*}
		a_{\#}\coloneq \{f(a)\mid f\in \hom(A,A')\text{ for all }A'\in A_{\#}
	\end{equation*} an \textbf{induced element of $a$ in $A_{\#}$}. Notice that for all $A'\in A_{\#}$ there is exactly one $a'\in a_{\#}$ that is contained in $A'$. 
\end{definition}

\begin{definition} \label{def:morseFiltr-postcomposedFunctor}
	Let $G$ be an Abelian group and $X$ be any set. We can \textbf{tag} $G$ with $X$ by defining the set $G\times \{X\}$ and endowing it with the algebraic information from $G$. (for example we define $(g,X)\cdot (g',X)=(g\cdot_Gg',X)$). Let $f:G\to L$ be a homomorphism. Then $f$ induces a homomorphism between the tagged $G\times \{X\}$ and $L\times \{Y\}$ by sending $(g,X)\mapsto (f(g),Y)$.
	Let $F:\TopP\to \Ab$ be a (contravariant) functor from pointed topological spaces to Abelian groups that factors over the homotopy category, i.e. is homotopy invariant. We first define the tagged functor $F^{tag}$ to map spaces to groups that are tagged with the spaces they come from:
	\begin{align*}
		F^{tag}: \TopP &\to \Ab\\
		X&\mapsto F(X)\times \{X\}\\
		\big(f:X\to Y\big)&\mapsto \big(F^{tag}(f):F(Y)\times \{Y\}\to F(X)\times\{X\}\big)
	\end{align*}
	For covariant functors the procedure is analogous.
	Now we can define the \textbf{postcomposed functor} $F_{\#}$ from the category of connected simple systems to the canonical isomorphism classes as follows:
	\begin{align*}
	F_{\#} : \CSS		&\to   \CIAB \\
	I_0 & \mapsto \{F^{tag}(X)\mid X \in I_0 \} \\
	I_m & \mapsto \{ F^{tag}(g) \mid [g]\in I_m \}\\
	\end{align*}
	Let $\Phi:(I_0,I_m)\to (J_0,J_m)$ be a morphism in $\CSS$. Then we define
	\begin{equation*}
		F_{\#}(\Phi): F_{\#}(J_0,J_m) \to  F_{\#}(I_0,I_m)\,,\quad 
		F_{\#}(\Phi) \coloneq \{F^{tag}(g)\mid [g]\in \Phi \}\,.
	\end{equation*}
This obviously becomes a new functor, where the tagging removed uniqueness problems for non-injective (in the literal sense not up to isomorphism) functors $F$. If $F(X)=F(Y)$ for different $X,Y\in I_0$ with $g:X\to Y\in I_m$, then without tagging there could be two different morphisms $F(X)\to F(X)$: the identity and $F(g)$. Since our functors are almost always injective, we will ignore the tagging from now on. 
\end{definition}
\begin{remark}
	Obviously, the procedure of postcomposing a functor works for any homotopy invariant functor from pointed topological spaces into any algebraic category and not just Abelian groups.
\end{remark}
\begin{definition}[Direct Sum in the Category of Canonical Isomorphism Classes]
	Let $A_{\#}$ and $B_{\#}$ be canonical isomorphism classes of Abelien groups. Then we define the \textbf{direct sum} $A_{\#} \oplus B_{\#}$ as follows: 
	The groups are given as all direct sums:
	\begin{equation*}
		\{A\oplus B \mid A\in A_{\#}\,, B\in B_{\#}\}
	\end{equation*}and for two groups $(A_1\oplus B_1)$ and $ (A_2\oplus B_2)$ together with the morphisms $f_{A,A'}\in \hom(A,A')$ and $g_{B,B'}\in \hom(B,B')$ we define the morphism
	\begin{equation*}
		f=f_{A,A'}\oplus f_{B,B'}: (A_1\oplus B_1) \to (A_2\oplus B_2)
	\end{equation*} that makes up the homomorphism set $\hom((A_1\oplus B_1),(A_2\oplus B_2))$.
	The direct sum comes together with the inclusions $(i_A)_{\#}$ and $(i_B)_{\#}$ that are defined as follows: 
	let $1_B\in B \in B_{\#}$. Then we define the morphism $(i_A)_{\#}=(a\mapsto (a,1_B))_{\#}$ to be the morphism induced by the natural inclsion of $A$ into $A\oplus B$. This is independent of the choosen $B\in B_{\#}$. Let $1_{B'}$ be the neutral element of another $B'\in B_{\#}$. Then the induced morphisms of canonical isomorphism classes agree, as all individual morphisms agree. This follows from the internal homomorphisms (for example $g_{B,B'}$) mapping $1_B\mapsto 1_{B'}$. 
	This constuction together with the inclusions makes up the coproduct in $\CIAB$, as it satisfies the universal property: Let $C_{\#}$ be another canonical isomorphism class of Abelien groups, together with two morphisms $f_{\#}:A_{\#}\to C_{\#}$ and $g_{\#}: B_{\#}\to C_{\#}$. Then there exists a unique morphism $f_{\#}\oplus g_{\#}$ such that the diagram comutes:
	\begin{center}
		\begin{tikzcd}
			A_{\#} \arrow[dr,"f_{\#}",bend left=30] \arrow[d,"(i_A)_{\#}"'] &\\
			A_{\#} \oplus B_{\#} \arrow[r,"f_{\#}\oplus g_{\#}"]& C_{\#}\\
			B_{\#} \arrow[u,"(i_B)_{\#}"] \arrow[ur,"g_{\#}"',bend right=30]&
		\end{tikzcd}
	\end{center} This morphism is induced by the direct sum of morphisms on all elements of the canonical isomorphism classes, meaning that $\pcomp{f}\oplus \pcomp{g}=\pcomp{(f\oplus g)}$. 
\end{definition}
\begin{remark}
	Let $A_{\#}=(A_1)_{\#}\oplus (A_2)_{\#}$ be the direct sum of two canonical isomorphism classes. Then $A_{\#}$ comes with two canonical projection morphisms
	\begin{equation*}
		(pr_i)_{\#}: A_{\#} \to (A_i)_{\#} \,,\quad (i=1,2)
	\end{equation*} that are induced by the projection $A_1\oplus A_2\to A_i$ for $i=1,2$ for any $A_1\oplus A_2\in A_{\#}$. Then, for any canonical isomorphism class $C_{\#}$ together with maps $\pcomp{(f_i)}: C_{\#} \to \pcomp{A_i}$ there is a unique morphism $\pcomp{(f_1)}\times \pcomp{f_2}:\pcomp{A}\to \pcomp{C}$ such that the diagram commutes
	\begin{center}
		\begin{tikzcd}
			 \pcomp{(A_1)}&\\
			\pcomp{A}\arrow[u,"\pcomp{(pr_1)}"] \arrow[d,"\pcomp{(pr_2)}"']&\pcomp{C}\arrow[ul,"\pcomp{(f_1)}"',bend right=30] \arrow[ld,"\pcomp{(f_2)}",bend left=30] \arrow[l,"\pcomp{(f_1)}\times \pcomp{f_2}"']\\
			\pcomp{(A_2)}&
		\end{tikzcd}
	\end{center}
	Again, the map $\pcomp{(f_1)}\times \pcomp{f_2}$ is induced by the product of the maps that induce the $\pcomp{(f_i)}$ and hence $\pcomp{(f_1)}\times \pcomp{f_2}=\pcomp{(f_1\times f_2)}$. 
\end{remark}
\begin{proposition}[Biproducts]\label{prop:morseFiltr-biproduct}
	Let $(A,i_j:A_j\to A)$ be a finite coproduct where $i_j$ denote the inclusions. By the universel property, any $x\in A$ has a unique description $x=\sum_ji_j(a_j)$. To see this, assume that $A'\subseteq A$ would be the subgroup of those $x$ that are such linear combinations. Now we have the two functions $p,0:A\to A/A'$, where $p$ is the quotient map and $0$ the trival map. the $A_i$ all include into $A/A'$ and both, the $0$ map and $p$ satisfy that $0\circ i_j=0=p\circ i_j$. by the uniqueness of the universal property we conclude that $p=0$ and thereby $A/A'=0$. Hence the description exists for all $x$. The uniqueness is deduced by fixing a $k$ and defining $f_k=\id_{A_k}:A_k\to A_k$ and $f_j=0$ $(j\neq k)A_j\to A_k$. The universal property gives us a uniqe map $pr_k:A\to A_k$ such that $pr_k \circ i_j=f_j$. Then $pr_k(\sum_ji_j(a_j)) =a_k$. If we argue the same for all $k$ we get that the $a_j$ are all unique. 

	Furthermore, $(A,pr_j)$ satisfies the universal property of the product: If $f_j:B\to A_j$, then $\sum_j i_j\circ f_j$ is the function that composed with any $pr_j$ gives back $f_j$ by definition of $pr_j$. The uniqueness follows from the unique description $x=\sum_ji_j(a_j)$.

	On the other hand, if we are given a finite product $(A,pr_j:A\to A_j)$, we can define the functions $f_j:A_k\to A_j$ that are given by the identity for $k=j$ and trivial else. Then the universal property gives us functions $i_k:A_k\to A$ such that $pr_j\circ i_k=f_k$. Then $(A,i_j)$ is a coproduct and if we are given functinos $f_j:A_j\to B$ for all $j$, the function $ f=\sum_j f_j\circ pr_j$  satisfies the universal property.
\end{proposition}
\begin{corollary}\label{cor:morseFilt-summandsAndFactorsOfMorphisms}
	Let $\pcomp{A}=\bigoplus_{i=1}^n\pcomp{(A_i)}=\bigtimes_{i=1}^n\pcomp{A_i}$ be the direct sum (and thereby also product) of finitly many canonical isomorphism classes and let $\pcomp{C}$ be another canonical isomorphism class. If we are given a morphism $\pcomp{f}: \pcomp{A}\to \pcomp{C}$, we call the morphisms 
	\begin{equation*}
		\pcomp{(f_i)}: \pcomp{(A_i)}\to \pcomp{C}\,,\quad i=1,\dots n
	\end{equation*} given by $\pcomp{f}\circ \pcomp{(i_i)}$ the \textbf{summand functions of $f$}. By the uniqueness of the universal property of the coproduct we know that 
	\begin{equation*}
		\pcomp{f}=\bigoplus_{i=1}^n \pcomp{(f_i)}=\pcomp{\big(\bigoplus_{i=1}^n f_i\big)}=\pcomp{\left(\sum_i f_i\circ pr_i\right)}\,.
	\end{equation*}
	Similar (as any coproduct is also a product), if we are given a function $\pcomp{g}:\pcomp{C}\to \pcomp{A}$ we call the functions
	\begin{equation*}
		\pcomp{(g_i)}\coloneq \pcomp{(pr_i)} \circ \pcomp{g} :\pcomp{C}\to \pcomp{(A_i)}
	\end{equation*} the \textbf{factor functions of $g$}. Again by uniqueness we have 
	\begin{equation*}
		\pcomp{g}=\bigtimes_{i=1}^n \pcomp{(g^i)}=\pcomp{\big(\bigtimes_{i=1}^n g^i\big)}=\pcomp{\left( \sum_i i_i\circ g^i\right)}\,.
	\end{equation*}
	Now assume that we are given a function $f:\bigoplus_{i=1}^n\pcomp{(A_i)}\to \bigoplus_{j=1}^m \pcomp{(B_j)}$. Then we can look at all the factor functions of all summand functions to get
	\begin{equation*}
		\pcomp{(\pcomp{(f_i)})^j}=\pcomp{((f_i)^j)}:\pcomp{(A_i)}\to (\pcomp{B_j})\,.
	\end{equation*} Similary, we could have looked at all the summands of all factor functions to get $\pcomp{((f^j)_i)}$. By definition we then have that 
	\begin{equation*}
		f=\bigoplus_{i=1}^n\big(
			\bigtimes_{j=1}^m (f_i)^j
		)=\bigtimes_{j=1}^m\big(\bigoplus_{i=1}^n (f^j)_i\big)
	\end{equation*}
	Note that by associativity of functions we have the equality that allows us to simplify the notation
	\begin{equation*}
		(f_i)^j=\pcomp{pr_j}\circ \pcomp{f}\circ \pcomp{i_i}=(f^j)_i\eqcolon f_i^j\,.
	\end{equation*} If we now write it down explicitly, we have 
	\begin{equation*}
		f= \sum_{i=1}^n\sum_{j=1}^m i_j \circ f^j_i \circ pr_i\,.
	\end{equation*}
\end{corollary}

\begin{remark}[Critical points as isolated compact invariant subsets] \label{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets}
	Let ${f:M\to \R}$ be a Morse function on a smooth Riemannian manifold $(M,g)$. Let $q\in \Crit_k(f)$. Around $q$ we have refined Morse charts $\phi:U \to \R^m\cong \R^{m-\lambda_q}\oplus \R^{\lambda_p}$ where  the splitting of $\R^m$ is taken such that $\diff \phi_q$ is expanding in $\R^{m-\lambda_q}$ and contracting in $\R^{\lambda_p}$. With this we can define the balls:
	\begin{align*}
		D^{s}_\epsilon &~\coloneq~\{v\in \R^{m-\lambda_q} \mid \norm{v}\leq \epsilon \}, \\
		D^{u}_\epsilon &~\coloneq~\{v\in \R^{\lambda_q\phantom{-m}} \mid \norm{v}\leq \epsilon \}. 
	\end{align*}
For epsilon small enough, they give rise to the index pair $N_q\coloneq\phi^{-1}( D^{s }_\epsilon\times  D^{u }_\epsilon)$ and $L_q\coloneq\phi^{-1}( D^{ s}_\epsilon\times \partial D^{u}_\epsilon)$. To see this, notice that in such a refined Morse chart $\phi$, the differential of the flow in the critical point splits the tangent space into $T^u_qM\times T^s_qM$. Finally, since the property of $\diff \phi_x$ being a contraction on $T^s_qM$ is an open condition and thereby there is an $\epsilon>0$ such that $\diff \phi_x$ is a contraction on $T^s_qM$ for all $x$ with $\norm{\phi^{-1}(x)}<\epsilon$. Thereby, the norm of $\phi_t(x)$ projected to $T^s_qM$ is strictly decreasing in $t$. This proves that $L_q$ is an exit set of $N_q$. Similarly, one sees that $L_q$ is positively invariant in $N_q$. 

This index pair is in fact regular, since $(N_q,L_q)\simeq  (D^{ s}_\epsilon\times  D^{u }_\epsilon, D^{s}_\epsilon\times \partial D^{u}_\epsilon) \simeq (D^{u}_{\epsilon},\partial D^{u}_{\epsilon} )$ and the latters regularity is easy to see by showing that it is an NDR pair. We will cal such an index pair a \textbf{standart index pair of $q$}.

Furthermore, the index pair $(N_q,L_q)$ is homotopically equivalent to $\overline{T^u_qM}$, which is clear, as $(D^{u}_{\epsilon},\partial D^{u}_{\epsilon} ) \simeq \overline{T^u_qM}$ by mapping $\R^{\lambda_q}\to T^u_qM$ via $\restr{\diff \phi^{-1}}{0}\circ R_0$, where $R_0:\R^m\to T_0\R^m$ is the linear identification from Derfinition \ref{def:vb-linearIdentification}. Finally, we just need to strech the ball to infinity. 


We know the K-group of such a pair for any $l\in \N$ to be:
	\begin{align} \label{eq: homology of index pair of critical point}
		K^{-i}(N_q,L_q)\cong \rK^{-i}(\overline{T^u_qM})\cong \begin{cases}
			\Z & \text{if } i-k\in 2\Z,\\
			0& \text{else. }
		\end{cases}
	\end{align}
	Notice that for $k=0$ we need to consider $\partial D^k=\emptyset$ to get an index pair. That's why we let $\partial$ denote the manifold boundary instead of the topological boundary. However, now we can identify for all $k$: \begin{equation} \label{ kritical points as direct sum of homology groups}
		C^{k}(M,\mathfrak{A},\tilde{K}^{-l}(pt))=\bigoplus_{ \Crit_k(f)} \Z \cong \bigoplus_{q\in \Crit_k(f)} K^{-k}(N_q,L_q).
	\end{equation}
	Or using the language of connected simple systems, we have the identification 
	\begin{equation} 
		C^{k}(M,\mathfrak{A},\tilde{K}^{-l}(pt))\cong \bigoplus_{q\in \Crit_k(f)} K_{\#}^{-k}(\mathcal{IP}_q)\,,
	\end{equation}where the isomorphism can be read as an isomorphism of canonical isomorphism classes with the left-hand side viewed as an object of $\CIAB$ consists of only one group. 
\end{remark}
\begin{remark}\label{rem:morseFiltr-indexPairForDifferentChart}
	For any chart $\chi$ that is centered at a critical point $q$ and $\epsilon>0$ we define the topological pair 
	\begin{equation*}
		(N_\chi,L_\chi)\coloneq (\chi^{-1}(D^s_{\epsilon}\times D^u_{\epsilon}), \chi^{-1}(D^s_{\epsilon}\times \partial D^u_{\epsilon}))\,.
	\end{equation*}The argument that shows that $(N_\psi,L_\psi)$ is a reglar index pair of $q$ only needed that $\restr{\diff \psi}{q}$ splitts. Hence if $\restr{\diff \chi}{q}=\restr{\diff \psi}{q}$, we can conclude, that there is an $\epsilon>0$ such that $(N_\chi,L_\chi)$ is also a regular index pair.
\end{remark}

\begin{definition}
	Let $a_{\#}$ be a canonical isomorphism class if Abelien groups. Let $a\in a_{\#}$, be an Abelian group and let $S\subseteq a$ be a generating set of $a$. Then we call the set
	\begin{equation*}
		S_{\#}\coloneq \{ G \mid G=f(S)\,,\text{where }f\in \hom(a,a')\text{ for all }a'\in a_{\#}\}
	\end{equation*} a \textbf{generating set} of $a_{\#}$. Notice, that for all $a'\in a_{\#}$ there is a $G\in S_{\#}$ such that $G$ is a generating set of $a'$. Furtermore, let $s_i$ be an element of $S$. Then we call the set 
	\begin{equation*}
		\pcomp{(s_i)}\coloneq \{ g_i \mid g_i=f(s_i)\,,\text{where }f\in \hom(a,a')\text{ for all }a'\in a_{\#}\}
	\end{equation*} a generator of $\pcomp{a}$. If all $s_i$ together generate $a$, then all $\pcomp{(s_i)}$ together build up $\pcomp{S}$. 
\end{definition}
\begin{remark} \label{rem:MorseFiltr-ProductStruktureOfKIPk}
	In Remark \ref{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets} we established an isomorphism for any $l\in \N$
	\begin{equation*}
		C^k(M,\mathfrak{A},\rK^{-l}(pt))\to \bigoplus_{q\in \Crit_k(f)}K^{-k}_{\#}(\mathcal{IP}_q)\,.
	\end{equation*} Now notice, that the disjoint union of all index pairs $\sqcup_{q\in \Crit_k(f)} (N_q,L_q)$ is an index pair for $\Crit_k(f)$. 
	Hence the group isomorphism given by the additivity of K-theory by Theorem \ref{thm:kTheo-pointedAdditivity} gives us a natural isomorphism 
	\begin{equation*}
		\bigoplus_{q\in \Crit_k(f)} \rK^{-k}(N_q/L_q) \to \rK^{-k}(\bigvee_{q\in \Crit_k(f)} (N_q/L_q))
	\end{equation*} which in turn defines a natural isomorphism of isomorphism classes of groups
	\begin{equation*}
		\bigoplus_{q\in \Crit_k(f)}K^{-k}_{\#}(\mathcal{IP}_q) \to K^{-k}_{\#}(\mathcal{IP}_k)
	\end{equation*} Hence, the class $K^{-k}_{\#}(\mathcal{IP}_k)$ satisfies the propertie of a coproduct with the maps
	\begin{align*}
		\pcomp{i_q}\coloneq \pcomp{(c_q^*)}:\rK^{-k}_{\#}(\mathcal{IP}_{q})&\to K^{-k}_{\#}(\mathcal{IP}_k) \quad\text{ where }\\
		c: \bigvee_{r\in \Crit_k}(N_r/L_r)& \to N_q/L_q\quad \text{ is the collapse,}
	\end{align*}and the properties of the product with the mapsto
	\begin{align*}
		\pcomp{pr_q}\coloneq \pcomp{(i_q)^*}:K^{-k}_{\#}(\mathcal{IP}_k) &\to \rK^{-k}_{\#}(\mathcal{IP}_{q})\quad\text{ where }\\
		i_q: N_q/L_q & \to \bigvee_{r\in \Crit_k}(N_r/L_r)\text{ is the inclusion.}
	\end{align*}
\end{remark}




\begin{remark}[Generators for Canonical Isomorphism Classes]\label{rem:MorseFiltr-canonicalIdentificationOfMorseChainGrou}
	Hence, there is an isomorphism 
	\begin{equation*}
		C^k(M,\mathfrak{A},\rK^{-l}(pt))\to K^{-k}_{\#}(\mathcal{IP}_k)\,.
	\end{equation*} This isomorphism can be made natural. For this we need to establish a natural generating set of $K^{-k}_{\#}(\mathcal{IP}_k)$, which is done by defining a generating set for any (and thereby for all) groups in $K^{-k}_{\#}(\mathcal{IP}_k)$. We pick the group $\rK^{-k}(\bigvee_{q\in \Crit_k(f)} (N_q/L_q))$. Additivity tells us that a generating set is given by including generators of the summands $\rK^{-k}(N_q/L_q)$. Now for each summand we have the isomorphism $\tilde{K}^{-\lambda_q}(N_q/L_q)\cong \tilde{K}^{-\lambda_q}(\overline{T^u_qM})$, which is natural and described in Remark \ref{rem:morseFiltr-criticalPointsAsIsolatedCompactInvariantSubsets}. Hence, if $T^u_qM$ is oriented, Corollary \ref{cor:orient-vectorspaceOrientationInducesKtheoreticOrientation} gives us just what we need, making the map canonical. The generator is then given by pulling back the generator of $K^{-\ind_p}(\overline{T^u_qM})$ along the map of pairs:
	\begin{align*}
	 \kappa_q\coloneq  \restr{\diff \phi^{-1}}{0}\circ R_0\circ s\circ \mathrm{pr}_u \circ \phi	\,:\,(\phi^{-1}( D^{s }_\epsilon\times  D^{u }_\epsilon),\phi^{-1}( D^{ s}_\epsilon\times \partial D^{u}_\epsilon))  \to (\overline{T^u_qM},\{\infty\})
	\end{align*} where $\mathrm{pr}_u$ is the projection onto the unstable coordinates and $s$ is the stretching $x\mapsto  (\frac{x}{\epsilon-\norm{x}})$ with $\restr{\diff s}{0}=\frac{1}{\epsilon}\Idm$. So for every critical point $q\in \Crit_k(f)$ we get a generator of $\alpha_q\in \tilde{K}^{-\lambda_q}(N_q/L_q)$ which pulls back along the collapsoiing map $c_q^*(\alpha_q)\in \rK^{-k}(\bigvee_{r\in \Crit_k}(N_q/L_r))$and then induces an element in $K^{-k}_{\#}(\mathcal{IP}_k)$. 	
	We denote said element as $(\alpha_q^{(N_q,L_q)})_{\#}$ and call it the \textbf{canonical generator corresponding} to $q$. 
	
	Notice the abuse of notation here, where $(\alpha_q^{(N_q,L_q)})_{\#}$ can either be an element of $K^{-k}_{\#}(\mathcal{IP}_k)$, or a generator of $K^{-k}_{\#}(\mathcal{IP}_{\{q\}})$. Those two meanings are related in the following sense: 
	\begin{align*}
		\pcomp{(c_q^*)}((\alpha_q^{(N_q,L_q)})_{\#})=(\alpha_q^{(N_q,L_q)})_{\#}\in K^{-k}_{\#}(\mathcal{IP}_k)\\
		\pcomp{(i_q^*)}((\alpha_q^{(N_q,L_q)})_{\#})=(\alpha_q^{(N_q,L_q)})_{\#}\in K^{-k}_{\#}(\mathcal{IP}_{\{q\}})
	\end{align*}
\end{remark}
	We could have picked another refined Morse chart, or another $\epsilon$ and hence another index pair $(N_q',L_q')$ of $q$. This would give another generator $\alpha_q'^{(N_q',L_q')}\in \tilde{K}^{-\lambda_q}(N_q'/L_q')$. However, as canonical generators, they would agree: For this we need to check the following: let $h:N_q/L_q\to N_q',L_q'$ be the homotopy equivalence from Theorem \ref{thm:morseFiltr-homotopyEquivalenceOfIndexPairs}. In their summands, $\alpha_q'^{(N_q',L_q')}$ and $\alpha_q^{(N_q,L_q)}$ are generators, and as $h^*$ induces an isomorphism we have $h^*(\alpha_q'{(N_q',L_q')})=\pm \alpha_q^{(N_q,L_q)}$. So we need to show, that the sign here is $+$. Whilst we could brute force a calculation here using the single preimage Lemma \ref{lem:orient-singleRegularPreimage} we will later in Remark \ref{rem:morseFiltr-canonicalGeneratorsIndependOfChartAndEpsilon} proof said statement a bit more elegant.
\begin{remark}
The independence of the chosen Morse chart was the main motivation for introducing the language of canonical isomorphism classes. 
The book \cite{banyaga2004lectures} proofes the homotopy equivalence of all index pairs and proceeds much as we do to show that singular homology and Morse homology are isomorphic, albeit in singular homology rather than complex K-theory. This is a written out version of Salamons proof of said statement in \cite{Salamon1990}. 
Similar to us, they also introduce a generator for an index pair by orientation and a Morse chart, but do not show the invariance of said chart (compare \cite[Example 7.17]{banyaga2004lectures}). Whilst this is no gap in the theory (the invariance is to be expected and the authors presumably considered it evident), without the language of canonical isomorphism classes, such an invariance is cumbersome even to state precisely.
\end{remark}

\subsection{Classification of Vector Bundles}
The first goal is to classify all vector bundles over sufficiently nice spaces. The latter refers to spaces admitting partitions of unity for which Tietze's extension theorem holds. For simplicity we just consider compact Hausdorff spaces. Along the way we will discover that the set of isomorphism classes of vector bundles is a homotopy invariance.

Before we go back to the analytical reals, we will further develop the theory of vector bundles:
\begin{definition}
	For a topological space $X$ we define $\Vect_n$ to be the set of isomorphism classes of complex vector bundles of rank $n$ and  $\Vect(X)$ to be the set of isomorphism classes of complex vector bundles of any rank. By declaring $\Vect(f)=f^*$ to be the pullback we get a contravariant functor $\Vect$ from $\Top$ to $\Set$. We can equip the set $\Vect(X)$ with the direct sum and tensor product operation. It is then easy to see that these operations descend to isomorphism classes and hence we get an abelian semi ring $(\Vect(X),\oplus,\otimes)$ (abelian semigroup with associative and distributive multiplication). In fact we even get a contravariant functor from $\Top$ to the category of abelian semirings.
\end{definition}
\begin{lemma}\label{lem:kTheo-sectionExtension}
	Let $X$ be a compact Hausdorff space, $Y\subseteq X$ a closed subspace and $E$ a vector bundle over $X$. Then any section $s:Y\to \restr{E}{Y}$ can be extended to a global section of $X$.
\end{lemma}
\begin{proof}
	For a detailed proof see Chapter 1.4 of \cite{atiyah1989k}. The general argument however is easy to follow: By Tietze's extension theorem we can extend the section to around every $x\in X$. By compactness we can pick finitely many of those extensions. Now using a partition of unity with respect to their domains we can build the sum of all the extensions to get a well defined global one. 
\end{proof}
\begin{corollary}\label{cor:kTheo-isomorphismExtension}
	Let $Y$ be a closed subspace of a compact Hausdorff space $X$, and let $E,F$ be bundles over $X$. If $f:\restr{E}{Y}\to \restr{F}{Y}$ is an isomorphism, then there is an open neighbourhood $U\supset Y$ and an extension $f:  \restr{E}{U}\to \restr{F}{U}$ which is an isomorphism.
\end{corollary}
\begin{proof}
	By corollary \ref{cor:vb-sectionsIntoHom} we can view $f$ as a section into the bundle $\Hom(\restr{E}{Y},\restr{F}{Y})$, which can be extended to a global one in $\Hom(E,F)$. Let $U$ be the open set for which the section maps to isomorphisms. Since $U$ contains $Y$ this yields the require extension.
\end{proof}
\begin{theorem}[Homotopy Invariance of the Pullback]\label{thm:kTheo-homotopyInvariance}
	Let $X,Y$ be compact Hausdorff spaces, $I=[0,1]$, and 
	\begin{equation*}
		f: Y\times I\to X
	\end{equation*} be a homotopy. For a fixed $t\in I$, we define the function $f_t:Y\to X$, $f_t(y)=f(t,y)$. Let $\pi: E\to X$ be a vector bundle. Then 
	\begin{equation*}
		f_0^*(E)\cong f_1^*(E)\,.
	\end{equation*}
\end{theorem}
\begin{proof}
	Consider the following two vector bundles over $Y\times I$:
	\begin{equation*}
		f^*E \quad \text{and} \quad \pi_1^*f_t^*E \,,
	\end{equation*}where $\pi_1:Y\times I\to Y$ denotes the projection and $t\in I$. 
Since the restriction is by definition the pullback along the inclusion 
$	i:Y\times \{t\}\to Y\times I\,.$ The equality of the two maps:
\begin{equation*}
	f\circ i = f_t \circ \pi_1 \circ i :Y\times\{t\} \to X\,
\end{equation*} together with the functoriality gives us an isomorphism 
\begin{equation*}
\Phi:	\restr{f^*E}{Y\times\{t\}} \to \restr{\pi_1^*f_t^*E}{Y\times\{t\}}\,.
\end{equation*}
By Corollary \ref{cor:kTheo-isomorphismExtension} we can extend the isomorphism  for some open neighbourhood of $Y\times \{t\}$ and by the product topology we can deduce that there exists a $\delta_t$ such that $U$ contains the thickened line $Y\times (t-\delta_t,t+\delta_t) $. The isomorphism class of $\restr{f^*E}{Y\times \{t\}}$ as a bundle over $Y$ is locally constant in $t$: Let $0<\epsilon<\delta$ and $r\in I$. Then define the inclusions
	\begin{equation*}
			\begin{array}{rl}
				i_t: Y&\to Y\times \{t\}\subseteq Y\times I \\
				y &\mapsto (y,t)
			\end{array}
		\quad\quad \text{and}\quad \quad
			\begin{array}{rl}
				i_{t+\epsilon}:Y&\to Y\times (t-\delta_t,t+\delta_t)\\
				y &\mapsto (y,t+\epsilon)
			\end{array}
	\end{equation*} Then we have the identities $f_t=f\circ i_t$ and $\pi_1\circ i_{t+\epsilon}=\id_Y$ and we can calculate:
	\begin{equation*}
		f^*_{t+\epsilon}E=i_{t+\epsilon}^*f^*E=i_{t+\epsilon}^*\pi_1^*f_t^*E=(f_t\circ \pi_1\circ i_{t+\epsilon})^*E=f_t^*E\,.
	\end{equation*} Finally, since $I$ is connected, any locally constant map from $I$ is globally constant and therefore $f_0^*E\cong f_1^*E$.
\end{proof}

\begin{corollary}
		\label{lem:kTheo-contractibleBundlesAreTivial}
	\begin{enumerate}
		\item If $f:X\to Y$ is a homotopy equivalence, $f^*:\vect(Y)\to \vect(X)$ is a semiring isomorphism.
		\item If $X$ is contractible, every bundle over $X$ is trivial and $\vect(X)\cong \N_0$
	\end{enumerate}
\end{corollary}

\begin{definition}[Trivialization of a Vector Bundle]
		\label{def:kTheo-trivialization}
	Let $Y$ be a closed subspace of $X$, $E$ a vector bundle over $X$ and $\alpha:\restr{E}{Y}\to Y\times V$ be an isomorphism. Then we call $\alpha$ \textbf{trivialisation of $E$ over $Y$}.
\end{definition} 



\begin{lemma}\label{lem:kTheo-trivialisationDefinesABundle}
	A trivialization $\alpha$ of a bundle $E$ over $Y\subseteq X$ defines a bundle $E\big/ \alpha$ over $X\big/ Y$. The isomorphism class of $E\big/ \alpha$ depends only on the homotopy class of $\alpha$. 
\end{lemma}
\begin{proof}
	We define the bundle $E/ \alpha $ as follows: Let $\pi_2: Y \times V \to V$ denote the projection and define an equivalence relation on $\restr{E}{Y}$ by 
	\begin{equation*}
		e\sim e' ~ \Leftrightarrow \pi_2 \alpha (e) =\pi_2 \alpha (e') \, .
	\end{equation*} We extend this relation by the identity on $\restr{E}{X\setminus Y}$ and we define $E\big/\alpha$ to denote the quotient space of $E$ defined by that relation.
	To see why this is a bundle we need to check the local triviality only around the base point $Y \big/ Y$ of $X \big/ Y$. This follows from the fact that by Corollary \ref{cor:kTheo-isomorphismExtension} we can extend $\alpha$ to an isomorphism $\Tilde{\alpha}:\restr{E}{U}\to U \times V$. Then $\Tilde{\alpha}$ induces an isomorphism
	\begin{equation*}
		\restr{E \big/ \alpha}{U\slash Y} \cong  (U \big/ Y) \times V
	\end{equation*}
	proving the local triviality.
	Now for the homotopy invariance of said bundle suppose that $\alpha_0$ and $\alpha_1$ are homotopic trivializations of $E$ over $Y$. This means that we have a trivialization $\beta$ of $E \times I$ over $Y \times I\subset X \times I$ that induces the $\alpha_i$ at the endpoints. Let $f: (X \big/ Y)\times I\to (X\times I) \big/ (Y\times I)$ be the natural map. Then $f^*((E \times I) \big/  \beta)$ is a bundle on $(X \big/ Y)\times I$ whose restriction to $(X \big/  Y)\times \{i\}$ is $E \big/  \alpha_i$ ($i=0,1$). Hence by the homotopy invariance of the pullback (Theorem \ref{thm:kTheo-homotopyInvariance}) we have 
	\begin{equation*}
		E \big/ \alpha_0 \cong E \big/ \alpha _1\,.
	\end{equation*}
\end{proof}
\begin{lemma}
	\label{lem:kTheo-collapsingOfContractibleSubspace}
	Let $Y\subseteq X$ be a closed contractible subspace of a compact Hausdorff space $X$. Then 
	\begin{equation*}
		f:X\to X\big/ Y \quad \text{induces a semiring isomorphism}\quad f^*: \Vect(X\big/Y) \to \Vect(X)\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Since $Y$ is contractible, $\restr{E}{Y}$ is trivial and thereby a trivialization $\alpha:\restr{E}{Y}\to Y\times V$ exists. Two such trivializations only differ by the automorphism  of $Y\times V$ which in turn corresponds to a map $Y\to \GL(V)\cong \GL_{\rank(E)}(\C)$. Now since $Y$ is contractible those two maps factor (up to homotopy) over any one point $y\in Y$ and are thereby homotopically equivalent to the constant map. Finally since $\GL_{\rank(E)}(\C)$ is connected all constant maps are homotopically equivalent. Thereby $\alpha$ is unique up to homotopy. By Lemma \ref{lem:kTheo-trivialisationDefinesABundle} the isomorphism class of $E\big/ \alpha$ only depends on the homotopy class of $\alpha$ we have a well defined map 
	\begin{equation*}
		\Vect(X)\to \Vect(X\big/ Y)\,,
	\end{equation*}which in turn is an inverse of $f^*$ concluding the proof.
\end{proof}
\begin{definition}[Gluing and Clutching of Vector Bundles]
	\label{def:kTheo-gluingAndClutching}
	Let 
	\begin{align*}
		X=X_1 \cup X_2\, , \quad A=X_1 \cap X_2 \, ,
	\end{align*}
	and all be compact Hausdorff spaces. 
	 Assume that $E_i$ is a vector bundle over $X_i$ and $\phi :\restr{E_1}{A} \to \restr{E_2}{A}$ is an isomorphism. Then we define a vector bundle $E_1 \cap_{\phi} E_2$ on $X$ as follows:
	 We topologize the space $E_1 \cup_{\phi} E_2$ as the quotient of $E_1 + E_2 \slash \sim$, where $\sim$ is the identification of points with their image under $\phi$.
	 Identifying $X$ with $X_1 +X_2 $ in the quotient space we get the natural projection $p:E_1 \cap_{\phi} E_2 \to X$ and the fibers have a natural vector space structure. It remains to show that this is locally trivial. This is obvious outside of $A$ so let $a\in A$ and $V_1$ be a closed neighbourhood of $a$ in $X_1$ such that $\restr{E_1}{V_1}$ is trivial giving us the isomorphisms
	 	\begin{align*}
	 	\theta_1: \restr{E_1}{V_1} &\to V_1 \times \C^n \, \\
	 	\theta_1^A: \restr{E_1}{V_1\cap A} &\to (V_1\cap A) \times \C^n \,
	 \end{align*}
	 	Finally, define
	 \begin{equation*}
	 	\theta^A_2: \restr{E_2}{V_1\cap A}\to (V_1\cap A) \times \C^n \,
	 \end{equation*} by composition $ \theta_1^A \circ \phi^{-1}$. Let $V_2$ be a neighborhood of $a$ in $X_2$ such that 
	 \begin{align*}
	 	\theta_2:\restr{E_2}{V_2}\to V_2 \times \C^n
	 \end{align*} is an extension of $\theta_2^A$. This extension exists again by the extending the section into the homomorphism bundle corresponding to the isomorphism $\theta_2^A$. Finally $\theta_1$ and $\theta_2$  defines a well defined isomorphism 
	 	\begin{equation*}
	 	\theta_1 \cap_{\phi} \theta _2: E_1 \cap_{\phi} E_2 \to (V_1 \cap V_2)\times \C^n\, .
	 \end{equation*} This proves the local triviality.
	 
	 We call such a triple $(E_1,E_2,\phi)$ a \textbf{clutching datum} and $\phi$ the \textbf{transition function}.
\end{definition}

\begin{corollary}[Elementary Properties of the Gluing and Clutching Construction]\label{cor:kTheo-elementaryPropertiesGluingAndClutching}
	Elementary properties of the gluing and clutching construction are:
	\begin{enumerate}
		\item If $E$ is a bundle over $X$, then the identity defines ans isomorphism and 
		\begin{equation*}
			E_1 \cup_{\id_A}E_2 \cong E.
		\end{equation*}
		\item If $\beta_i:E_i \to E_i'$ are isomorphisms on $X_i$ and $\phi' \beta_1=\beta_2\phi$, then  
		\begin{equation*}
			E_1\cup_{\phi}E_2 \cong E'_1\cup_{\phi'}E'_2 \, .
		\end{equation*}
		\item If $(E_i,\phi)$ and $(E'_i,\phi')$ are two "clutching data" on $X_i$, then 
		\begin{align*}
			(E_1 \cup _{\phi}E_2)\oplus (E'_1 \cup _{\phi'}E'_2)&\cong E_1\oplus E'_1 \bigcup_{\phi \oplus \phi'}E_2\oplus E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)\otimes (E'_1 \cup_{\phi'}E'_2)&\cong E_1\otimes E'_1 \bigcup_{\phi \otimes \phi'}E_2\otimes E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)^* &\cong E^*_1\bigcup_{(\phi^*)^{-1}}E^*_2 \, .
		\end{align*} $(-)^*$ denotes the dual here.
	\end{enumerate}
\end{corollary}

\begin{lemma}
	\label{lem:kTheo-homotopyInvarianceClutchingFunction}
	The isomorphism class of $E_1\cup_\phi E_2$ depends only on the homotopy class of the isomorphism $\phi:\restr{E_1}{A}\to \restr{E_2}{A}$\,.
	\end{lemma}
\begin{proof}
	A homotopy of isomorphisms is a map 
\begin{equation*}
	\Phi: \restr{E_1}{A} \times I\to \restr{E_2}{A}\,,
\end{equation*} such that for all $t$ the map 
\begin{equation*}
	\Phi_t: \restr{E_1}{A} \to \restr{E_2}{A}\quad \Phi_t(\xi)\coloneq \Phi(\xi,t)
\end{equation*} is a bundle isomorphism. We can encode this $\Phi$ into a bundle isomorphism as follows: Let $\pi: X\times I\to X$ be the projection. Then elements $\xi\in \pi^*E_1$ are by definition \ref{def:vb-pullbackOfVerctorBundles} of the form $\xi_{x,t}=((x,t),\xi_x)\in (\pi^*E_1)_{x,t}$. With this notation define the bundle isomorphism: 
\begin{equation*}
	\Psi: \restr{\pi^*(E_1)}{A\times I} \to   \restr{\pi^*(E_2)}{A\times I}\,,\quad 	\Psi(\xi_{x,t})\coloneq ((x,t),\Phi_t( \xi_x))
\end{equation*}
This is a bundle isomorphism, since $\Phi_t$ is one. Now denote the inclusion 
\begin{equation*}
	i_t: X \to X \times I\,,\quad i_t(x)\coloneq (x,t)\,,
\end{equation*}and notice that the pullback along $i_t$ is the same as restricting to $X\times\{t\}$. 

With those definitions it is clear by the construction of the gluing, that 
\begin{equation}\label{eq:kTheo-gluingHomotopyInvariant}
	E_1\,\cup_{\Phi_t}\,E_2\,\cong\, i_t^*\left(\pi^*(E_1)\, \cup_{\Psi}\, \pi^*(E_2)\right)\,.
\end{equation} The isomorphism can be seen by part (2) of Corollary  \ref{cor:kTheo-elementaryPropertiesGluingAndClutching}. as follows: 
First notice the isomorphism $\restr{\pi^*(E_i)}{A\times \{t\}}\cong E_i$ for $i=1,2$ and any $t$. By construction of the gluing this leads to the isomorphism 
\begin{equation*}
		E_1\,\cup_{\Phi_t}\,E_2\,\cong\,	\restr{\pi^*(E_1)}{A\times \{t\}} \, \cup_{(\Phi_t,\id)} \, \restr{\pi^*(E_2)}{A\times \{t\}}\,.
\end{equation*} Now again by construction we notice, that gluing restricted spaces is the same as gluing the whole space and then restricting. Furthermore the map $(\Phi_t,\id)$ is the restriction of $\Psi$ to $X\times\{t\}$. Therefore: 
\begin{equation*}
	\restr{\pi^*(E_1)}{A\times \{t\}} \, \cup_{(\Phi_t,\id)} \, \restr{\pi^*(E_2)}{A\times \{t\}}\,\cong \, 	\restr{\left(\pi^*(E_1) \cup_{\Psi} \pi^*(E_2)\right)}{X\times\{t\}}
\end{equation*} Finally, pulling back via $i_t$ is the same as restricting $X\times i$ to $X\times \{t\}$ and identifying $X$ with $X\times \{t\}$. This concludes the chain of arguments:
\begin{equation*}
	\restr{\left(\pi^*(E_1) \cup_{\Psi} \pi^*(E_2)\right)}{X\times\{t\}} \, \cong \,
	i_t^*\left(\pi^*(E_1)\, \cup_{\Psi}\, \pi^*(E_2)\right)\,.
\end{equation*}
Since the pullback is homotopy invariant we can conclude by the isomorphism (\ref{eq:kTheo-gluingHomotopyInvariant}):
\begin{equation*}
	E_1\,\cup_{\Phi_0}\,E_2\,\cong\,E_1\,\cup_{\Phi_1}\,E_2\,.
\end{equation*}
\end{proof}
\begin{definition}\label{def:kTheo-homotopyClassesOfMaps}
	Let $X,Y$ be topological spaces. By $[X,Y]$ we denote the set of homotopy classes of maps from $X$ to $Y$.
\end{definition}

\begin{lemma}\label{lem:kTheo-gluingAndHomotopyClassesToGLn}
	Let $X=X_1\cup X_2$ with $X_1$ and $X_2$ closed contractible spaces. Then there is a canonical bijection 
	\begin{equation*}
		\Vect_n(X) \to [A,\GL_n(\C)]\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Let $[E]$ be any vector bundle over $X$ of rank $n$. Then $E$ can be represented by a vector bundle obtained by gluing that is trivial on both $X_i$: Since $X_i$ is contractible we can choose two trivializations
	\begin{equation*}
		\alpha_1:\restr{E}{X_1}\to X_1\times \C^n,\quad \text{and}\quad 	\alpha_2:\restr{E}{X_2}\to X_2\times \C^n\,.
	\end{equation*} With part (2) of Corollary \ref{cor:kTheo-elementaryPropertiesGluingAndClutching} we can conclude:
	\begin{equation*}
		E\,\cong\,\restr{E}{X_1}\cup_{\id_A} \restr{E}{X_2}\, \cong\, 
		X\times \C^n \cup_{\restr{\alpha_2\circ \alpha_1^{-1}}{A}} X\times \C^n
	\end{equation*} Since a map between the two trivial bundles $A\times \C^n\to A\times \C^n$ can be identified with a map $A\to \GL_n(\C)$ we get an assignment from bundles over $X$ together with the trivializations to maps $A\to \GL_n(\C)$. The next step is to show that two isomorphic vector bundles give homotopic maps: Assume that we are given two vector bundles $E,F$ over $X$ that are isomorphic via $T:E\to F$ and without loss of generality we assume that they are given as the gluing of two trivial bundles: 
	\begin{equation*}
		E: X_1\times \C^n \,\cup_{\phi}\, X_2\times \C^n\,,\quad\text{and}\quad  
		F: X_1\times \C^n \,\cup_{\psi}\, X_2\times \C^n\,.
	\end{equation*}
	We now want to define a homotopy between $psi$ and $\phi$. For this we start by splitting the isomorphism $T$ into its two parts over $X_1$ and $X_2$:
	\begin{equation*}
		\restr{T}{X_i}: X_i\times \C^n \to X_i\times \C^n\,.
	\end{equation*} These maps are given by maps 
	\begin{equation*}
		T_i: X_i\to \GL_n(\C)\,,
	\end{equation*} where the gluing compatibility of the $\restr{T}{X_i}$ yield for any $a\in A$:
	\begin{equation*}
		\psi(a)T_1(a)=T_2(a)\phi(a)\,,\quad \text{ equivalently }\quad \psi(a)=T_2(a)\phi(a)T_1^{-1}(a)\,.
	\end{equation*} By the contractibility of $X_i$ we have for any fixed $x_i\in X_i$ a homotopy $h^i_t:X_i\to X_i$ such that $h^i_0=c_{x_i}$ is the collapsing map whilst $h^i_1=\id_{X_i}$. Furthermore, since $\GL_n(\C)$ is connected we have a path $\gamma^i: I\to \GL_n(\C)$ that connects $T_i(x_i)=\gamma^i(1)$ to $\Idm_n=\gamma^i(0)$. With this we can define the homotopies
	\begin{align*}
		T_{i,t}: X_i&\to \GL_n(\C)\\
		x &\mapsto \begin{cases}
			T_i\circ h^i_{2t}(x)& \text{ if }t\leq \frac{1}{2}\,,\\
			\gamma^i(2(t-\frac{1}{2}))& \text{ if }t > \frac{1}{2}\,,
		\end{cases}
	\end{align*} that satisfy $T_{i,1}=T_i$ and $T_{i,0}(x)=\Idm_n$ for all $x$. Finally we define the homotopy $H_t:A \to \GL_n(\C)$ by
	\begin{equation*}
		H_t(a)\coloneq T_{2,t}(a)\phi(a)T_{1,t}^{-1}(a)\,
	\end{equation*} to get a homotopy from $\phi=H_0$ to $\psi=H_1$. Hence, we get a well defined map from isomorphism classes of bundles over $X$ of rank $n$ to the homotopy classes of maps from $A$ to $\GL_n(\C)$ that is the inverse of the gluing construction and hence is bijective.
\end{proof}
\begin{definition}[Metrics on Bundles]\label{def:kTheo-metricsOnBundles}
	A \textbf{metric} on a complex bundle $E$ is a section $h:X\to \Herm(E)$ such that $h(x)$ is positive definite for all $x$. $\Herm(E)$ denotes the bundle of hermitian forms (conjugate-symmetric sesquilinear) in each fiber. A bundle together with a metric is a \textbf{Hermitian bundle}.
\end{definition}

\begin{lemma}[Metric Induced Complement] \label{lem:kTheo-metricInducedComplement}
	Suppose that $F$ is a sub-bundle of $E$. Then a metric provides a canonical complementary sub-bundle.
\end{lemma}
\begin{proof}Let $h$ denote the metric.	For each $x$ we have the orthonormal projection $P_x:E_x\to F_x$ inducing a map $P:E \to F$. Let $(f_1,\dots f_n)$ be an orthonormal frame of $F$. Then fiberwise with $\xi\in E_x$ the orthogonal projection is given by
		\begin{equation*}
		P_x(\xi)=\sum_i h_x(\xi,f_i(x))f_i(x)\,.

\begin{lemma}[Splitting Prinziple]\label{lem:kTheo-splittingPrinziple}
	Every short exact sequence of bundles over the same base space splits.

		\end{tikzcd}
	\end{center}
	Now it is well known that any short exact sequence of vector spaces splits. We just need to ensure that this splitting is continuous. For this we give $E$ a metric. Now since $\phi'$ is injective, $\phi'(E')$ is a subbundle. By Lemma \ref{lem:kTheo-metricInducedComplement} we can split $E\cong \phi'(E')\oplus L$. 	\begin{equation*}
		\begin{tikzcd}
				0 \arrow[r]&E'\arrow[r,"i_1"]&E'\oplus E'' \arrow[r,"\pi_2"]&E'' \arrow[r]&0
		\end{tikzcd}
	\end{equation*}
\end{proof}
\begin{definition}\label{def:kTheo-ampleSubspace}
	A subspace $V\subseteq \Gamma(E)$, where $\Gamma(E)$ is viewed as a $\C$-vector space is called \textbf{ample}, if 
	\begin{equation*}
		\phi: X\times V \to E,\quad \phi(x,s)\coloneq s(x)
	\end{equation*}is a surjection.
\end{definition}
\begin{corollary}
	If the subspace $V$ is ample and $\dim(V)=\rank(E)$, then $E$ is trivializable.
\end{corollary}
\begin{lemma}\label{lem:kTheo-ampleSubspaceExists}
	For any vector bundle $E$ over a compact Hausdorff space $X$, there exists an finite dimensional ample subspace of $\Gamma(E)$.
\end{lemma}
\begin{proof}
	Let $\{U_\alpha\}$ be a finite open cover of $X$ such that $\restr{E}{U_\alpha}$ is trivial. Let $\{s_{\alpha,i}\}$ be the corresponding local frames, and define $V_\alpha$ to be the finite dimensional ample subspace $V_\alpha\subseteq \Gamma(\restr{E}{U_\alpha})$. Let $\rho_\alpha$ be a partition of unity subordinate to the cover $\{U_\alpha\}$. Define 
	\begin{equation*}
		\theta_\alpha: V_\alpha \to \Gamma(E)\,,\quad \theta_\alpha(s_\alpha)(x)=\begin{cases}
			s_\alpha(x)\cdot \rho_\alpha(x) & \text{ if } x\in U_\alpha\,,\\
			0 & \text{ else.}
		\end{cases}
	\end{equation*} Then the $\theta_\alpha$ give rise to a homomorphism 
	\begin{equation*}
		\theta: \Pi_\alpha V_\alpha \to \Gamma(E)\,,
	\end{equation*} and the image of $\theta$
	is finite dimensional and ample. The surjectivity is clear, as for any $x$ there exists a $\rho_\alpha(x)>0$ making the map 
	\begin{equation*}
		\theta_\alpha(V_\alpha)\to E_x\,
	\end{equation*} a surjection.
\end{proof}
\begin{corollary}\label{cor:kTheo-epiFromTrivial}
	If $E$ is any bundle over $X$, there exists an epimorphism $\phi:X\times \C^m \to E$ for some integer $m$.
\end{corollary}
\begin{proof}
	Just take a basis $(s_1,\dots,s_m)$ of an ample subspace $V$ and define $\phi(x,e_i)=s_i(x)$, where $e_i$ is the $i$-th standard basis vector of $\C^m$.
\end{proof}
\begin{corollary}\label{cor:kTheo-splittingTrivial}
	If $E$ is a bundle over $X$, there exists a bundle $F$ such that $E\oplus F$ is trivial.
\end{corollary}
\begin{proof}
	By Corollary \ref{cor:kTheo-epiFromTrivial} there exists an epimorphism $\phi:X\times \C^m \to E$. The kernel of $\phi$ is a subbundle of $X\times \C^m$ and hence by Lemma \ref{lem:kTheo-splittingPrinziple} we have a splitting of the short exact sequence 
	\begin{center}
		\begin{tikzcd}
			0 \arrow[r]&\ker(\phi)\arrow[r]&X\times \C^m \arrow[r,"\phi"]&E \arrow[r]&0
		\end{tikzcd}
	\end{center} Hence, the middle term becomes isomorphic to the direct sum of the outer terms and we get
	$X\times \C^m\cong E\oplus \ker(\phi)$.
\end{proof}
\begin{definition}[Grassmann Manifold] \label{def:kTheo-grassmanManifold}
	Let $V$ be a vector space and $n$ be any integer. The \textbf{Grassmann manifold} $\Gr_n(V)$ is the set of all subspaces of $V$ with \textbf{codimenion} $n$. If $V$ is given a Hermitian metric, we can topologize $\Gr_n(V)$ as a subspace of the set of all projection operators on $V$ and this defines a map 
	\begin{equation*}
		P: \Gr_n(V) \to \End(V)\,,
	\end{equation*} where $\End(V)$ is the space of all endomorphisms of $V$ with the operator norm topology.
	We give $\Gr_n(V)$ the subspace topology of $\End(V)$.
\end{definition}
\begin{definition}\label{def:kTheo-classifyingMap}
	Suppose that $E$ is a bundle over $X$, $V$ is a vector space and $\phi:X\times V \to E$ is an epimorphism. Then we can define a map $f:X\to \Gr_{\dim(V)}(V)$ by sending $x$ to the kernel of $\phi_x: V\to E_x$. We call $f$ to be induced by $\phi$. This map is continouos for any metric on $V$. To see this notice that the topology on $V$ is metric invariant, as $V$ is finite dimensional. Now the map $f$ is continuous if the map for the map \begin{equation*}
		P: \Gr_n(V) \to \End(V)
	\end{equation*} the composition $P\circ f$ is continuous. But $P\circ f$ is given by sending $x$ to the projection onto $\ker(\phi_x)$, which is continuous as $\phi$ is.
\end{definition}
\begin{definition}[Classifying Bundle]\label{def:kTheo-classifyingBundle}
	Let $V$ be a vector space and $F\subseteq \Gr_n(V)\times V$ be the subbundle consisting of all points $(g,v)$ such that $v\in g$. We call $F$ the \textbf{tautological bundle} over $\Gr_n(V)$. Furthermore, if $E=(\Gr_n(V)\times V)/F$ is the quotient bundle, we call $E$ the \textbf{classifying bundle} over $\Gr_n(V)$.
\end{definition}

\begin{corollary}
	If $E'$ is a bundle over $X$, and $\phi: X\times V\to E'$ is an epimorphism, then $\phi$ induces a map $f:X\to \Gr_{m}(V)$ such that $E'\cong f^*E$, wehere $E$ is the classifying bundle over $\Gr_{m}(V)$. The isomorphism follows from a fiberwise construction because $E_x\cong V\big/\ker(\phi_x)$. The latter however is exactly the fiber over $\ker(x)$ in the classifying bundle and hence the pullback via the $\phi$ induced map gives an isomorphism. That motivates the name "classifying bundle" for $E$ and "classifying map"for $f$. 
\end{corollary}
 \begin{lemma}\label{lem:kTheo-GrassmanTopologyMetricInvariant}
	The topology on $\Gr_n(V)$ is independent of the choice of the metric on $V$.
 \end{lemma}
 \begin{proof}
	Let $h$ and $h'$ be two metrics on $V$. Then we denote the topological space $\Gr_n(V)$ with the topology induced by $h$  by $\Gr(V_h)$ and the one induced by $h'$ by $\Gr(V_{h'})$. We need to show that the identity map $\id:\Gr(V_h)\to \Gr(V_{h'})$ is a homeomorphism. For this we need to show that $\id$ is continuous.The projection $\Gr_n(V_h)\times V_{h'} \to E$ (where $E$ is the classifying bundle) is an epimorphism and hence induces a classifying map $f:\Gr_n(V_h)\to \Gr_n(V_{h'})$ such that $f^*E\cong E$. However that map is given by the identity on $\Gr_n(V)$ and hence the identity is continuous by Definition \ref{def:kTheo-classifyingMap}. 
 \end{proof}
 \begin{definition}[Direct Limit of Sets]\label{def:kTheo-directLimitOfSets}	Let $X_1\to X_2\to \cdots$ be a sequence of sets and injective maps. Then the \textbf{direct limit} $\lim_{m\to \infty} X_m$ is the union $\bigcup_{m=1}^\infty X_m$ modulo the equivalence relation identifying $x$ with $y$ if there exists an $m$ such that $i_m(x)=i_m(y)$. We call $\lim_{m\to \infty} X_m$ the \textbf{direct limit} of the sequence $X_1\to X_2\to \cdots$.
 \end{definition}
 \begin{definition}[Direct Limit of Topological Spaces]\label{def:kTheo-directLimitOfTopologicalSpaces}
	Let $X_1\to X_2\to \cdots$ be a sequence of topological spaces and continuous injective maps. Then the \textbf{direct limit} $\lim_{m\to \infty} X_m$ is the set theoretic direct limit  with the topology given by the final topology with respect to the maps $X_m\to \lim_{m\to \infty} X_m$. I.e. a subset $U\subseteq \lim_{m\to \infty} X_m$ is open if and only if $U\cap X_m$ is open for all $m$.
 \end{definition}
 \begin{lemma}[Universal Property of the Direct Limit]\label{lem:kTheo-universalPropertyOfDirectLimit}
	Let 
	\begin{center}
		\begin{tikzcd}
			X_1 \arrow[r,"i_1"] & X_2 \arrow[r,"i_2"] & \cdots
		\end{tikzcd}
	\end{center} be a sequence of sets (or topological spaces). Then any family of (continuous) maps $f_m:X_m\to Y$ that are compatible with the $i_m$ (i.e. $f_{m+1}\circ i_m=f_m$) factors uniquely through a (continuous) map $f:\lim_{m\to \infty} X_m \to Y$.	
 \end{lemma}
 \begin{proof}
	We start by showing the existence of $f$ by defining $f(x)=f_m(x)$ for $x\in X_m$. This is well defined as the $f_m$ are compatible with the $i_m$. Now we need to show that $f$ is continuous. For this we need to show that $f^{-1}(U)$ is open for any open subset $U\subseteq Y$. But by definition of the topology on the direct limit, it suffices to show that $f^{-1}(U)\cap X_m$ is open for all $m$. 
	This is clear as $f^{-1}(U)\cap X_m=f_m^{-1}(U)$, which is open as $f_m$ is continuous. Finally, the uniqueness of $f$ follows from the fact that the maps $X_m\to \lim_{m\to \infty} X_m$ are injective and their images cover $\lim_{m\to \infty} X_m$.
 \end{proof}
 \begin{corollary}
	The inverse system given by the projections 
	\begin{equation*}
		\pi: \C^m \to \C^{m-1},\quad (x_1,\dots,x_m)\mapsto (x_1,\dots,x_{m-1})\,,
	\end{equation*}induces an direct system of Grassmann manifolds with the induced continuous maps
	\begin{equation*}
		i_{m-1}: \Gr_n(\C^{m-1}) \to \Gr_n(\C^{m})\,,\quad U\mapsto \pi^{-1}(U)\,.
	\end{equation*} If we denote the classifying bundle over $\Gr_n(\C^m)$ by $E_m$, then the pullback of $E_m$ along $i_{m-1}$ is isomorphic to $E_{m-1}$.
 \end{corollary}
 \begin{theorem}
	\label{thm:kTheory-classifyingSpace}
	The map 
	\begin{equation*}
		\lim_{m\to \infty} [X,\Gr_n(\C^m)] \to \Vect_n(X)\,,
	\end{equation*} induced by $f\mapsto f^*E_m$ is a bijection for all compact Hausdorff spaces $X$.
 \end{theorem}
 \begin{proof}
	We shall contrstuct and inverse map. If $E$ is a bundle over $X$, then by Corollary \ref{cor:kTheo-epiFromTrivial} there exists an epimorphism $\phi:X\times \C^m \to E$ for some integer $m$. Let $f:X\to \Gr_n(\C^m)$ be the map induced by $\phi$. If we manage to show that the homotopy class of $f$ as a map to $\Gr_n(\C^{m'})$ for some large enough $m'$ is independent of the choice of $\phi$ and $m$, then we get a well defined map from $\Vect_n(X)$ to $\lim_{m\to \infty} [X,\Gr_n(\C^m)]$ that is inverse to the classifying map. 

	Suppose that $\phi_i :X\times C^{m_i}\to E$ for $i=1,2$ are two epimorphisms. Let $g_i:X\to \G_n(C^{m_i})$ be the maps induced by $\phi_i$. We want to show that $g_1$ and $g_2$ are homotopic as maps to $\Gr_n(\C^{m_1+m_2})$. For this we define the map
	\begin{equation*}
		\psi_t: X\times \C^{m_1}\times C^{m_2} \to E\,,\quad \psi_t(x,v_1,v_2)=t\phi_1(x,v_1)+(1-t)\phi_2(x,v_2)\,.
	\end{equation*} Since the $\phi_i$ are epimorphisms, this map is an epimorphism for any $t$. Let 
	\begin{equation*}
		f_t:X\to \Gr_n(\C^{m_1+m_2})
	\end{equation*} be the mapes induced by $\psi_t$. (where we identified $\C^{m_1}\times \C^{m_2}$ with $\C^{m_1+m_2}$ by mapping $(v_1,v_2)$ to $(v_1,v_2)$). Then 
	\begin{equation*}
		f_0=j_0\circ g_1\quad \text{and}\quad f_1=T \circ j_1\circ g_2\,,
	\end{equation*} where $j_i:\Gr_n(\C^{m_i})\to \Gr_n(\C^{m_1+m_2})$ is the map induced by the inclusion $\C^{m_i}\to \C^{m_1+m_2}$ and $T$ is the map induced by the isomorphism $\C^{m_1+m_2}\to \C^{m_1+m_2}$ that sends $(v_1,v_2)$ to $(v_2,v_1)$. Since $T$ is homotopic to the identity, we can conclude that $j_0\circ g_1$ and $j_1\circ g_2$ are homotopic as maps to $\Gr_n(\C^{m_1+m_2})$. Hence, the homotopy class of the map induced by an epimorphism is independent of the choice of the epimorphism and the integer $m$. This concludes the proof.
 \end{proof}
 \begin{lemma}For any compact $X$ and sequence of injective maps $i_m:X_m\to X_{m+1}$ we have a canonical bijection between the set theoretic direct limit of homotopy classes of maps and the homotopy classes of maps to the topological direct limit:
\begin{equation*} 
	\lim_{m\to \infty} [X,\Gr_n(\C^m)] \cong [X,\lim_{m\to \infty}\Gr_n(\C^m)]\,.
\end{equation*} Here $[X,\Gr_n(\C^m)] \to [X,\Gr_n(\C^{m-1})]$ is the map induced by $i_{m-1}:\Gr_n(\C^{m-1})\to \Gr_n(\C^m)$.
 \end{lemma}
 \begin{proof}
Since $X$ is compact, any map $f:X\to \lim_{m\to \infty}\Gr_n(\C^m)$ has image that is also compact and hence contained in some $\Gr_n(\C^m)$ for $m$ big enough (Just cover it with open maps and reduce that cover to a finite one). Hence there exists a $f_m$ such that $f=i\circ f_m$. where $i$ denotes the inclusion of $\Gr_n(\C^m)$ into the direct limit. This assignement is independent of the choosen $m$, once we view $f_m$ as an element of the the direct limit. This assignement is well defined on homotopy classes becuse if $f,s:X\to \lim_{m\to \infty}\Gr_n(\C^m)$ are homotopic, then there exists a homotopy $H:X\times I\to \lim_{m\to \infty}\Gr_n(\C^m)$ between them. By the same argument as before, the image of $H$ is compact and hence contained in some $\Gr_n(\C^m)$. Hence, $f_m$ and $s_m$ are homotopic via $H_m$ as maps to $\Gr_n(\C^m)$. Hence we get a well defined canonical map 
\begin{equation*}
	\Phi: [X,\lim_{m\to \infty}\Gr_n(\C^m)] \to \lim_{m\to \infty} [X,\Gr_n(\C^m)]\,.
\end{equation*} This map is injective, as if $f_m$ and $s_m$ are homotopic as maps to $\Gr_n(\C^m)$, then they are also homotopic as maps to the direct limit. Finally $\Phi$ is surjective, as any element of the direct limit is represented by some $f_m:X\to \Gr_n(\C^m)$ and hence $\Phi([i\circ f_m])=[f_m]$. 
 \end{proof}

 \begin{definition}
	We need the following categories of topological spaces:
	\begin{itemize}
		\item $\TopC$ is the category of compact Hausdorff spaces.
		\item $\TopCPoint$ is the category of pointed compact Hausdorff spaces.
		\item $\TopCPair$ is the category of pairs of compact Hausdorff spaces.
	\end{itemize}
	Here, we have the natural functors:
	\begin{itemize}
		\item $\TopCPair \to \TopCPoint$ by sending a pair $(X,Y)\mapsto (X/Y,[Y])$ with obvious morphisms. If $Y=\emptyset$ this is the disjoint union of $X$ with a point.
		\item $\TopC \to \TopCPair$ by sending $X\mapsto (X,\emptyset)$ again with obvious morphisms.
		\item $\TopC \to \TopCPoint$ as the composition.
	\end{itemize}
	If we decent to the category where the morphisms are homotopy classes of maps, we get the categories $\hTopC$, $\hTopCPoint$ and $\hTopCPair$. Furthermore, the construction gives a functor from the categorie of abelien semigroups to the categorie of abelien groups. 
\end{definition}

  \begin{definition}[Grothendiek Group and Ring]\label{def:kTheo-grothendiekGroupAndRing}
	Let $A$ be an abelien semigroup. Then the \textbf{Grothendiek group} of $A$ is the abelian group $K(A)$ together with a homomorphism $\phi:A\to K(A)$ such that for any other group $G$ together with a homomorphism $\psi:A\to G$, there exists a unique homomorphism $\psi:K(A)\to G$ such that the following diagram commutes:	
	\begin{center}
		\begin{tikzcd}
			A \arrow[r,"\phi"] \arrow[rd,swap,"\psi"] & K(A) \arrow[d,dashed,"\tilde{\psi}"]\\
			 & G
		\end{tikzcd}	
	\end{center} If $G$ is a semiring, then $K(G)$ inherits a ring structure and we call $K(G)$ the \textbf{Grothendiek ring} of $G$.
  \end{definition}
  \begin{corollary}\label{cor:kTheo-ConstructionOfGrothendiekGroup} 
	The Grothendiek group exists for any abelian semigroup and is unique up to canonical isomorphism. If $G$ is a semiring, then the Grothendiek ring exists and is unique up to canonical isomorphism.
  \end{corollary}
  \begin{proof}The construction follows the following idea: We define touples in which $(a,b)$ that are to be interpreted as differences "$a-b$". For this constructino to yeald something useful we want to enforce the rule that "$a-b=a+c-(b+c)$" or said differently $(a+c,b+c)=(a,b)$. Finally for the univerall property we want to use the smallest relation that makes a group and satisf yealds $(a+c,b+c)=(a,b)$. This gives us the following construction: Let $A$ be an abelien semigroup and $\Delta:A\to A\times A$ be the diaglinal homomorphism that sends $a$ to $(a,a)$. Let $K(A)\coloneq A\times A\big/ \Delta(A)$ .By this notation we mean: $\Delta(A)$ acts on $A\times A$ via
	\begin{equation*}
		\Delta(A) \times (A\times A ) \to A\times A\,\quad ((a,a),(b,c))\mapsto (a+b,b+c)\,.
	\end{equation*} With this we have the set of orbits which gives rise to an relation:
	\begin{equation*}
		\mathcal{R}\coloneq \{((a,b),(a+c,b+c))| c\in A\}
	\end{equation*}Now we take the smallest equivalence relation on $A\times A$ that contains $\mathcal{R}$ (This relation is made explicit in Remark \ref{rem:kTheo-grothendieckEquivalence}). We define $K(A)$ to be $A\times A$ module said relation. Then $K(A)$ is a priori an abelien semigroup, but for $a,b\in A$ we have
\begin{equation*}
	(a,b)+(b,a)=(a+b,b+a)\in \Delta(A)\,,
\end{equation*} and hence $K(A)$ is an abelian group. Also we can see that 
\begin{equation*}
\overline{(a,b)}=\overline{(a,0)}-\overline{(b,0)}=\alpha_A(a)-\alpha_A(b)
\end{equation*} Together with the homomorphism $\alpha_A:A\to K(A)$ that sends $a$ to the class of $(a,0)$, we get the universal property of the Grothendiek group:

Let $G$ be another abelian group and $\psi: A\to G$ a semigroup homomorphism. From this define 
\begin{equation*}
	\tilde{\psi}:K(A)\to G\,,\quad  \overline{(a,b)} \to \psi(a)-\psi(b)
\end{equation*} The well definintion of this map can be verified by a calculation. By calculating it also follows that $\tilde{\psi}$ is a group homomorphism and that $\tilde{\psi}\circ \alpha_A=\psi$. Hence, for the universal property we only need to verify uniqueness. So assume that $\theta:K(A)\to G$ is another group homomorphism. Then:
\begin{equation*}
	\theta\left(\overline{(a,b)}\right)=\underbrace{\theta(\alpha_A(a))}_{\psi(a)}-\underbrace{\theta(\alpha_A(b))}_{\psi(b)}=\tilde{\psi}\left(\overline{(a,b)}\right)
\end{equation*}
This concludes the construction of the grothendiek group. To see the functoriality we first need to define what $K(\beta)$ means, when $\beta: A\to B$ defines a semigroub homomorphism between to abelien semigroups. For this we use the universal property and define $K(\beta)$ to be the unique group homomorphism comming from $\alpha_B\circ \beta : A \to K(B)$. Now the uniqueness of said definition shows the equialities $K(\id)=\id$ and $K(f\circ g)=K(f)\circ K(g)$.
\end{proof}



\begin{remark}\label{rem:kTheo-grothendieckEquivalence}
In $K(A)$ we get the following equivalence for any two pairs $\overline{(a,b)}$ and $\overline{(a',b')}$:
\begin{equation*}
	\overline{(a,b)}=\overline{(a',b')}\,\Leftrightarrow \text{ There exists a } c\in A \text{ such that } (a+b'+c)=(a'+b+c)\,.
\end{equation*}	
\end{remark}
\begin{proof}
	We start by proofing "$\Leftarrow$". This direction follows from the propertys of an equivalence relation, as then, every element of an equivalence class is also a representant:
	\begin{equation} \label{eq:kTheo-theGrothendiekRelation}
		\overline{(a,b)}=\overleftarrow{(a+(a'+b+c),b+(a+b'+c))}=\overline{(a'+(a+b+c),b'+(a+b+c))}=\overline{(a',b')}\,.
	\end{equation} The other direction follows from the minimality of the equivalence relation: To see this notice how the argument above works in every equivalence relation containing the orbits of $\Delta(A)$ and hence works in the smallest one. Furthermore, \begin{equation*}
		(a,b)\sim(a',b')\Leftrightarrow\text{ There exists a } c\in A \text{ such that } (a+b'+c)=(a'+b+c)
	\end{equation*}defines an equivalence relation that contains the orbits and hence this is the smallest one.
\end{proof}


\begin{remark}[A simpler Construction of the Grothendiek Group?]
	Atiyah defines in Chapter II.1 \cite{atiyah1989k} word for word: 

	\textit{ A slightly different construction of $K(A)$ which is sometimes convinient is the following: Let $\Delta: A\to A$ be the diagonal homomorphism of semi-groups, and let $K(A)$ denote the set of cosets of $\Delta(A)$ in $A\times A$. It is a quotient semi-group, but the interchange of factors in $A\times A$ introduces an inverse in $K(A)$ so that $K(A)$ is a group.}
	
	Whilst this definition feels easy to the reader, it carries a lot of baggage. First of all, if one is not familiar with commutative monoid or abelien semigroup theory, one might want to generalice the notion of coset from group theory that states that for a subgroub $A\subseteq G$ of an abelien group $G$ the cosets are of the form $g+A$. Whist this notion can be generaliced to semigroubs, it would yeald something useless. To see this just take $\N$. In the semigroup $\N\times \N \big/ \Delta(\N)$ the cosets $(2,0)+\N$ and $(3,1)+\N$ are different, as $(2,0)$ is not contained in the latter. So Atiyah must have meant something different. However, the literature on semigroups from that time was aware of that problem and defines the notion of qutionts in the setting of semigroubs with the help of \textit{conguence} instead of cosets. The latter refers to an equivalence relation that behaves well with the operation of said semigroup. Compare Chapter 1.5 \cite{CliffordPreston1961}. It was even shown that the notion of congruence and coset are equivalent for groups(Compare for example: Chapter 6.1 \cite{Green1965}). Regardless of what Atiyah meant by his notion of coset, the construction became known and cited. (\cite{Brodzki1996}, \cite{Landweber2005},\cite{Courtney2004}) None of those reference defined coset in semigroubs and hence this little detour into the history should function as a reminder, that we need to be careful.
\end{remark}

\begin{definition}
	For the sake of intuition, I want to give a bit of insight into tensor products without beeing to technical and opening the box of pandora of monoidial categories. We define a tensor product in a caregory simply as the space over which bilinear maps factor uniquely. To capture bilinearity in such a categorie wie define the following. Let $\pi_i:X_0\times X_1 \to X_i$ be the canonical projection for $i=0,1$.  Any resonalble product should come with such surjective maps. Since those are surjective, the $\pi_i$ omitt a family of right inverses $r_i: X_i\to X_0\times X_1$. We call such a right inverse "product preserving" if the compositions $\pi_{2}\circ r_1$ and $\pi_1\circ r_2$ are constant. Hence the right invers maps into just one fiber. Now bilinearity in this general setting states, that for any product preserving right invers of the projections $r_i$, the composition $f\circ r_i$ is structure preserving.
\end{definition}
\begin{remark}[What is the Smash Product]
	We can think of the smash product as the tensor product in the pointed topological spaces as follows. Let $f:(X,x_0)\times (Y,y_0)\to (Z,z_0)$ be bilinear. For any $y\in Y$ we have the product preserving right inverse of $\pi_1$ given by
	\begin{equation*}
		r_1^y: (X,x_0)\to (X,x_0)\times (Y,y_0),\quad x \mapsto (x,y)\,.
	\end{equation*} Analogously we define $r_2^x$ for any $x\in X$ as
	\begin{equation*}
		r_2^x: (Y,y_0)\to (X,x_0)\times (Y,y_0),\quad y \mapsto (x,y)\,.
	\end{equation*}
	 Now for any $x\in X$ we have $f(x,y_0)=f\circ i^x_Y(y_0)=z_0$ as the composition is said to be basepoint preserving. Similar we see that $f(x_0,y)=z_0$ for any $y\in Y$. Hence $f$ factors over the quotient space \begin{equation*}
		X\times Y \big/ (X\times \{y_0\} \cup (Y\times \{x_0\})) \,,
	 \end{equation*} where the nominator is the canonically embedded $X\vee Y$ into the produc. Furtermore by the universal property of quotiens spaces, this factorization is unique.
\end{remark}

\subsection{Homotopy-Theoretic definition of $K$}

\begin{definition}\label{def:kTheo-trivialBundleAndLimitInclusinos}
	for any natural number $n\in \N$ we define $\underline{n}$ to be the $n$-dimensional trivial bundle over a base space $X$. With this we define the inclusions
	\begin{equation*}
		i_n: \Vect_n(X)\to \Vect_{n+1}(X)\,,\quad \xi \mapsto \xi \oplus [\underline{1}],
	\end{equation*} and this gives rise to the direct limit $\lim_n \Vect_n(X)$.
\end{definition}
\begin{definition}[Stabely Equivalence]
	Two vector bundles $E,F$ over the same base space are called \textbf{stably equivalent} if there exists an $n$ such that $E$ and $F$ become isomorphic after adding the trivial bundle: 
	\begin{equation*}
		E\oplus \underline{n}\cong F\oplus \underline{n}\,.
	\end{equation*}
\end{definition}

\begin{lemma}
	In $K(X)$ two equivalence classes $[E]$ and $[F]$ are the same, if and only if $E$ and $F$ are stably equivalent.
\end{lemma}
\begin{proof}
	Assume that two vector bundles satisfy
	\begin{equation*}
		E\oplus \underline{n}\cong F\oplus \underline{n}\,,
	\end{equation*} for some $n\in \N$. By abuse of notation we let $E$ (and $\underline{n}$) denote a vectorbundle and its isomorphismclass simultaniously. Then by Remark \ref{rem:kTheo-grothendieckEquivalence}
	\begin{equation*}
		[E]=\overline{(E,0)}=\overline{(E+\underline{n},\underline{n})}=\overline{(F+\underline{n},\underline{n})}=\overline{(F,0)}=[F]\,.
	\end{equation*}
	On the other hand, if $[E]=[F]$, then by the same Remark we can conclude:
	\begin{equation*}
		(E,0)\sim (F,0)\\
		\Leftrightarrow \text{ there exists } G \text{ such that} E\oplus G=F\oplus G \,.
	\end{equation*} Now by Corollary \ref{cor:kTheo-splittingTrivial} we can find a vector bundle $G'$ such that $G\oplus G'=\underline{n}$ for some $n$. Then
	\begin{equation*}
		E\oplus G\oplus G'=F\oplus G \oplus G' \Rightarrow E\oplus \underline{n}=F\oplus \underline{n}
	\end{equation*}
\end{proof}
\begin{lemma}
	The set of equivalenc classes using the notion of stably equivalence is the limit 
	\begin{equation*}
		\lim_n \Vect_n(X)
	\end{equation*}
\end{lemma}
\begin{proof}Let $\mathrm{SVect}(X)$ be the set of equivalence classes using the notion of stabely equivalence. This satisfies the universal property of the direct limit. To see this let $f_m:\Vect_m(X)\to Y$ be a familiy of maps, that commute with the inclusions $i_n$ from Definition \ref{def:kTheo-trivialBundleAndLimitInclusinos}. From those we define the map 
	\begin{equation*}
		f: \mathrm{SVect}(X) \to Y\,, \quad [\xi] \mapsto f_{\rank(\xi)}(\xi)\,.
	\end{equation*} This map is well defined. To see this assume that $\xi$ and $\zeta$ are stabely equivalent vector bundles and hence for some $n$ we have
	\begin{equation*}
		\xi \oplus \underline{n}\cong \zeta \oplus \underline{n}\,. 
	\end{equation*} Hence, if we include $\xi$ and $\zeta$ into $\Vect_{n+\rank(\xi)}$ they define the same class and since the maps $f_m$ commute with those inclusions, the map $f$ is well defined. The uniquenes of this map follows directly as every element of $\mathrm{SVect}(X)$ is represented in such a way.
\end{proof}
\begin{theorem}
	For a compact connected space $X$ there is a natural bijection
	\begin{equation*}
		K(X) \cong \Z \times \lim_n \Vect_n(X)\,.
	\end{equation*}
\end{theorem}
\begin{proof}
	We will construct such a map. Let $K\in K(X)$, then by corollary \ref{cor:kTheo-ConstructionOfGrothendiekGroup} we know that $K$ can be represented as the difference of two equivalenc classes of vector bundles $K=[E]-[F]$. Now let $G$ be a bundle such that $F\oplus G=\underline{n}$. Then 
	\begin{equation*}
		K=[\underbrace{E\oplus G}_{\eqcolon H}]-[F\oplus G]=[H]-[\underline{n}]\,.
	\end{equation*} Obviously, $G$ is not unique, as $G\oplus \underbrace{1}$ would also work. To omit this issue we choose $G$ such that $n$ is minimal. Then $G$ is unique: Assume that $G$ and $G'$ satisfy the minimality condition. Then $F\oplus G=F\oplus G'=\underline{n}$. 
\end{proof}

\subsection{Classification of Vector Bundles}
The first goal is to classify all vector bundles over sufficiently nice spaces. The latter refers to spaces admitting partitions of unity for which Tietze's extension theorem holds. For simplicity we just consider compact spaces. Along the way we will discover that the set of isomorphism classes of vector bundles is a homotopy invariance.

Before we go back to the analytical reals, we will further develop the theory of vector bundles:
\begin{definition}
	For a topological space $X$ we define $\Vect_n$ to be the set of isomorphism classes of complex vector bundles of rank $n$ and  $\Vect(X)$ to be the set of isomorphism classes of complex vector bundles of any rank. By declaring $\Vect(f)=f^*$ to be the pullback we get a contravariant functor $\Vect$ from $\Top$ to $\Set$. We can equip the set $\Vect(X)$ with the direct sum and tensor product operation. It is then easy to see that these operations descend to isomorphism classes and hence we get an abelian semi ring $(\Vect(X),\oplus,\otimes)$ (abelian semigroup with associative and distributive multiplication). In fact we even get a contravariant functor from $\Top$ to the category of abelian semirings.
\end{definition}
\begin{lemma}\label{lem:kTheo-sectionExtension}
	Let $X$ be a compact space, $Y\subseteq X$ a closed subspace and $E$ a vector bundle over $X$. Then any section $s:Y\to \restr{E}{Y}$ can be extended to a global section of $X$.
\end{lemma}
\begin{proof}
	For a detailed proof see Chapter 1.4 of \cite{atiyah1989k}. The general argument however is easy to follow: By Tietze's extension theorem we can extend the section to around every $x\in X$. By compactness we can pick finitely many of those extensions. Now using a partition of unity with respect to their domains we can build the sum of all the extensions to get a well defined global one. 
\end{proof}
\begin{corollary}\label{cor:kTheo-isomorphismExtension}
	Let $Y$ be a closed subspace of a compact space $X$, and let $E,F$ be bundles over $X$. If $f:\restr{E}{Y}\to \restr{F}{Y}$ is an isomorphism, then there is an open neighbourhood $U\supset Y$ and an extension $f:  \restr{E}{U}\to \restr{F}{U}$ which is an isomorphism.
\end{corollary}
\begin{proof}
	By corollary \ref{cor:vb-sectionsIntoHom} we can view $f$ as a section into the bundle $\Hom(\restr{E}{Y},\restr{F}{Y})$, which can be extended to a global one in $\Hom(E,F)$. Let $U$ be the open set for which the section maps to isomorphisms. Since $U$ contains $Y$ this yields the require extension.
\end{proof}
\begin{theorem}[Homotopy Invariance of the Pullback]\label{thm:kTheo-homotopyInvariance}
	Let $X,Y$ be two compact spaces, $I=[0,1]$, and 
	\begin{equation*}
		f: Y\times I\to X
	\end{equation*} be a homotopy. For a fixed $t\in I$, we define the function $f_t:Y\to X$, $f_t(y)=f(t,y)$. Let $\pi: E\to X$ be a vector bundle. Then 
	\begin{equation*}
		f_0^*(E)\cong f_1^*(E)\,.
	\end{equation*}
\end{theorem}
\begin{proof}
	Consider the following two vector bundles over $Y\times I$:
	\begin{equation*}
		f^*E \quad \text{and} \quad \pi_1^*f_t^*E \,,
	\end{equation*}where $\pi_1:Y\times I\to Y$ denotes the projection and $t\in I$. 
Since the restriction is by definition the pullback along the inclusion 
$	i:Y\times \{t\}\to Y\times I\,.$ The equality of the two maps:
\begin{equation*}
	f\circ i = f_t \circ \pi_1 \circ i :Y\times\{t\} \to X\,
\end{equation*} together with the functoriality gives us an isomorphism 
\begin{equation*}
\Phi:	\restr{f^*E}{Y\times\{t\}} \to \restr{\pi_1^*f_t^*E}{Y\times\{t\}}\,.
\end{equation*}
By Corollary \ref{cor:kTheo-isomorphismExtension} we can extend the isomorphism  for some open neighbourhood of $Y\times \{t\}$ and by the product topology we can deduce that there exists a $\delta_t$ such that $U$ contains the thickened line $Y\times (t-\delta_t,t+\delta_t) $. The isomorphism class of $\restr{f^*E}{Y\times \{t\}}$ as a bundle over $Y$ is locally constant in $t$: Let $0<\epsilon<\delta$ and $r\in I$. Then define the inclusions
	\begin{equation*}
			\begin{array}{rl}
				i_t: Y&\to Y\times \{t\}\subseteq Y\times I \\
				y &\mapsto (y,t)
			\end{array}
		\quad\quad \text{and}\quad \quad
			\begin{array}{rl}
				i_{t+\epsilon}:Y&\to Y\times (t-\delta_t,t+\delta_t)\\
				y &\mapsto (y,t+\epsilon)
			\end{array}
	\end{equation*} Then we have the identities $f_t=f\circ i_t$ and $\pi_1\circ i_{t+\epsilon}=\id_Y$ and we can calculate:
	\begin{equation*}
		f^*_{t+\epsilon}E=i_{t+\epsilon}^*f^*E=i_{t+\epsilon}^*\pi_1^*f_t^*E=(f_t\circ \pi_1\circ i_{t+\epsilon})^*E=f_t^*E\,.
	\end{equation*} Finally, since $I$ is connected, any locally constant map from $I$ is globally constant and therefore $f_0^*E\cong f_1^*E$.
\end{proof}

\begin{corollary}
		\label{lem:kTheo-contractibleBundlesAreTivial}
	\begin{enumerate}
		\item If $f:X\to Y$ is a homotopy equivalence, $f^*:\vect(Y)\to \vect(X)$ is a semiring isomorphism.
		\item If $X$ is contractible, every bundle over $X$ is trivial and $\vect(X)\cong \N_0$
	\end{enumerate}
\end{corollary}

\begin{definition}[Trivialization of a Vector Bundle]
		\label{def:kTheo-trivialization}
	Let $Y$ be a closed subspace of $X$, $E$ a vector bundle over $X$ and $\alpha:\restr{E}{Y}\to Y\times V$ be an isomorphism. Then we call $\alpha$ \textbf{trivialisation of $E$ over $Y$}.
\end{definition} 




\begin{lemma}\label{lem:kTheo-trivialisationDefinesABundle}
	A trivialization $\alpha$ of a bundle $E$ over $Y\subseteq X$ defines a bundle $E\big/ \alpha$ over $X\big/ Y$. The isomorphism class of $E\big/ \alpha$ depends only on the homotopy class of $\alpha$. 
\end{lemma}

\begin{proof}
	We define the bundle $E/ \alpha $ as follows: Let $\pi_2: Y \times V \to V$ denote the projection and define an equivalence relation on $\restr{E}{Y}$ by 
	\begin{equation*}
		e\sim e' ~ \Leftrightarrow \pi_2 \alpha (e) =\pi_2 \alpha (e') \, .
	\end{equation*} We extend this relation by the identity on $\restr{E}{X\setminus Y}$ and we define $E\big/\alpha$ to denote the quotient space of $E$ defined by that relation.
	To see why this is a bundle we need to check the local triviality only around the base point $Y \big/ Y$ of $X \big/ Y$. This follows from the fact that by Corollary \ref{cor:kTheo-isomorphismExtension} we can extend $\alpha$ to an isomorphism $\Tilde{\alpha}:\restr{E}{U}\to U \times V$. Then $\Tilde{\alpha}$ induces an isomorphism
	\begin{equation*}
		\restr{E \big/ \alpha}{U\slash Y} \cong  (U \big/ Y) \times V
	\end{equation*}
	proving the local triviality.
	Now for the homotopy invariance of said bundle suppose that $\alpha_0$ and $\alpha_1$ are homotopic trivializations of $E$ over $Y$. This means that we have a trivialization $\beta$ of $E \times I$ over $Y \times I\subset X \times I$ that induces the $\alpha_i$ at the endpoints. Let $f: (X \big/ Y)\times I\to (X\times I) \big/ (Y\times I)$ be the natural map. Then $f^*((E \times I) \big/  \beta)$ is a bundle on $(X \big/ Y)\times I$ whose restriction to $(X \big/  Y)\times \{i\}$ is $E \big/  \alpha_i$ ($i=0,1$). Hence by the homotopy invariance of the pullback (Theorem \ref{thm:kTheo-homotopyInvariance}) we have 
	\begin{equation*}
		E \big/ \alpha_0 \cong E \big/ \alpha _1\,.
	\end{equation*}
\end{proof}

\begin{corollary} \label{cor:kTheo-trivializationIsRightInverse}
	The trivialization acts as a right inverse of the pullback along the collaps: If we call $f: X\to X\big/Y$ the collaps, and $fE$ can be trivialized over $Y$ via $\alpha$, then 
	\begin{equation*}
		f^*(E\big/ \alpha)\, \cong\, E\,.
	\end{equation*}
\end{corollary}
\begin{proof}
	Intuituvely this is clear, as the pullback along the collaps builds a bundle, that is trivial over $Y$ and agrees with $E$ outside of $Y$. To proof the statement we will construct a bundle map, that is an isomorphism in each fibre: 
	Let $q:E\to E\big/\alpha$ denote the quotient map from the construction and let $\pi_E:E\to X$, $\pi_{E/\alpha}:E\big/\alpha\to X\big/Y$ be the bundle projections. By construction of $E\big/\alpha$ we have $\pi_{E/\alpha}\circ q = f\circ \pi_E$, so
    \begin{equation*}
        \Phi:E\to f^*\big(E\big/\alpha\big)\,,\qquad
        \Phi(e) = \big(\pi_E(e),\, q(e)\big)
    \end{equation*}
    is a well-defined bundle morphism. We show $\Phi$ is a fiberwise linear isomorphism, and hence a bundle isomorphism by Lemma \ref{thm:vb-isomorphicVectorBundles}.

    \emph{Case $x\notin Y$:} Here the equivalence relation defining $E/\alpha$ restricts to the identity on $E_x$, so
    \begin{equation*}
        q_{x}:E_x \;\xrightarrow{\ \cong\ }\; (E\big/\alpha)_{f(x)}=(E\big/\alpha)_x
    \end{equation*}
    is a linear isomorphism.

    \emph{Case $x\in Y$:} By definition of the equivalence relation, two points $e,e'\in E_x$ are identified with $\pi_2\alpha(e)=\pi_2\alpha(e')\in V$, and all fibers $E_x$ ($x\in Y$) are glued to the single fiber $(E/\alpha)_{y_0}\cong V$ over the basepoint $y_0=Y\big/Y$. Hence
    \begin{equation*}
        q_{x} = \pi_2\circ\alpha_x : E_x \rightarrow V = (E\big/\alpha)_{y_0}\,,
    \end{equation*}
    where $\alpha_x:E_x\to V$ is the (linear) restriction of $\alpha$ to the fiber over $x$. Since $\alpha_x$ is a linear isomorphism, so is $q_{x}$.

    Thus $\Phi$ is a bundle morphism which is a linear isomorphism on every fiber.
\end{proof}

\begin{lemma}
	\label{lem:kTheo-collapsingOfContractibleSubspace}
	Let $Y\subseteq X$ be a closed contractible subspace of a compact space $X$. Then 
	\begin{equation*}
		f:X\to X\big/ Y \quad \text{induces a semiring isomorphism}\quad f^*: \Vect(X\big/Y) \to \Vect(X)\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Since $Y$ is contractible, $\restr{E}{Y}$ is trivial and thereby a trivialization $\alpha:\restr{E}{Y}\to Y\times V$ exists. Two such trivializations only differ by the automorphism  of $Y\times V$ which in turn corresponds to a map $Y\to \GL(V)\cong \GL_{\rank(E)}(\C)$. Now since $Y$ is contractible those two maps factor (up to homotopy) over any one point $y\in Y$ and are thereby homotopically equivalent to the constant map. Finally since $\GL_{\rank(E)}(\C)$ is connected all constant maps are homotopically equivalent. Thereby $\alpha$ is unique up to homotopy. By Lemma \ref{lem:kTheo-trivialisationDefinesABundle} the isomorphism class of $E\big/ \alpha$ only depends on the homotopy class of $\alpha$ we have a well defined map 
	\begin{equation*}
		\Vect(X)\to \Vect(X\big/ Y)\,,
	\end{equation*}which in turn is an inverse of $f^*$ concluding the proof.
\end{proof}

\begin{definition}[Gluing and Clutching of Vector Bundles]
	\label{def:kTheo-gluingAndClutching}
	Let 
	\begin{align*}
		X=X_1 \cup X_2\, , \quad A=X_1 \cap X_2 \, ,
	\end{align*}
	and all be compact spaces. 
	 Assume that $E_i$ is a vector bundle over $X_i$ and $\phi :\restr{E_1}{A} \to \restr{E_2}{A}$ is an isomorphism. Then we define a vector bundle $E_1 \cap_{\phi} E_2$ on $X$ as follows:
	 We topologize the space $E_1 \glue{\phi} E_2$ as the quotient of $E_1 + E_2 \slash \sim$, where $\sim$ is the identification of points with their image under $\phi$.
	 Identifying $X$ with $X_1 +X_2 $ in the quotient space we get the natural projection $p:E_1 \cap_{\phi} E_2 \to X$ and the fibers have a natural vector space structure. It remains to show that this is locally trivial. This is obvious outside of $A$ so let $a\in A$ and $V_1$ be a closed neighbourhood of $a$ in $X_1$ such that $\restr{E_1}{V_1}$ is trivial giving us the isomorphisms
	 	\begin{align*}
	 	\theta_1: \restr{E_1}{V_1} &\to V_1 \times \C^n \, \\
	 	\theta_1^A: \restr{E_1}{V_1\cap A} &\to (V_1\cap A) \times \C^n \,
	 \end{align*}
	 	Finally, define
	 \begin{equation*}
	 	\theta^A_2: \restr{E_2}{V_1\cap A}\to (V_1\cap A) \times \C^n \,
	 \end{equation*} by composition $ \theta_1^A \circ \phi^{-1}$. Let $V_2$ be a neighborhood of $a$ in $X_2$ such that 
	 \begin{align*}
	 	\theta_2:\restr{E_2}{V_2}\to V_2 \times \C^n
	 \end{align*} is an extension of $\theta_2^A$. This extension exists again by the extending the section into the homomorphism bundle corresponding to the isomorphism $\theta_2^A$. Finally $\theta_1$ and $\theta_2$  defines a well defined isomorphism 
	 	\begin{equation*}
	 	\theta_1 \cap_{\phi} \theta _2: E_1 \cap_{\phi} E_2 \to (V_1 \cap V_2)\times \C^n\, .
	 \end{equation*} This proves the local triviality.
	 
	 We call such a triple $(E_1,E_2,\phi)$ a \textbf{clutching datum} and $\phi$ the \textbf{transition function}.
\end{definition}

\begin{corollary}[Elementary Properties of the Gluing and Clutching Construction]\label{cor:kTheo-elementaryPropertiesGluingAndClutching}
	Elementary properties of the gluing and clutching construction are:
	\begin{enumerate}
		\item If $E$ is a bundle over $X$, then the identity defines ans isomorphism and 
		\begin{equation*}
			E_1 \glue{\id_A}E_2 \cong E.
		\end{equation*}
		\item If $\beta_i:E_i \to E_i'$ are isomorphisms on $X_i$ and $\phi' \beta_1=\beta_2\phi$, then  
		\begin{equation*}
			E_1\glue{\phi}E_2 \cong E'_1\glue{\phi'}E'_2 \, .
		\end{equation*}
		\item If $(E_i,\phi)$ and $(E'_i,\phi')$ are two "clutching data" on $X_i$, then 
		\begin{align*}
			(E_1 \cup _{\phi}E_2)\oplus (E'_1 \cup _{\phi'}E'_2)&\cong E_1\oplus E'_1 \Glue{\phi \oplus \phi'}E_2\oplus E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)\otimes (E'_1 \glue{\phi'}E'_2)&\cong E_1\otimes E'_1 \Glue{\phi \otimes \phi'}E_2\otimes E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)^* &\cong E^*_1\Glue{(\phi^*)^{-1}}E^*_2 \, .
		\end{align*} $(-)^*$ denotes the dual here.
	\end{enumerate}
\end{corollary}

\begin{lemma}
	\label{lem:kTheo-homotopyInvarianceClutchingFunction}
	The isomorphism class of $E_1\glue\phi E_2$ depends only on the homotopy class of the isomorphism $\phi:\restr{E_1}{A}\to \restr{E_2}{A}$\,.
	\end{lemma}
\begin{proof}
	A homotopy of isomorphisms is a map 
\begin{equation*}
	\Phi: \restr{E_1}{A} \times I\to \restr{E_2}{A}\,,
\end{equation*} such that for all $t$ the map 
\begin{equation*}
	\Phi_t: \restr{E_1}{A} \to \restr{E_2}{A}\quad \Phi_t(\xi)\coloneq \Phi(\xi,t)
\end{equation*} is a bundle isomorphism. We can encode this $\Phi$ into a bundle isomorphism as follows: Let $\pi: X\times I\to X$ be the projection. Then elements $\xi\in \pi^*E_1$ are by definition \ref{def:vb-pullbackOfVerctorBundles} of the form $\xi_{x,t}=((x,t),\xi_x)\in (\pi^*E_1)_{x,t}$. With this notation define the bundle isomorphism: 
\begin{equation*}
	\Psi: \restr{\pi^*(E_1)}{A\times I} \to   \restr{\pi^*(E_2)}{A\times I}\,,\quad 	\Psi(\xi_{x,t})\coloneq ((x,t),\Phi_t( \xi_x))
\end{equation*}
This is a bundle isomorphism, since $\Phi_t$ is one. Now denote the inclusion 
\begin{equation*}
	i_t: X \to X \times I\,,\quad i_t(x)\coloneq (x,t)\,,
\end{equation*}and notice that the pullback along $i_t$ is the same as restricting to $X\times\{t\}$. 

With those definitions it is clear by the construction of the gluing, that 
\begin{equation}\label{eq:kTheo-gluingHomotopyInvariant}
	E_1\glue{\Phi_t} E_2\,\cong\, i_t^*\left(\pi^*(E_1) \glue{\Psi} \pi^*(E_2)\right)\,.
\end{equation} The isomorphism can be seen by part (2) of Corollary  \ref{cor:kTheo-elementaryPropertiesGluingAndClutching}. as follows: 
First notice the isomorphism $\restr{\pi^*(E_i)}{A\times \{t\}}\cong E_i$ for $i=1,2$ and any $t$. By construction of the gluing this leads to the isomorphism 
\begin{equation*}
		E_1 \glue{\Phi_t} E_2\,\cong\,	\restr{\pi^*(E_1)}{A\times \{t\}}  \glue{(\Phi_t,\id)} \restr{\pi^*(E_2)}{A\times \{t\}}\,.
\end{equation*} Now again by construction we notice, that gluing restricted spaces is the same as gluing the whole space and then restricting. Furthermore the map $(\Phi_t,\id)$ is the restriction of $\Psi$ to $X\times\{t\}$. Therefore: 
\begin{equation*}
	\restr{\pi^*(E_1)}{A\times \{t\}}  \glue{(\Phi_t,\id)}  \restr{\pi^*(E_2)}{A\times \{t\}}\,\cong \, 	\restr{\left(\pi^*(E_1) \glue{\Psi} \pi^*(E_2)\right)}{X\times\{t\}}
\end{equation*} Finally, pulling back via $i_t$ is the same as restricting $X\times i$ to $X\times \{t\}$ and identifying $X$ with $X\times \{t\}$. This concludes the chain of arguments:
\begin{equation*}
	\restr{\left(\pi^*(E_1) \glue{\Psi} \pi^*(E_2)\right)}{X\times\{t\}} \, \cong \,
	i_t^*\left(\pi^*(E_1)   \glue{\Psi} \pi^*(E_2)\right)\,.
\end{equation*}
Since the pullback is homotopy invariant we can conclude by the isomorphism (\ref{eq:kTheo-gluingHomotopyInvariant}):
\begin{equation*}
	E_1 \glue{\Phi_0} E_2\,\cong\,E_1 \glue{\Phi_1} E_2\,.
\end{equation*}
\end{proof}
\begin{definition}\label{def:kTheo-homotopyClassesOfMaps}
	Let $X,Y$ be topological spaces. By $[X,Y]$ we denote the set of homotopy classes of maps from $X$ to $Y$.
\end{definition}

\begin{lemma}\label{lem:kTheo-gluingAndHomotopyClassesToGLn}
	Let $X=X_1\cup X_2$ with $X_1$ and $X_2$ closed contractible spaces. Then there is a canonical bijection 
	\begin{equation*}
		\Vect_n(X) \to [A,\GL_n(\C)]\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Let $[E]$ be any vector bundle over $X$ of rank $n$. Then $E$ can be represented by a vector bundle obtained by gluing that is trivial on both $X_i$: Since $X_i$ is contractible we can choose two trivializations
	\begin{equation*}
		\alpha_1:\restr{E}{X_1}\to X_1\times \C^n,\quad \text{and}\quad 	\alpha_2:\restr{E}{X_2}\to X_2\times \C^n\,.
	\end{equation*} With part (2) of Corollary \ref{cor:kTheo-elementaryPropertiesGluingAndClutching} we can conclude:
	\begin{equation*}
		E\,\cong\,\restr{E}{X_1}\glue{\id_A} \restr{E}{X_2}\, \cong\, 
		X\times \C^n \glue{\restr{\alpha_2\circ \alpha_1^{-1}}{A}} X\times \C^n
	\end{equation*} Since a map between the two trivial bundles $A\times \C^n\to A\times \C^n$ can be identified with a map $A\to \GL_n(\C)$ we get an assignment from bundles over $X$ together with the trivializations to maps $A\to \GL_n(\C)$. The next step is to show that two isomorphic vector bundles give homotopic maps: Assume that we are given two vector bundles $E,F$ over $X$ that are isomorphic via $T:E\to F$ and without loss of generality we assume that they are given as the gluing of two trivial bundles: 
	\begin{equation*}
		E: X_1\times \C^n \glue{\phi} X_2\times \C^n\,,\quad\text{and}\quad  
		F: X_1\times \C^n \glue{\psi} X_2\times \C^n\,.
	\end{equation*}
	We now want to define a homotopy between $psi$ and $\phi$. For this we start by splitting the isomorphism $T$ into its two parts over $X_1$ and $X_2$:
	\begin{equation*}
		\restr{T}{X_i}: X_i\times \C^n \to X_i\times \C^n\,.
	\end{equation*} These maps are given by maps 
	\begin{equation*}
		T_i: X_i\to \GL_n(\C)\,,
	\end{equation*} where the gluing compatibility of the $\restr{T}{X_i}$ yield for any $a\in A$:
	\begin{equation*}
		\psi(a)T_1(a)=T_2(a)\phi(a)\,,\quad \text{ equivalently }\quad \psi(a)=T_2(a)\phi(a)T_1^{-1}(a)\,.
	\end{equation*} By the contractibility of $X_i$ we have for any fixed $x_i\in X_i$ a homotopy $h^i_t:X_i\to X_i$ such that $h^i_0=c_{x_i}$ is the collapsing map whilst $h^i_1=\id_{X_i}$. Furthermore, since $\GL_n(\C)$ is connected we have a path $\gamma^i: I\to \GL_n(\C)$ that connects $T_i(x_i)=\gamma^i(1)$ to $\Idm_n=\gamma^i(0)$. With this we can define the homotopies
	\begin{align*}
		T_{i,t}: X_i&\to \GL_n(\C)\\
		x &\mapsto \begin{cases}
			T_i\circ h^i_{2t}(x)& \text{ if }t\leq \frac{1}{2}\,,\\
			\gamma^i(2(t-\frac{1}{2}))& \text{ if }t > \frac{1}{2}\,,
		\end{cases}
	\end{align*} that satisfy $T_{i,1}=T_i$ and $T_{i,0}(x)=\Idm_n$ for all $x$. Finally we define the homotopy $H_t:A \to \GL_n(\C)$ by
	\begin{equation*}
		H_t(a)\coloneq T_{2,t}(a)\phi(a)T_{1,t}^{-1}(a)\,
	\end{equation*} to get a homotopy from $\phi=H_0$ to $\psi=H_1$. Hence, we get a well defined map from isomorphism classes of bundles over $X$ of rank $n$ to the homotopy classes of maps from $A$ to $\GL_n(\C)$ that is the inverse of the gluing construction and hence is bijective.
\end{proof}
\begin{example} \label{ex:kTheo-tautopogicalLineBundleOverCP1}
	We now want to analyze the tautological line bundle $l$ over $\C\PP^1$. For this we cut $\C\PP^1$ into the two pices whic are contractible as they come with homeomorphisms into the unit disc in $\C$:
	\begin{align*}
		H^+&\coloneq \{[x:y]\in \C\PP^1\,|\, \abs{x}\geq \abs{y}\}\, \to D\subset \C\,,\quad [x:y]\mapsto \frac{y}{x}\\
		H^-&\coloneq \{[x:y] \in \C\PP^1\,|\, \abs{x}\leq \abs{y}\}\,\to D\subset \C\,,\quad [x:y]\mapsto \frac{x}{y}
	\end{align*} Those maps give us unique representations of elements in $H^+$ and $H^-$. We now want to evaluate the bijection $\Vect_1(\C\PP^1)\to [H^+\cap H^-,\GL_1(\C)]$ at $[l]$. 
	The image of the intersection $H^+\cap H^-$ under the two maps is the bundary of the Disk and hence a circle. Furthermore, $\GL_1(\C)\cong C^*$. Hence, we know that the image is isomorphic tp $[S^1,\C^*]\cong \pi(S^1)\cong \Z$. This tells us, that there must be nontrivial vector bundles on $\C\PP^1$. By the proof of \ref{lem:kTheo-gluingAndHomotopyClassesToGLn} we know that $[l]$ gets mapped to the function gluing together trivializations over $H^+$ and $H^-$ alog their intersection. To define the trivializations we need define local frames over $H^+$ and $H^-$. Those frames are given by a single section defines as:
	\begin{align*}
		s_+:H^+\to l\,,\quad [x:y]\mapsto (1,\frac{y}{x})\,,\\
		s_-:H^-\to l\,,\quad [x:y]\mapsto (\frac{x}{y},1)\,.
	\end{align*} With those we define the trivializations:
	\begin{align*}
		t_+:\,\restr{l}{H^+}\to H^+\times \C\,,\quad ([x:y],\lambda\cdot s_+[x:y])\mapsto ([x:y],\lambda)\\
		t_-:\,\restr{l}{H^-}\to H^-\times \C\,,\quad ([x:y],\mu\cdot s_-[x:y])\mapsto ([x:y],\mu)
	\end{align*}in each fiber over the intersection we have
	\begin{equation*}
		\lambda\cdot s_+([x:y])=\lambda(1,\frac{y}{x})=\lambda \cdot \frac{y}{x}(\frac{x}{y},1)=\lambda\cdot \frac{y}{x}s_-([x:y])
	\end{equation*}
Hence in each fibre over $[x:y]\in H^+\cap H^-$ the gluing function can be calculated:
\begin{equation*}
	\lambda\mapsto \restr{t_-}{[x:y]}\circ \restr{t_+}{[x:y]}^{-1}(\lambda)=\restr{t_-}{[x:y]}(\lambda \cdot s_+([x:y]))=\restr{t_-}{[x:y]}(\lambda\cdot \frac{y}{x} \cdot s_-([x:y]))=\lambda \cdot \frac{y}{x}
\end{equation*}
Hence, as a map from $H^+\cap H^-\to GL_1(\C)\equiv \C^*$ this map reads:
\begin{equation*}
	H^+\cap H^-\to \C^*\,\quad [x:y]\mapsto \frac{y}{x}
\end{equation*} viewing this as an element of $\pi(S^1)$ by identifying the intersection with a circle via the homeomorphism from $H^+\to D\subseteq \C$ we notice that the gluing map corresponds to the identity in $\pi(S^1)$. If the bundle $l$ would be trivial, the gluing functinon would be homotopically equivalent to the constant one-function $H^+\cap H^-\to \C^*$ that maps all $[x:y]\mapsto 1$. This cannot be, because interpreted as elements in $\pi(S)$ they are different.
\end{example}

\begin{lemma}\label{lem:kTheo-ampleSubspaceExists}
	For any vector bundle $E$ over a compact space $X$, there exists an finite dimensional ample subspace of $\Gamma(E)$.
\end{lemma}
\begin{proof}
	Let $\{U_\alpha\}$ be a finite open cover of $X$ such that $\restr{E}{U_\alpha}$ is trivial. Let $\{s_{\alpha,i}\}$ be the corresponding local frames, and define $V_\alpha$ to be the finite dimensional ample subspace $V_\alpha\subseteq \Gamma(\restr{E}{U_\alpha})$. Let $\rho_\alpha$ be a partition of unity subordinate to the cover $\{U_\alpha\}$. Define 
	\begin{equation*}
		\theta_\alpha: V_\alpha \to \Gamma(E)\,,\quad \theta_\alpha(s_\alpha)(x)=\begin{cases}
			s_\alpha(x)\cdot \rho_\alpha(x) & \text{ if } x\in U_\alpha\,,\\
			0 & \text{ else.}
		\end{cases}
	\end{equation*} Then the $\theta_\alpha$ give rise to a homomorphism 
	\begin{equation*}
		\theta: \Pi_\alpha V_\alpha \to \Gamma(E)\,,
	\end{equation*} and the image of $\theta$
	is finite dimensional and ample. The surjectivity is clear, as for any $x$ there exists a $\rho_\alpha(x)>0$ making the map 
	\begin{equation*}
		\theta_\alpha(V_\alpha)\to E_x\,
	\end{equation*} a surjection.
\end{proof}

\begin{definition}[Grassmann Manifold] \label{def:kTheo-grassmanManifold}
	Let $V$ be a vector space and $n$ be any integer. The \textbf{Grassmann manifold} $\Gr_n(V)$ is the set of all subspaces of $V$ with \textbf{codimenion} $n$. If $V$ is given a Hermitian metric, we can topologize $\Gr_n(V)$ as a subspace of the set of all projection operators on $V$ and this defines a map 
	\begin{equation*}
		P: \Gr_n(V) \to \End(V)\,,
	\end{equation*} where $\End(V)$ is the space of all endomorphisms of $V$ with the operator norm topology.
	We give $\Gr_n(V)$ the subspace topology of $\End(V)$.
\end{definition}
\begin{definition}\label{def:kTheo-classifyingMap}
	Suppose that $E$ is a bundle over $X$, $V$ is a vector space and $\phi:X\times V \to E$ is an epimorphism. Then we can define a map $f:X\to \Gr_{\dim(V)}(V)$ by sending $x$ to the kernel of $\phi_x: V\to E_x$. We call $f$ to be \textbf{induced by $\phi$}. This map is continouos for any metric on $V$. To see this notice that the topology on $V$ is metric invariant, as $V$ is finite dimensional. Now the map $f$ is continuous if the map for the map \begin{equation*}
		P: \Gr_n(V) \to \End(V)
	\end{equation*} the composition $P\circ f$ is continuous. But $P\circ f$ is given by sending $x$ to the projection onto $\ker(\phi_x)$, which is continuous as $\phi$ is.
\end{definition}
\begin{definition}[Classifying Bundle]\label{def:kTheo-classifyingBundle}
	Let $V$ be a vector space and $F\subseteq \Gr_n(V)\times V$ be the subbundle consisting of all points $(g,v)$ such that $v\in g$. We call $F$ the \textbf{tautological bundle} over $\Gr_n(V)$. Furthermore, if $E=(\Gr_n(V)\times V)/F$ is the quotient bundle, we call $E$ the \textbf{classifying bundle} over $\Gr_n(V)$.
\end{definition}

\begin{corollary}
	If $E'$ is a bundle over $X$, and $\phi: X\times V\to E'$ is an epimorphism, then $\phi$ induces a map $f:X\to \Gr_{m}(V)$ such that $E'\cong f^*E$, wehere $E$ is the classifying bundle over $\Gr_{m}(V)$. The isomorphism follows from a fiberwise construction because $E_x\cong V\big/\ker(\phi_x)$. The latter however is exactly the fiber over $\ker(x)$ in the classifying bundle and hence the pullback via the $\phi$ induced map gives an isomorphism. That motivates the name "classifying bundle" for $E$ and "classifying map"for $f$. 
\end{corollary}
 \begin{lemma}\label{lem:kTheo-GrassmanTopologyMetricInvariant}
	The topology on $\Gr_n(V)$ is independent of the choice of the metric on $V$.
 \end{lemma}
 \begin{proof}

\begin{definition}[Directed System]
	Let $\mathcal{C}$ be a categorie and $(I,\leq)$ be a partial orderd set where every finite subset is bounded from above. Let $(A_i)_{i\in I}$ be a family of objects from $\mathcal{C}$ and $f_{ij}:A_i\to A_j$ be a morphism for all $i\leq j$ that satisfies:
	\begin{enumerate}
		\item $f_{ii}= \id_{A_i}$ for all $i\in I$.
		\item $f_{kj}\circ f_{ik}=f_{ij}$ for all $i\leq k\leq j$.
	\end{enumerate}
	The pair $\big((A_i)_{i\in I},(f_{ij})_{i\leq j}\big)$ is called a \textbf{direct system}. 
\end{definition}
\begin{definition}[Direkt Limit]
	Let $\big((A_i)_{i\in I},(f_{ij})_{i\leq j}\big)$ be a direct system in a categorie $\mathcal{C}$.  A pair $\big(X,(\phi_i)_{i\in I}\big)$ where $X$ is an object of $\mathcal{C}$ and $\phi_i:A_i \to X$ are morphisms such that $\phi_j\circ f_{ij}=\phi_i$ for all $i\leq j$ is called a direct limit, if the following universal property holds: 
	For any family of morphisms $g_i:A_i\to Y$ that satisfies $g_i\circ f_{ji}=g_j$ for all $i\leq j$ there exists a unique morphism $g:X\to Y$ such that $g_i=g\circ \phi_i$. We denote the space $X$ by $\colim_{i}A_i$ (or just $\colim A_i$), where the maps $f_{ij}$ are omitted from the notation.
\end{definition}
\begin{corollary}
	As always, the direct Limit is (if it exists) unique up to unique isomorphism.
\end{corollary}
\begin{remark}[Construction of Direct Limits]
	We will now see three constructions of direct limits. Let $\big((A_i)_{i\in I},(f_{ij})_{i\leq j}\big)$ denote a directed system in the noted categorie. 

\textit{ In the categorie of sets} we define the direct limit by
\begin{equation*}
	\colim_i A_i\coloneq \bigsqcup A_i \Big/ \sim  \,,
\end{equation*} where $x\sim y$ if there exists $i,j,k\in I$ such that $f_{ik}(x)=f_{jk}(y)$. This gives an equivalence relation. The maps $\phi_i: A_i\to \colim_{i}A_i$ are the composition of the inclusion into the disjoint uninion with the projection onto the quotient space. The universal property is easy to verify. For a family of suitable maps $g_i:A_i\to B$ we define the map $g:\colim_{i}A_i\to B$ as $g(x)=g_m(x)$ for $x\in A_m$. This is welldefined since the $g_i$ are compatible with the $f_{ij}$ and unique by definition.

\textit{ In the categorie of topological spaces} we define the direct limit to be the settheoretic direct limit equipped with the final toplogy of the inclusions, i.e. a subset $U\subseteq \colim_i A_i$ is open if and only if $\phi_i^{-1}(U)$ is open for all $m$. Again the universal property is easy to prove by using the construction of the universal map from the settheoretic limit. To show the continuoity we show that $g^{-1}(U)$ is open for any open $U$ which is checked by verifying openes of $\phi_i^{-1}\big(g^{-1}(U)\big)$ for all $i$, which in turn equals $g_i^{-1}(U)$ and is open as the $g_i$ are contiuous. Uniqueness follows from the uniqueness of $g$ esteblished already in set. 

\textit{ In an algebraic categorie like semigroubs, groubs, rings } we define the direct limit again as the settheoretic direct limit, where the algebraic operations are defined such that the $\phi_i$ are homomorphisms meaning for example (if $+$ is an operation in the $A_i$) $a\in A_i$ and $b\in A_j$ we define $a+b$ in $\colim{i}A_i$ by including the sum $f_{ik}(a)+f_{jk}(b)$ which is defined in $A_k$. Again the universal morphism $g$ is the one from the settheoretic limit which now is a homomorphism by construction.	
\end{remark}
\begin{remark}\label{rem:kTheo-staircaseMapsClosedEmbedding}
If we build the direct limit of sets and all the maps of the direct system are inkjective, then the inclusions into the direct limit are injective. If all the maps of the direct system are closed embeddings, then the inclusions into the direct are also closed embeddings.
\end{remark}
\begin{proof}
	Recall thet the inclusions $\phi_i:A_i\to \colim A_i$ are defined to be the composition of the inclusion into the disjoint union with the projection onto the quotient. Now assume that all maps $f_{ij}$ in the direct system are injective and $\phi_i(x)=\phi_i(y)$. Then $x\sim y$ in the quotient and hence, there exists $i,j,k\in I$ such that $f_{ik}(x)=f_{jk}(y)$ which implies that $x=y$.

	Now assume that all $f_{ij}$ are closed embeddings. Then, the inclusions are injective by the settheoretic argument. Furthermore, the images of $\phi_i:A_i\to \colim A_i$ are closed. That follows from subsets in the final topology being closed, if and only if all their preimages are closed. Hence we check if $\phi_j(\phi(A_i))$ is closed for all $j\in I$:
	\begin{equation} \label{eq:kTheo-staircaseMapsClosedEmbedding}
		\begin{array}{llll}
			\text{ If }j\leq i& \text{then:} & \phi_j^{-1}(\phi_i(A_i))=(\phi_i\circ f_{ji})^{-1}(\phi_i(A_i))=f_{ji}^{-1}(A_i) &\text{which is closed.}\\
			\text{ If }i<j	 & \text{then:}	&\phi_j^{-1}(\phi_i(A_i)) =\underbrace{\phi_j^{-1}(\phi_j}_{\phi \text{ is injective}} f_{ij}(A_i))=f_{ij}(A_i) &\text{which is closed.}
		\end{array}
	\end{equation} To whow that $\phi_i^{-1}: \im(\phi_i)\to A_i$ is continouos we assume that $B\subset A_i$ is open. then $\phi_i(B)$ is open if and only if for any $j$ $\phi_j^{-1}(\phi_i(B))$ is open. This follows from the same calculations as in equation (\ref{eq:kTheo-staircaseMapsClosedEmbedding}) using that the $f_{ij}$ are homeomorphisms onto their image. 
\end{proof}

	\begin{equation*}
		f_0=j_0\circ g_1\quad \text{and}\quad f_1=T \circ j_1\circ g_2\,,
	\end{equation*} where $j_i:\Gr_n(\C^{m_i})\to \Gr_n(\C^{m_1+m_2})$ is the map induced by the inclusion $\C^{m_i}\to \C^{m_1+m_2}$ and $T$ is the map induced by the isomorphism $\C^{m_1+m_2}\to \C^{m_1+m_2}$ that sends $(v_1,v_2)$ to $(v_2,v_1)$. Since $T$ is homotopic to the identity, we can conclude that $j_0\circ g_1$ and $j_1\circ g_2$ are homotopic as maps to $\Gr_n(\C^{m_1+m_2})$. Hence, the homotopy class of the map induced by an epimorphism is independent of the choice of the epimorphism and the integer $m$. This concludes the proof.
 \end{proof}
 \begin{lemma}\label{lem:kTheo-colimUndHomVertauschen}
	For any compact $X$ and direct system $\big((Y_m)_{m\in \N},(i_m)_{m\in \N}\big)$ where all $Y_m$ are Hausdorff and all $i_m$ are closed embeddings we have a canonical bijection between the set theoretic direct limit of homotopy classes of maps and the homotopy classes of maps to the topological direct limit:
\begin{equation*} 
	\colim_m [X,X_m] \cong [X,\colim_mX_m]\,.
\end{equation*}
 \end{lemma}
 \begin{proof}
	We denote $\colim_mY_m=Y$ and the coresponding inclusion $\phi_m:Y_m\to Y$ . Since $X$ is compact, any map $f:X\to Y$ has image that in some $\im(\phi_m)$ for $m$ big enough. Assume that this was not the case, then would exist an increasing series $(j_i)_{i\in \N}$ such that for all $i$ there is an $x_i$ with $f(x_i)\in \im \phi_{j_i}\setminus \im \phi_{j_i-1}$. With this we define the set 
	\begin{equation*}
		A\coloneq \{f(x_i)\,|\, i\in \N \}\subseteq \im f \subseteq Y\,.
	\end{equation*}Then $A$ is closed, since for all $m$ the intersection $\im\phi_m\cap A$ is finite and since $\im(\phi_m)$ is closed, said intersection is closed. By the final topology we can cnclude that the whole $A$ is closed and since it is contained in the image if $f$ it is compact (Here we use that the $Y_i$ are Hausdorff!). Furthermore, the same argument shows that every subset $B\subset A$ is closed, because again $B\cap \im \phi_m$ is finite. This tells us that $A$ as a subset of $Y$ carries the discrete topology: Let $C\subseteq A$ be an arbitrary subset. Then $A\setminus C\subseteq A$ is closed in $Y$ and hence its complement in $Y$ is an open and intersected with $A$ yealds $C$. Hence, $C$ is open in the subspace topology. This however is an contradiction, as $A$ cannot be infinite, compact and carry the discrete topology.
	Hence any map $f:X\to Y$ has image that in some $\im(\phi_m)$ for $m$ big enough. But then there exists a $f_m$ such that $f=\phi_m\circ f_m$.  This assignement is well defined on homotopy classes becuse if $f,s:X\to Y$ are homotopic, then there exists a homotopy $H:X\times I\to Y$ between them: By the same argument as before, the image of $H$ is compact and hence contained in some $\im \phi_m$. Hence, $f_m$ and $s_m$ are homotopic via $H_m$ as maps to $\im \phi_m$. Hence, we get a well defined canonical map 
\begin{equation*}
	\Phi: [X,\colim_m Y_m] \to \colim_m [X,Y_m]\,.
\end{equation*}This map is injective, as if $f_m$ and $s_m$ are homotopic as maps to $\im\phi_m$, then they are also homotopic as maps to the direct limit. Finally $\Phi$ is surjective, as any element of the direct limit is represented by some $f_m:X\to Y_m$ and hence $\Phi([\phi_m\circ f_m])=[f_m]$. 
 \end{proof}

 \begin{definition}
	We need the following categories of topological spaces:
	\begin{itemize}
		\item $\TopC$ is the category of compact spaces.
		\item $\TopCPoint$ is the category of well-pointed compact spaces.
		\item $\TopCPair$ is the category of pairs of compact spaces, where the inclusions are cofibrations.
	\end{itemize}
	Here, we have the natural functors:
	\begin{itemize}
		\item $\TopCPair \to \TopCPoint$ by sending a pair $(X,Y)\mapsto (X/Y,[Y])$ with obvious morphisms. If $Y=\emptyset$ this is the disjoint union of $X$ with a point.
		\item $\TopC \to \TopCPair$ by sending $X\mapsto (X,\emptyset)$ again with obvious morphisms. the inclusion of $\setminus$ into $X$ is a cofibration, as the empty set itself is a open neighbourhood and hence the pair is a NDR pair.
		\item $\TopC \to \TopCPoint$ as the composition. Again, the inclusion of $[\emptyset]$ into the disjoint union $X \sqcup [\emptyset]$ is a cofibration, as $\{[\emptyset]\}$ is an open neighbourhood of itself.
	\end{itemize}
	If we decent to the category where the morphisms are homotopy classes of maps, we get the categories $\hTopC$, $\hTopCPoint$ and $\hTopCPair$. Furthermore, the construction gives a functor from the categorie of abelien semigroups to the categorie of abelien groups. 
\end{definition}

  \begin{definition}[Grothendiek Group and Ring]\label{def:kTheo-grothendiekGroupAndRing}
	Let $A$ be an abelien semigroup. Then the \textbf{Grothendiek group} of $A$ is the abelian group $K(A)$ together with a homomorphism $\phi:A\to K(A)$ such that for any other group $G$ together with a homomorphism $\psi:A\to G$, there exists a unique homomorphism $\psi:K(A)\to G$ such that the following diagram commutes:	
	\begin{center}
		\begin{tikzcd}
			A \arrow[r,"\phi"] \arrow[rd,swap,"\psi"] & K(A) \arrow[d,dashed,"\tilde{\psi}"]\\
			 & G
		\end{tikzcd}	
	\end{center} If $G$ is a semiring, then $K(G)$ inherits a ring structure and we call $K(G)$ the \textbf{Grothendiek ring} of $G$.
  \end{definition}
  \begin{corollary}\label{cor:kTheo-ConstructionOfGrothendiekGroup} 
	The Grothendiek group exists for any abelian semigroup and is unique up to canonical isomorphism. If $G$ is a semiring, then the Grothendiek ring exists and is unique up to canonical isomorphism.
  \end{corollary}
  \begin{proof}The construction follows the following idea: We define touples in which $(a,b)$ that are to be interpreted as differences "$a-b$". For this constructino to yeald something useful we want to enforce the rule that "$a-b=a+c-(b+c)$" or said differently $(a+c,b+c)=(a,b)$. Finally, to accfount for the universal property, we want to use the smallest relation that obeys siad rule. This gives us the following construction: Let $A$ be an abelien semigroup and $\Delta:A\to A\times A$ be the diaglinal homomorphism that sends $a$ to $(a,a)$. Let $K(A)\coloneq A\times A\big/ \Delta(A)$ .By this notation we mean: $\Delta(A)$ acts on $A\times A$ via
	\begin{equation*}
		\Delta(A) \times (A\times A ) \to A\times A\,\quad ((a,a),(b,c))\mapsto (a+b,a+c)\,.
	\end{equation*} With this we have the set of orbits which gives rise to an relation:
	\begin{equation*}
		\mathcal{R}\coloneq \{((a,b),(a+c,b+c))| c\in A\}
	\end{equation*}Now we take the smallest equivalence relation on $A\times A$ that contains $\mathcal{R}$ (This relation is made explicit in Remark \ref{rem:kTheo-grothendieckEquivalence}). We define $K(A)$ to be $A\times A$ module said relation. Then $K(A)$ is a priori an abelien semigroup, but for $a,b\in A$ we have
\begin{equation*}
	(a,b)+(b,a)=(a+b,b+a)\in \Delta(A)\,,
\end{equation*} and hence $K(A)$ is an abelian group. Also we can see that 
\begin{equation*}
\overline{(a,b)}=\overline{(a,0)}-\overline{(b,0)}=\alpha_A(a)-\alpha_A(b)
\end{equation*} Together with the homomorphism $\alpha_A:A\to K(A)$ that sends $a$ to the class of $(a,0)$, we get the universal property of the Grothendiek group:

Let $G$ be another abelian group and $\psi: A\to G$ a semigroup homomorphism. From this define 
\begin{equation*}
	\tilde{\psi}:K(A)\to G\,,\quad  \overline{(a,b)} \to \psi(a)-\psi(b)
\end{equation*} The well definintion of this map can be verified by a calculation. By calculating it also follows that $\tilde{\psi}$ is a group homomorphism and that $\tilde{\psi}\circ \alpha_A=\psi$. Hence, for the universal property we only need to verify uniqueness. So assume that $\theta:K(A)\to G$ is another group homomorphism. Then:
\begin{equation*}
	\theta\left(\overline{(a,b)}\right)=\underbrace{\theta(\alpha_A(a))}_{\psi(a)}-\underbrace{\theta(\alpha_A(b))}_{\psi(b)}=\tilde{\psi}\left(\overline{(a,b)}\right)
\end{equation*}
This concludes the construction of the grothendiek group. To see the functoriality we first need to define what $K(\beta)$ means, when $\beta: A\to B$ defines a semigroub homomorphism between to abelien semigroups. For this we use the universal property and define $K(\beta)$ to be the unique group homomorphism comming from $\alpha_B\circ \beta : A \to K(B)$. Now the uniqueness of said definition shows the equialities $K(\id)=\id$ and $K(f\circ g)=K(f)\circ K(g)$.
\end{proof}



\begin{remark}\label{rem:kTheo-grothendieckEquivalence}
In $K(A)$ we get the following equivalence for any two pairs $\overline{(a,b)}$ and $\overline{(a',b')}$:
\begin{equation*}
	\overline{(a,b)}=\overline{(a',b')}\,\Leftrightarrow \text{ There exists a } c\in A \text{ such that } (a+b'+c)=(a'+b+c)\,.
\end{equation*}	
\end{remark}
\begin{proof}
	We start by proofing "$\Leftarrow$". This direction follows from the propertys of an equivalence relation, as then, every element of an equivalence class is also a representant:
	\begin{equation} \label{eq:kTheo-theGrothendiekRelation}
		\overline{(a,b)}=\overline{(a+(a'+b+c),b+(a+b'+c))}=\overline{(a'+(a+b+c),b'+(a+b+c))}=\overline{(a',b')}\,.
	\end{equation} The other direction follows from the minimality of the equivalence relation: To see this notice how the argument above works in every equivalence relation containing the orbits of $\Delta(A)$ and hence works in the smallest one. Furthermore, \begin{equation*}
		(a,b)\sim(a',b')\Leftrightarrow\text{ There exists a } c\in A \text{ such that } (a+b'+c)=(a'+b+c)
	\end{equation*}defines an equivalence relation that contains the orbits and hence this is the smallest one.
\end{proof}


\begin{remark}[A simpler Construction of the Grothendiek Group?] One might be wondering, why we defined the relation on the pair $A\times A$ as the smallest equivalence relation starting identifying the orbits of the operation of the diagonal, when Remark \ref{rem:kTheo-grothendieckEquivalence} meight look and feel different. It was even shown that the notion of congruence and coset are equivalent for groups(Compare for example: Chapter 6.1 \cite{Green1965}). Regardless of what Atiyah meant by his notion of coset, the construction became known and cited. (\cite{Brodzki1996}, \cite{Landweber2005},\cite{Courtney2004}) None of those reference defined coset in semigroubs and hence this little detour into the history should function as a reminder, that we need to be careful when generalizing things from the known categories. We are so used to cosets giving equivalenc relations, that its easy to overlook the needed care. 
\end{remark}

\begin{definition}
	For the sake of intuition, I want to give a bit of insight into tensor products without beeing to technical and opening the box of pandora of monoidial categories. We define a tensor product in a caregory simply as the space over which bilinear maps factor uniquely. To capture bilinearity in such a categorie wie define the following. Let $\pi_i:X_0\times X_1 \to X_i$ be the canonical projection for $i=0,1$.  Any resonalble product should come with such surjective maps. Since those are surjective, the $\pi_i$ omitt a family of right inverses $r_i: X_i\to X_0\times X_1$. We call such a right inverse "product preserving" if the compositions $\pi_{2}\circ r_1$ and $\pi_1\circ r_2$ are constant. Hence the right invers maps into just one fiber. Now bilinearity in this general setting states, that for any product preserving right invers of the projections $r_i$, the composition $f\circ r_i$ is structure preserving.
\end{definition}
\begin{remark}[What is the Smash Product]
	We can think of the smash product as the tensor product in the pointed topological spaces as follows. Let $f:(X,x_0)\times (Y,y_0)\to (Z,z_0)$ be bilinear. For any $y\in Y$ we have the product preserving right inverse of $\pi_1$ given by
	\begin{equation*}
		r_1^y: (X,x_0)\to (X,x_0)\times (Y,y_0),\quad x \mapsto (x,y)\,.
	\end{equation*} Analogously we define $r_2^x$ for any $x\in X$ as
	\begin{equation*}
		r_2^x: (Y,y_0)\to (X,x_0)\times (Y,y_0),\quad y \mapsto (x,y)\,.
	\end{equation*}
	 Now for any $x\in X$ we have $f(x,y_0)=f\circ i^x_Y(y_0)=z_0$ as the composition is said to be basepoint preserving. Similar we see that $f(x_0,y)=z_0$ for any $y\in Y$. Hence $f$ factors over the quotient space \begin{equation*}
		X\times Y \big/ (X\times \{y_0\} \cup (Y\times \{x_0\})) \,,
	 \end{equation*} where the nominator is the canonically embedded $X\vee Y$ into the produc. Furtermore by the universal property of quotiens spaces, this factorization is unique.
\end{remark}

\subsection{Homotopy-Theoretic definition of $K$}

\begin{definition}\label{def:kTheo-trivialBundleAndLimitInclusinos}
	for any natural number $n\in \N$ we define $\underline{n}$ to be the $n$-dimensional trivial bundle over a base space $X$. With this we define the inclusions
	\begin{equation*}
		i_n: \Vect_n(X)\to \Vect_{n+1}(X)\,,\quad \xi \mapsto \xi \oplus [\underline{1}],
	\end{equation*} and this gives rise to the direct limit $\colim_n \Vect_n(X)$.
\end{definition}
\begin{definition}[Stabely Equivalence]
	Two vector bundles $E,F$ over the same base space are called \textbf{stably equivalent} if there exists an $n$ such that $E$ and $F$ become isomorphic after adding the trivial bundle: 
	\begin{equation*}
		E\oplus \underline{n}\cong F\oplus \underline{n}\,.
	\end{equation*}
\end{definition}

\begin{lemma}
	In $K(X)$ two equivalence classes $[E]$ and $[F]$ are the same, if and only if $E$ and $F$ are stably equivalent.
\end{lemma}
\begin{proof}
	Assume that two vector bundles satisfy
	\begin{equation*}
		E\oplus \underline{n}\cong F\oplus \underline{n}\,,
	\end{equation*} for some $n\in \N$. By abuse of notation we let $E$ (and $\underline{n}$) denote a vectorbundle and its isomorphismclass simultaniously. Then by Remark \ref{rem:kTheo-grothendieckEquivalence}
	\begin{equation*}
		[E]=\overline{(E,0)}=\overline{(E+\underline{n},\underline{n})}=\overline{(F+\underline{n},\underline{n})}=\overline{(F,0)}=[F]\,.
	\end{equation*}
	On the other hand, if $[E]=[F]$, then by the same Remark we can conclude:
	\begin{equation*}
		(E,0)\sim (F,0)\\
		\Leftrightarrow \text{ there exists } G \text{ such that} E\oplus G=F\oplus G \,.
	\end{equation*} Now by Corollary \ref{cor:kTheo-splittingTrivial} we can find a vector bundle $G'$ such that $G\oplus G'=\underline{n}$ for some $n$. Then
	\begin{equation*}
		E\oplus G\oplus G'=F\oplus G \oplus G' \Rightarrow E\oplus \underline{n}=F\oplus \underline{n}
	\end{equation*}
\end{proof}
\begin{definition}
	We call to vector bundles $E,F$ over the same base space \textbf{eventually equivalent}, if there exist $n,m\in \N$ such that $E\oplus n\cong F\oplus m$.
\end{definition}
\begin{lemma}
	The set of equivalenc classes using the notion of eventual equivalence is the limit 
	\begin{equation*}
		\colim_n \Vect_n(X)
	\end{equation*}
\end{lemma}
\begin{proof}Let $\mathrm{EVect}(X)$ be the set of equivalence classes using the notion of eventually equivalence. This satisfies the universal property of the direct limit. To see this let $f_m:\Vect_m(X)\to Y$ be a familiy of maps, that commute with the inclusions $i_n$ from Definition \ref{def:kTheo-trivialBundleAndLimitInclusinos}. From those we define the map 
	\begin{equation*}
		f: \mathrm{EVect}(X) \to Y\,, \quad [\xi] \mapsto f_{\rank(\xi)}(\xi)\,.
	\end{equation*} This map is well defined, because $f$ commuting with the inclusions $i_n$ is equivalent to $f$ being invariant under eventual equivalence. Uniqueness follows from ever element of $\mathrm{EVect}(X)$ being represented by a vecor bundle.
\end{proof}
\begin{lemma}
	The set of equivalenc classes using the notion of eventual equivalence  together with the direct sum is an abelian group
\end{lemma}
\begin{proof}
	Everything except inverses is clear. The neutral element is the class of $\underline{0}$. For inverses we take a bundle $E$ and by Corollary \ref{cor:kTheo-splittingTrivial} there exists a $G$ such that $E\oplus G\cong\underline{n}$. Now $[E]+[G]=[\underline{n}]=\underline[0]$. 
\end{proof}
\begin{lemma}\label{lem:kTheo-reducedKisEventually}
	For a compact connected space $X$ there is a natural group isomorphism.
	\begin{equation*}
		K(X) \cong \Z\times \mathrm{EVect}(X) \cong\Z \times \colim_n \Vect_n(X)\,.
	\end{equation*}
	Furthermore if $(X,p)$ is pointed, there is a natural group isomorphism 
	\begin{equation*}
		\tilde{K}(X)\cong \mathrm{EVect}(X) \cong \colim_n \Vect_n(X)\,.
	\end{equation*}
\end{lemma}
\begin{proof}
Let $K$ be an element of $K(X)$. Then by Corollary \ref{cor:kTheo-ConstructionOfGrothendiekGroup} we can write $K$ as the difference of two classes of vector bundles $K=[E]-[F]$. We abuse the notation of equivalence classes in this proof meaning we won't differenciate between isomorphism classes of vector bundles in $\Vect_n$, eventually or stable isomorphism classes or equivalence classes of naturaly included vector bundles in $K(X)$. The context should aways be clear.  With this we define the funktion
	\begin{equation*}
		\rank(K): K(X)\to \Z\,,\quad K=[E]-[F] \mapsto \rank(E)-\rank(F)\,.
	\end{equation*} This is well defined as if $K=[E]-[F]=[E']-[F']$, then by the relatino in $K(X)$ we know that there is an $n$ such that
	\begin{equation*}
		E\oplus F' \oplus \underline{n}\cong E' \oplus F \oplus \underline{n}\,.
	\end{equation*} and hence:
	\begin{equation*}
		\rank(E)-\rank(F)=\rank(E')-\rank(F')
	\end{equation*} Furthermore, $\rank$ has a natural right inverse given by
	\begin{equation*}
		r: \Z \to K(X)\,,\quad m\mapsto \begin{cases}
			[m]-[0]& \text{ if }m\geq 0\\
			[0]-[-m]& \text{ if }m<0\,.
		\end{cases}
	\end{equation*}
	Now we claim that the kernal $\ker(\rank)\cong \mathrm{EVect}(X)$. For this we define the map 
	\begin{equation*}
		\phi: \mathrm{EVect}(X)\to K(X)\,,\quad [E]\mapsto [E] -[\underline{\rank(E)}]
	\end{equation*}Tho see the well definition we assume that $E$ and $E'$ are eventually isomorphic and hence there are $n,n'\in \N$ such that $E\oplus \underline{n}\cong E'\oplus \underline{n'}$. But then in $K(X)$ we have:
	\begin{equation*}
		[E]-[\underline{\rank(E)}]=[E\oplus \underline{n}]-[\underbrace{\underline{\rank(E)}\oplus\underline{n}}_{=\underline{\rank(F)}\oplus \underline{n'}}]=[E'\oplus \underline{n'}]-[\underline{\rank(F)}\oplus \underline{n'}]=[F]-[\underline{\rank(F)}]\,.
	\end{equation*}
	Obviously $\rank\circ \phi \equiv 0$. To get the other inclusion we first notice the following: Let $K\in K(X)=[E]-[F]$ be arbitrary. By Corollary \ref{cor:kTheo-splittingTrivial} there exists a $G$ such that $F\oplus G\cong \underline{n}$ for some $n\in \N$. Hence if $K$ is in the kernal of $\rank$ we have:
	\begin{equation*}
		K=[E]-[F]=[E \oplus G]-[\underline{n}]=[E\oplus G]-[\underline{\rank(E\oplus G)}]= \phi([E \oplus G])
	\end{equation*}
	This leads to the naturally splitting short exact sequence
\begin{center}
		\begin{tikzcd}
			0 \arrow[r]& \mathrm{EVect}(X)\arrow[r,"\phi"]&K(X)\arrow[r,"\rank"']&\Z \arrow[r] \arrow[l,bend right,"r"']& 0
		\end{tikzcd}
\end{center}
This already proofs the isomorphisms \begin{equation*}
		K(X) \cong \Z\times \mathrm{EVect}(X) \cong\Z \times \colim_n \Vect_n(X)\,.
	\end{equation*}
Now we define the group isomorphism 
\begin{equation*}
	l:\Z \to K(p)\,,\quad m\mapsto \begin{cases}
			[m]-[0]& \text{ if }m\geq 0\\
			[0]-[-m]& \text{ if }m<0\,.
		\end{cases}
\end{equation*}
This gives the commutative diagramm:
\begin{center}
	\begin{tikzcd}
			0 \arrow[r]& \mathrm{SVect}(X)\arrow[r,"\phi"]&K(X)\arrow[d,"\id"]\arrow[r,"\rank"']&\Z \arrow[d,"l"] \arrow[r] \arrow[l,bend right,"r"']& 0 \\
			0 \arrow[r]& \tilde{K}(X)\arrow[r,"inclusion"]&K(X)\arrow[r,"i^*"]&K(p) \arrow[r] \arrow[l,bend left,"c_p^*"]& 0
		\end{tikzcd}
\end{center}
Where $i$ is the inclusion $\{p\}\to X$ and $c_p$ is the collaps $X\to \{p\}$. Then by some diagram chasing the map 
\begin{equation*}
	\tilde{\phi}: \mathrm{SVect}(X)\to \tilde{K}(X) 
\end{equation*} is well defined and by the five lemma (add zeros the short sequences on the left) it is an isomorphism.
\end{proof} 
\begin{remark}\label{rem:kTheo-elementsInReducedKtheoryAreGivenByEMionusN}
	In the proof above we have seen that every element of $k\in K(X)$ is given as the difference of a vector bundle and a trivial bundle $k=[\xi]-[\underline{n}]$ for some vector bundle $\xi$ and $n\in \N$. We also saw that 
	$\tilde{K}(X)$ is in the kernal of $\rank$ and hence any $k\in \tilde{K}(X)$ is given by $k=[\xi]-[\underline{\rank(xi)}]$ where $\xi$ is a vector bundle.
\end{remark}

\begin{definition}

The convention is that lower indices show for which colimit the maps form a directed system. As an example, the maps $i^n_m$ are used for the colimit over $m$. We start with the the inclusion

		\Big)_m
	\end{equation*} to induce maps $f_n:BU(n)\to BU(n+1)$. which in turn give us a directed system and lets us define:

	\end{equation*} 
\end{definition}
\begin{lemma}
	The maps $f_n:BU(n)\to BU(n+1)$ are closed embeddings.
\end{lemma}
\begin{proof}
	Since $\Gr_n(\C^m)$ is compact and Hausdorf for any $n\leq m$ the maps $f_n^m$ are closed embeddings. Furthermore, the natural maps $\phi^n_m:\Gr_n(\C^m)\to BU(n)$ are also closed embeddings. Now assume that $f_n(x)=f_n(x')$. Then for $r$ big enough (omitting the inclusions) $f_n^r(x)=f_n^r(x')$ which contradicts the injectivity of the $f_n^r(x)$. Now we want to show that $f_n$ hase a closed image. For this we let $C\coloneq \im f_n \subset BU(n+1)$ and call the natural inclusions $\phi^{n+1}_m: \Gr_{n+1}(\C^m)\to BU(n+1)$. Then $C$ is closed, if and only if $C\cap \im \phi_m^{n+1}$ is closed for all $m>n$. To show this we first show that $C\cap \im \phi_m^{n+1}=\im\, \phi^{n+1}_m\circ f_n^{m-1}$. The first inclusion $\subseteq$ is clear. The other inclusion follows directly from the constructino of $f_n$ that satisfies $f_n\circ \phi^n_{m-1}=\phi^{n+1}_m\circ f^{m-1}_n$. By Remark \ref{rem:kTheo-staircaseMapsClosedEmbedding} we know that $\phi^{n+1}_m \circ f^{m-1}_n$ is a closed embedding hence $C\cap \im \phi_m^{n+1}$ is closed for all $m>n$ and hence $C$ is closed. Now finally we need to show that $f_n$ are homeomorphisms onto their image. For this we show that they are closed maps, meaning they map closed sets to closed sets. This property together with injectivity implies the closed embedding property.
	Let $U\subset BU(n)$ be closed. By construction of the direkt limit we can wirte $U=\bigcup_m (\im(\phi^n_m)\cap U)=\bigcup_m (\phi^n_m(U_m))$, where $U_m\coloneq (\phi^n_m)^{-1}(U)$. Now by the same argument as befor we have the equiality $f_n(U)\cap \im\phi^{n+1}_{m+1}= \phi^{n+1}_{m+1}\circ f^m_n(U_m)$. This is closed for all $m$ because the maps $\phi^{n+1}_{m+1}$ and $f_n^m$ are closed embeddings and hence closed. But then $f^n(U)$ is closed in the final topology and hence $f_n$ is an closed embedding.
\end{proof}

\begin{lemma}\label{lemma:kTheo-generalGroupContractsToUnitary}
	The general linear Goup $\GL_n(\C)$ retracts onto the unitary Group $U(n)$ and the retraction commutes with the inclusions $A\mapsto \begin{pmatrix}
		A&0\\
		0&1
	\end{pmatrix}\,.$
\end{lemma}
\begin{proof}
	Let $A\in \GL_n(\C)$. Then $A^*A$ is a positive-definite Hermitian matrix, since for $v \in \C^n$:
	\begin{equation*}
		v^*A^*Av=(Av)^*(Av)=\norm{Av}^2\geq 0\,.
	\end{equation*}  Hence, it is diagonalizable with real non negative eigenvalues $\lambda_1,\cdot \lambda_n>0$. We define $P_1\coloneq \sqrt{A^*A}$ to be the matrix, that scales all eigenspaces by $\sqrt{\lambda_i}$. Then $P^2 =A^*A$, and $\sqrt{A^*A}$ is also positive definite Hermitian, as all eigenvalues are positive and hence, it is invertable. Notice, that 
	\begin{equation*}
		\sqrt{
		\begin{pmatrix}
			A&0\\0&1
		\end{pmatrix}^* 
				\begin{pmatrix}
			A&0\\0&1
		\end{pmatrix}
		}=
				\begin{pmatrix}
			\sqrt{A^*A}&0\\0&1
		\end{pmatrix}\,.
	\end{equation*}
	We then define $U_A\coloneq A  (P_A)^{-1}$. By definition $A=U_A P_A$, and since $P_A$ is Hermitian, $P_A^{-1}$ is also hermitan letting us conclude that $U_A$ is unitary.
	\begin{equation*}
		U_A^*U_A=((P_A)^{-1})^*\underbrace{A^*A}_{P_A^2}  (P_A)^{-1}=\Idm\,.
	\end{equation*} Notice again, that for $\tilde{A}=A\oplus \Idm_1\in GL_{n+1}(\C)$, we have $U_{\tilde{A}}=U_A\oplus \Idm_1\in U(n+1)$. Now the space of positive-definite Hermitian matrizies is convex. This is just a calculation of all the axioms for the matricies on the connecting line of two positive-definit Hermitian matricies. But as a convex space, it is contractible.
\end{proof}
\begin{lemma}\label{lem:kTheo-reducedSuspendedKGroupHomotopically}
	For a connected pointed space $X$ with well sitting basepoint, we have a natural homomorphism
	\begin{equation*}
		\tilde{K}(\Sigma X)\cong [X,U]
	\end{equation*} The group structure in $[X,U]$ comes from viewing $[X,U]$ as the direct limit of abelien groups $\colim[X,\GL_n(\C)]$, where the group structure is derived from the one in $\GL_n(\C)$.
\end{lemma}

\begin{proof}
	If the base point is well sitting, we can apply Corollary \ref{cor:kTheo-reducedSuspensionIsSuspension}, which gives us the homotopy equivalence
	\begin{equation*}
		\Sigma X \simeq S(X)\,.
	\end{equation*} Now $S(X)$ itself can be cut into two contractible cones 
	\begin{equation*}
		\begin{array}{ll}
			C_+(X) \coloneq & \{(x,t)\in S(X) \,|\, t\geq \frac{1}{2} \}\\
			C_-(X) \coloneq & \{(x,t)\in S(X) \,|\, t\leq \frac{1}{2} \} \,.	
		\end{array} 
	\end{equation*}
	Those cones intersect in $\C_+\cap C_-= X\times \{\frac{1}{2}\}\cong X$. Now denote the trivial $n$ bundle over $C_+(X)$ by $\underline{n}_+$ and over $C_-(X)$ by $\underline{n}_-$. With this notation we know by the proof of Lemma \ref{lem:kTheo-gluingAndHomotopyClassesToGLn} that the map
	\begin{align}\label{eq:kTheo-SXtoHomotoy}
		\phi_n: [X,\GL_n(\C)]&\to \Vect_n(\Sigma X)\\
		f&\mapsto \underline{n}_+ \Glue{f} \underline{n}_- \nonumber
	\end{align} in an bijection for all $n$. Furthermore, the inclusion of $\GL_n(\C)$ into $\GL_{n+1}(\C)$ gives an inclsion $[X,\GL_n(\C)]$  into $[X,\GL_n(\C)]$ which commutes with the inclusion $\Vect_n(\Sigma X)$ into $\Vect_{n+1}(\Sigma X)$ unter the bijection (\ref{eq:kTheo-SXtoHomotoy})meaning that for any $n\leq m$ the diagram 
	\begin{center}
        \begin{tikzcd}
            {[X,\GL_n(\C)]} \arrow[r,"\phi_n"] \arrow[d,"i_{nm}"] &\Vect_n(\Sigma X) \arrow[d,"i_{nm}"]\\
            {[X,\GL_m(\C)]} \arrow[r,"\phi_m"]           &\Vect_m(\Sigma X)
\end{tikzcd}
\end{center}commutes. Lemma \ref{lem:kTheo-reducedKisEventually} gives us 
	\begin{equation*}
		\tilde{K}(X)\cong \colim_n [X,\GL_n(\C)]\cong [X,\colim_n\GL_n(\C)]\,.
	\end{equation*} Tha latter bjectino follows from Lemma \ref{lem:kTheo-colimUndHomVertauschen}.
	Finally, Lemma \ref{lemma:kTheo-generalGroupContractsToUnitary} gives us a retraction $\GL_n(\C) \to U(n)$ that commutes with the inclusions and hence we get the same direct limit.
	\begin{equation*}
		[X,\colim_n\GL_n(\C)]\cong [X,\colim_nU(n)]=[X,U]
	\end{equation*} concluding that we have a natural bijection. To get the group strucure we need to look into said mapping. For this we inspect the mapping (\ref{eq:kTheo-SXtoHomotoy}) closer. Let 
	\begin{equation*}
		\Phi_n:[X,\GL_n(\C)]\to \colim \Vect_n(\Sigma X)
	\end{equation*} be the composition of $\phi_n$ with the inclusion $\Vect_n(\Sigma X) \to \colim \Vect_n(\Sigma X)$ By the commutative diagramm above, the maps $\phi_n$ induce a map from $\colim [X,\GL_n(\C)]$ which is a group homomorphism, if all $\Phi_n$ are one. Hence take two functions $f,g:X\to \GL_n(\C)$. then by definition 
	\begin{equation*}
		f\oplus g: X \to GL_n(\C)\,,\quad x\mapsto f(x)\times g(x)
	\end{equation*} Now $f\oplus g$ can be included into $[X,\GL_{2n}(\C)]$:
	\begin{equation*}
		f\oplus g: X \to GL_{2n}(\C)\,,\quad x\mapsto \begin{pmatrix}
			f(x)\times g(x) & 0 \\
			0& \Idm_n\
		\end{pmatrix}\,.
	\end{equation*}
	Now we have that $f\oplus g$ is homotopic to 
	\begin{equation*}
		\widetilde{f\oplus g}:X\to X \to GL_{2n}(\C)\,, \quad 		x \mapsto\begin{pmatrix}
			f(x) & 0 \\
			0& g(x)
		\end{pmatrix} 
	\end{equation*} via the homotopy $\rho_t(x)$ for $t\in (0,\frac{\pi}{2})$:
	\begin{equation} \label{eq:kTHeo-homotopyBetweendirectSumAndProduct}
		\rho_t(x)=
		\begin{pmatrix}	f(x)&0		   \\0&\Idm_n 		  \end{pmatrix}
		\begin{pmatrix}	\cos(t)&\sin(t)\\ \sin(t)&\cos(t) \end{pmatrix}
		\begin{pmatrix} \Idm_n &0	   \\0&g(x)			  \end{pmatrix}
		\begin{pmatrix}	\cos(t)&\sin(t)\\ \sin(t)&\cos(t) \end{pmatrix}
	\end{equation} Finally we can look at the image of $[f\oplus g]=[\widetilde{f\oplus g}]$ under $\Phi_{2n}$ which is given by inclusion of
	\begin{equation*}
		\phi_n([\widetilde{f\oplus g}])=\underline{n}_+\oplus \underline{n}_+ \Glue{\widetilde{f\oplus g}} \underline{n}_-\oplus \underline{n}_-=[\phi_n(f)\oplus \phi_n(g)]
	\end{equation*} into the colimit, where the equation
	\begin{equation*}
	[\phi_n(f)\oplus \phi_n(g)]=[\phi_n(f)]\oplus [\phi_n(g)]
	\end{equation*} holds. The second last equiality follows from the properties of the gluing construction given in Corollary \ref{cor:kTheo-elementaryPropertiesGluingAndClutching}.
\end{proof}

\begin{lemma}\label{lem:kTheo-inversesInKGivenByHomotopicallyMethods}
		Let $Y$ be a pointed space. Define
		\begin{align*}
			T:S^1 &\to S^1 \\
			t     &\mapsto 1-t
		\end{align*} and let $\Sigma T:SY\to SY$ be the map induced by $T$ on $S^1$ and the identity on $Y$. Then:
		\begin{align*}
			(\Sigma T)^*:\tilde{K}(SY)   &\to \tilde{K}(SY)\\
			y                           &\mapsto -y
		\end{align*}
\end{lemma}
\begin{proof}
	We have the following commutative diagramm: 
	\begin{center}
		\begin{tikzcd}
			\tilde{K}(SX) \arrow[d, "(\Sigma T)^*"'] & {\colim[Y,\GL(n,\C)]} \arrow[d, "f\mapsto f^{-1} (pointwise)"] \arrow[l] \\
			\tilde{K}(SX)                             & {\colim[Y,\GL(n,\C)]} \arrow[l]                                         
		\end{tikzcd}
	\end{center} To see why this commutes, notice how we defined the map by gluing via $f$, where $\Sigma T$ changes the place of the upper and lower vector bundle. Hence the gluing function needed is now the inverse. 
	The horizontal maps are natural group homomorphisms and on the right side the inverse is taken. Thereby the commutativity tells us that the left side is also the same as taking the inverse.
\end{proof}
\begin{example}\label{ex:kTheo-LineBundlesSatlquadequalslPlusl}
	Let $l$ and $l'$ be line bundles over $S^2$. Then the clutching functino for the vector bundle $l\otimes l'$ is given by the tensor product of clutching functions. So let $f$ and $f'$ be two clutching functions
	\begin{equation*}
		f,f':S \to 	  \GL_1(\C)
	\end{equation*} Then the product bundle $l\otimes l'$ is given by the clutching function 
	\begin{equation*}
		f\otimes f': S\to \GL(\C\otimes \C)\cong \GL_1(\C)\,,\quad x\mapsto f(x)f'(x)
		\end{equation*}
	Hence, the product in $\tilde{K}(S^2)$ is given by  $f\otimes f'=f\cdot f'$.
	Now, the homotopy (\ref{eq:kTHeo-homotopyBetweendirectSumAndProduct}) of the proof above shows that 
$f\cdot f'\oplus \id_1\simeq f \oplus f'$ and hence $l\otimes l'\oplus \underline{1}\cong l\oplus l'$.
\end{example}

\begin{proposition}
	For compactly generated pointed Hausdoff spaces $(X,x_0),(Y,y_0)$ there is a natural bijection
	\begin{equation*}
		[SX,Y]\to [X,\Omega Y]\,,
	\end{equation*} where $\Omega$ denotes the loop space
	\begin{equation*}
		\Omega Y \coloneq\{f:S^1\to Y\,| \, f \text{ continuous and }f(1)=y_0\}\,.
	\end{equation*} together with the compact open topology.
\end{proposition}
\begin{proof}
 For any three $(X,x_0),(Y,y_0),(Z,z_0)$ we have that a basepoint preserving map 
 $f:X\wedge Y\to Z$ can be lifted to a map $\tilde{f}:X \times Y \to Z;f(z)=\tilde{f}(\overline{z})$ that maps the whole $X\vee Y$ to $z_0$. This map is continuous by the universal property of the quotient topology. Hence we get a bijection
 \begin{equation*}
	C(X\wedge Y,Z)\to \{f\in C(X\times Y,Z)\,|\, \restr{f}{X\vee Y} \equiv z_0\}
 \end{equation*} its inverse is given by the quotient induced maps. 
 We can compose this map with the natural currying map $C(X\times Y,Z)\to C(X,C(Y,Z))$. For $Y=S$ we gain a map 
 \begin{equation*}
	C(SX,Z)\to C(X,\Omega Z)
 \end{equation*} that is again a bijection, since any $f:X\to \Omega Z$ gives a map $\tilde{f}:X\times S\to Z$ $\tilde{f}(x,s)\mapsto f(x)(s)$. If $(x,s)\in X\vee S\subseteq X\times S$, then either $x=x_0$ or $s=1$. In the first case. $f(x_0)$ is the constant loop and hence $f(x_0)(s)=z_0$ and in the second case we evaluate a based loop at its base and hence $f(x)(1)=z_0$. Hence, the $\tilde{f}$ factors over $X\vee S$ giving a well defined element in $C(SX,Z)$. This is its inverse. The continuoty part of the argument follows from Proposition \ref{prop:kTheo-curryingIsContinuous}.
\end{proof}

\begin{lemma} \label{lem:kTheo-exactShortPartForThePair}
	Let $(X,Y)\in ob(\TopCPair)$ and define $i:(Y,\emptyset) \to (X,\emptyset)$ and $j:(X,\emptyset)\to(X,Y)$ to be the inclusions. Then we have an exact sequence:
	\begin{center}
		\begin{tikzcd}
			{K(X,Y)} \arrow[r, "j^*"] & K(X) \arrow[r, "i^*"] & K(Y)
		\end{tikzcd}
	\end{center}
	Here, we identify $K(X)=K(X,\emptyset)$ and the same with $K(Y)=K(Y,\emptyset)$ such that we have well defined induced functions from the same functor in the sequence. This is just a formall note, as $K(X,\emptyset)\cong K(X)$ by Corollary \ref{cor:kTheo-K(X)istK(X,nix)}.
\end{lemma}
\begin{proof}
	First, we show that $\im(i^*)\subseteq \ker(j^*)$. To see this, notice that $i^*\circ j^*=(j\circ i)^*$, and $j\circ i$ factors over $(Y,Y)$:
	\begin{center}
		\begin{tikzcd}
			{(Y,\emptyset)} \arrow[r, "i"] \arrow[d] & {(X,\emptyset)} \arrow[d, "j"] \\
			{(Y,Y)} \arrow[r]                        & {(X,Y)}                       
		\end{tikzcd}
	\end{center}
	Hence, $j^*\circ i^*$ factors over the trivial ring $K(Y,Y)$ and thereby is trivial.
	Now suppose $k\in \ker(i^*)$. 
	The intuition now is that if we have a vector bundle that gets mapped to $0$ this means that it is trivial over $Y$ which lets us define a vector bundle over the quotient $X\slash Y$ by a trivialization: 
	We write $k=[\xi]-[\underline{n}]$ for $\xi$ being a vector bundle over $X$. (Compare Remark \ref{rem:kTheo-elementsInReducedKtheoryAreGivenByEMionusN}) Since $i^*\xi=0$ it follows that $i^*([\xi]-[n])=[\restr{\xi}{Y}]-[\restr{\underline{n}}{Y}]=0$ and hence $E$ and $n$ are stably equivalent over $Y$ giving us an $m$ such that 
	\begin{equation*}
		\restr{\left(\xi\oplus \underline{m}\right)}{Y}\cong \restr{\xi}{Y}\oplus\underline{m}\cong \underline{n}\oplus \underline{m}
	\end{equation*}
	Hence, there is a trivialization $\alpha: \restr{\left(\xi\oplus \underline{m}\right)}{Y} \to Y\times \C^{\rank(\xi)+m}$ that defines a bundle $\restr{\left(\xi\oplus \underline{m}\right)}{Y} \big/\alpha$ as shown in the proof of Lemma \ref{lem:kTheo-trivialisationDefinesABundle} over $X\big/Y$. Now we define 
	\begin{equation*}
		\eta=[\restr{\left(\xi\oplus \underline{m}\right)}{Y} \big/\alpha]-[\underline{n+m}]\in \tilde{K}(X\big/Y)=K(X,Y)
	\end{equation*} and sine the trivialization acts as a right inverse of the pullback of the restriction (compare Corollary \ref{cor:kTheo-trivializationIsRightInverse})
	\begin{equation*}
		j^*(\eta)=[\xi \oplus \underline{m}]-[\underline{n+m}]=k\,.
	\end{equation*} 
\end{proof}

\begin{lemma}\label{lem:kTheo-pointedUnionExactSequence}
	Let $X,Y$ be pointed topological spaces and define the maps
	\begin{equation*}
		i:X\to X\vee Y\,\text{ the inclusion, and }j:X\vee Y\to Y \text{ the collaps of $X$.}
	\end{equation*} Then the sequence
	\begin{center}
		\begin{tikzcd}
			0 \arrow[r]&\tilde{K}(X)\arrow[r,"j^*"]&\tilde{K}(X\vee Y)\arrow[r,"i^*"]&\tilde{K}(Y) \arrow[r]&0
		\end{tikzcd}
	\end{center} is exact and naturally splits via the map $q^*$, that is a right inverse of $j^*$, where 
	$
		q:X\vee Y \to Y 
	$is the collaps.
\end{lemma}
\begin{proof}
	Notice how there is a natural homeomoprhism $X\cong (X\vee Y)\big/ Y$.  Hence, Lemma \ref{lem:kTheo-exactShortPartForThePair} provides the exactnes at $K(X\vee Y)$. Now the map $q$ is a left inverse of $j$ as continuous maps, and hence $q^*$ is a right inverse of $j^*$ in the K group, proofing the injectivity of $j^*$. now only the surjectivity of $i^*$ is missing. However the inclusion $l:Y\to X\vee Y$ is a topological right inverse of $i$ and thereby $l^*$ is a left inverse of $i^*$ giving us the injectivity and hence prooving the exactness and splitting of the short exact sequence.
\end{proof}
\begin{theorem}[Pointed Additivity] \label{thm:kTheo-pointedAdditivity}
	Let $X,Y$ be pointed topological spaces and denote the inclusions of $X$ (resp. $Y$) into $X\vee Y$ by $i_x$ (resp. $i_Y$). Then the map 
	\begin{equation*}
		i_X^*\oplus  i_Y^*:\tilde{K}(X\vee Y)\to \tilde{K}(X)\oplus \tilde{K}(Y)
	\end{equation*} is a natural isomorphism with inverse 
	\begin{equation*}
		\psi: \tilde{K}(X)\oplus \tilde{K}(Y) \to \tilde{K}(X\vee Y)\,,\quad (x,y)\mapsto c_X^*(x)+c_Y^*(y)\,,
	\end{equation*}where $c_X$ (resp. $c_Y$) denotes the collaps of $X$ (hence onto $Y$) respectifly the collaps of $X$ and hence onto $Y$.
\end{theorem}
\begin{proof}
	This is a direct consequence of Lemma \ref{lem:kTheo-pointedUnionExactSequence} and the algebraic mashinery of splitting short exact sequences.
\end{proof}
\begin{corollary}[Unpointed Additivity] \label{thm:kTheo-unpointedAdditivity}
	Let $X,Y$ be topological spaces, then there is a natural isomorphism 
	\begin{equation*}
		K(X\sqcup Y)\to K(X)\oplus K(Y)
	\end{equation*}
\end{corollary}
\begin{proof}
	This follows from $K(A)\cong \tilde{K}(A,[\emptyset])$ for all topological spaces. Furthermore, $(X\sqcup Y,[\emptyset])\cong ((X,[\emptyset])\vee (Y,[\emptyset]))$. Hence, using those isomorphisms, the statement is a direct consequence of the pointed additivity.
\end{proof}

\begin{definition}\label{def:kTheo-coneOverSubspace}
	Let $X$ be a topological space and $I=[0,1]$ be the unit intervall. We define the cone over $X$ to be the space
	\begin{equation*}
		CX\coloneq (X\times I)\big/ X\times\{0\}\,.
	\end{equation*}
	Let $(X,Y)$ be a pair of compact topological spaces. We define the cone over the subspace $Y$ to be 
	\begin{equation*}
		\Big(X\sqcup CY\Big)\Big/ Y\sim Y\times \{ 1\}
	\end{equation*} and denote that with $(X\cup CY)$.
\end{definition}
\begin{proposition}
	If we define for $f:X\to Y$ the induced maps on the cones $Cf$ by
	\begin{equation*}
		Cf:CX\to CY\,\quad \overline{(x,t)}\mapsto \overline{(f(x),t)},
	\end{equation*}the cone defines a functor $\Top\to \TopP$. 
	The cone over $Y\subseteq X$ defines in similar fashion an functop from $\mathbf{CGWH}^2_{\mathbf{cof}}$ to $\TopP$.
	There is a natural isomorphism:
	\begin{equation*}
		T:(X\cup CY)\big/ X \to CY\big/Y
	\end{equation*} Here, the left and right hand side both define Functors from $\mathbf{CGWH}^2_{\mathbf{cof}}$ to $\TopP$ with the obvious morphisms. Furthermore, we have that $CY\big/Y$ is homotopically equivalent to $\Sigma Y$ which is homotopically equivalent to $SY$ as long as $(Y,y_0)$ is well pointed. All the equivalences are natural up to homotopy.  
\end{proposition}

\end{corollary}

\begin{definition}[Connecting Homomorphism]\label{def:kTheo-connectingHomomorphism}
Let $(X,Y)$ be a topological pair of compact spaces and $Y$ be pointed. Define the two maps
\begin{align}
	m: (X \cup CY,\emptyset)&\to (X\cup CY,X)\\
	\pi: (X\cup CY,\emptyset)& \to X\big/Y
\end{align} where $m$ is the inclusion in the second space, and $\pi$ is the collaps of $CY$. Since $CY$ is contractible, the induced map $\pi^*:\tilde{K}(X\big/ Y)\to K(X \cup CY,\emptyset)$ is an isomorphism by Lemma \ref{lem:kTheo-collapsingOfContractibleSubspace}. 	\begin{equation*}
		\theta: K(X\cup CY,Y)\to \tilde{K}(SY)=\tilde{K}^{-1}(Y)\,.
	\end{equation*}

	\begin{equation*}
		\delta\coloneq (\pi^*)^{-1}\circ m^*\circ \theta^{-1}:\tilde{K}^{-1}(Y)\to K(X,Y)

\end{definition}

	\begin{lemma}\label{lem:kTheo-doubleConeFlip}
		Let $Y$ be a wellpointed space and denote the points of $CY$ and $\Sigma Y$ by $(x,t)$ such that $CY$ has the apex $[\{0\}\times Y]$ and base $\{1\}\times Y\cong Y$. Let 
		\begin{equation*}
			D\coloneq C_1Y \cup C_2Y = \big(C_1Y \sqcup C_2 Y\big) \Big/ \big((x,1)_1\sim (x,1)_2\big)
		\end{equation*} denote the double cone glued along a common base. Now define the maps
		\begin{itemize}
			\item $c_i:D \to C_i Y \big/Y$ the collaps of the complemantary cone $C_jY$ $j\neq i$.
			\item $\overline{\pi_1}:C_iY\big/Y \to \Sigma Y\,\quad [(x,t)]\mapsto (x,t)$.
			\item $p_i\coloneq \overline{\pi_1}\circ c_i: D\to \Sigma Y$.
			\item $R:D\to D\,\quad (x,t)_i\mapsto (x,t)_j$ where $j\neq i$, the cone swap.
			\item $\Sigma T: \Sigma Y \to \Sigma Y\,,\quad (x,t)\mapsto (x,1-t)$ the suspension flip.
			\item $\Phi: D\to \Sigma Y\,,\quad (x,t)_1\mapsto (x,1-\frac{t}{2}),\quad (x,t)_2\mapsto (x,\frac{t}{2})$.
		\end{itemize} Then we have have that 
		\begin{equation*}
			p_2=p_1\circ R\quad \text{and}\quad \Phi\circ R=\Sigma T\circ \Phi\,.
		\end{equation*}
	\end{lemma}
	\begin{proof}[Proof of Lemma \ref{lem:kTheo-doubleConeFlip}]
		The first statement $p_2=p_1\circ R$ is clear, as:
		\begin{align*}
			p_2(x,t)_2&=[(x,t)]=p_1(x,t)_1=p_1\circ R (x,t)_2\,,\quad \text{ and }\\
			p_2(x,t)_1&=[\{1\} \times Y]=p_1(x,t)_2=p_1 \circ R (x,t)_1\,.
		\end{align*}
Similar, the second statement is also just a calculation:
\begin{align*}
	\Sigma T\circ \Phi (x,t)_1&=\Sigma T (x,1-\frac{t}{2})=(x,\frac{t}{2})=\Phi(x,t)_2=\phi \circ R (x,t)_1 \\
	\Sigma T\circ \Phi (x,t)_2&=\Sigma T (x,\frac{t}{2})=(x,1-\frac{t}{2})=\Phi(x,t)_1=\phi \circ R (x,t)_2\,.
\end{align*}
\end{proof}
By the lemma above, we know that $\Sigma T$ makes the diagramm commute and by Lemma \ref{lem:kTheo-inversesInKGivenByHomotopicallyMethods} we know that 
\begin{equation*}
	(\Sigma T)^* :\tilde{K}(SY)\to \tilde{K}(SY)
\end{equation*} is multiplication by minus one. Hence, the diagramm where all maps are either topologically induced or topologically induced and inverted:

			&\tilde{K}(C_1Y\cup C_2Y) \arrow[dr]&                            &                                                         \\
			\tilde{K}^{-1}(X) \arrow[uu,"\rho"] \arrow[rr,"i^*"]& & \tilde{K}(C_2Y\slash Y) \arrow[r] & \tilde{K}^{-1}(Y)      \arrow[uu, equal, Rightarrow,"(-1)^*"]                            
		\end{tikzcd}
	\end{center} is commutative. By construction (just composition of collapsing maps) we have that 
	\begin{equation*}
		l^*=\rho \circ {\psi}_1\,,\quad  \text{ and by the diagram } \quad {\psi}_2\circ \rho  =- i^*\,.
	\end{equation*}
	 Which gives us in total:
	 \begin{equation*}
		i_*\circ {\psi}_1=-{\psi}_2 \circ \rho \circ {\psi}_1=-{\psi}_2 \circ l^* 
	 \end{equation*} and thereby the diagramm commutes up to sign:
	\begin{center}
		\begin{tikzcd}
			{K\big((X\cup C_1Y)\cup C_2X,X\cup C_1Y\big)} \arrow[r, "l^*"] \arrow[d,"{\psi}_1"] & \tilde{K}\big((X\cup C_1Y)\cup C_2X\big) \arrow[r, "s^*"] \arrow[d,"{\psi}_2"] & \tilde{K}(X\cup C_1Y) \arrow[d,"{\psi}_3"] \\
			\tilde{K}^{-1}(X) \arrow[r, "i^*"]                                      & \tilde{K}^{-1}(Y) \arrow[r, "\delta"]                               & {K(X,Y)}                       
		\end{tikzcd}
	\end{center} The exactnes in the top row now prooves exactnes in the bottom.
\end{proof}




Bevor we start constructing a Cohomology theory, we extend the K groups to be defined not oly over the negativs, but over the whole $\Z$ which is done via an extension. The extension relies on periodic behaviour. Socalled Bott periodicity.
\begin{definition}[External Product]
	The \textbf{external product } $\mu:K(X)\otimes K(Y)\to K(X\times Y)$ is defined via the two projections $\pi_X:X\times Y\to X$ and $\pi_Y:X\times Y\to Y$ by:
	\begin{align*}
		\mu:K(X)\otimes K(Y)&\to K(X\times Y)\\
		(a\otimes b)&\mapsto (\pi_X)^*(a)\cdot (\pi_Y)^*(b) \, ,
	\end{align*} where $\cdot$ denotes the multiplication in the K-group, which was induced by the tensor Product. This $\mu$ is a ring homomorphism, since:
	\begin{align*}
		\mu((a \otimes b)(c \otimes d)) &= \mu(ac \otimes bd) \\
		&= \pi_X^*(ac)\pi_Y^*(bd) \\
		&= \pi_X^*(a)\pi_X^*(c)\pi_Y^*(b)\pi_Y^*(d) \\
		&= \pi_X^*(a)\pi_Y^*(b)\pi_X^*(c)\pi_Y^*(d) \\
		&= \mu(a \otimes b)\mu(c \otimes d)
	\end{align*}
\end{definition}
By example \ref{ex:kTheo-LineBundlesSatlquadequalslPlusl} we know that for any line bundle $l$, $[l\otimes l \oplus \underline{1}]=[l]^2+[\underline{1}]=[l]+[l]$ which is equivalent to $([l]-1)^2=[l^2]-[l]+[l]+[\underline{1}]=0$. Now $\{[\underline{1}],[l]\}$ is an addative basis of $\Z\big/([l]^2-[\underline{1}])$. The mapping that is defined on a basis of the tensor product as follows
	\begin{align*}
		K(X)\otimes \Big(\Z([l])\Big/ ([l]^2-[\underline{1}])\Big)&\to K(X\times S^2)\\
		[\xi]\otimes [\underline{1}] &\mapsto \mu([\xi], [\underline{1}])\\
		[\xi]\otimes [l] &\mapsto \mu([\xi], [l])
	\end{align*} is well defined. 
\begin{theorem}[The Fundamental Product Theorem]\label{thm:kTheo-fundamentalProductTheorem}
	For any compact $X$, and the tautological line bundle $l$ on $\C\PP^1\cong S^2$, the mapping 
	\begin{equation*}
		K(X)\otimes \Big(\Z([l])\Big/ ([l]^2-[\underline{1}])\Big)\to K(X\times S^2)
	\end{equation*}is an isomorphism of rings. 
\end{theorem}
\begin{proof}
	The proof is omitted here. There are elementary prooves given in Chapter 2.1 \cite{hatcher2002algebraic} or Chapter 2.3 \cite{atiyah1989k}. 
\end{proof}
\begin{corollary}
	If we take $X$ to be a point, then Theorem \ref{thm:kTheo-fundamentalProductTheorem} implies that
	\begin{equation*}
		\Phi: \Big(\Z([l])\Big/ ([l]^2-[\underline{1}])\Big)\to K(S^2)
	\end{equation*} is an isomorphism. Hence, $K(S^2)$ is generated by $\{[\underline{1}],[l]\}$. Furtermore together with the map 
	\begin{equation*}
		\rank: \Big(\Z([l])\Big/ ([l]^2-[\underline{1}])\Big)\to \Z
	\end{equation*} that maps the generatort to their rank, and 
	\begin{equation*}
		\nu: \Z \to K(pt);\,,\quad n \mapsto [\underline{n}]
	\end{equation*} we get the commutative diagramm:
	\begin{center}
	\begin{tikzcd}
		{\Z([l])\Big/ ([l]^2-[\underline{1}]) }\arrow[r,"\rank"] \arrow[d,"\Phi"] & \Z\arrow[d,"\nu"]\\
 		K(S^2) \arrow[r,"i_{pt}^*"] & K(pt) 	
	\end{tikzcd}
	\end{center}
	By the commutativity, we get a isomoprhism induced by $\Phi$: 
	\begin{equation*}
		\tilde{K}(S^2)=\ker(i_{pt}^*) \cong \ker(rank)=\mathrm{span}([H]-[\underline{1}])
	\end{equation*}
\end{corollary}

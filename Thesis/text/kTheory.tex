\subsection{Classification of Vector Bundles}
The first goal is to classify all vector bundles over sufficiently nice spaces. The latter refers to spaces admitting partitions of unity for which Tietze's extension theorem holds. For simplicity we just consider compact Hausdorff spaces. Along the way we will discover that the set of isomorphism classes of vector bundles is a homotopy invariance.

Before we go back to the analytical reals, we will further develop the theory of vector bundles:
\begin{definition}
	For a topological space $X$ we define $\Vect_n$ to be the set of isomorphism classes of complex vector bundles of rank $n$ and  $\Vect(X)$ to be the set of isomorphism classes of complex vector bundles of any rank. By declaring $\Vect(f)=f^*$ to be the pullback we get a contravariant functor $\Vect$ from $\Top$ to $\Set$. We can equip the set $\Vect(X)$ with the direct sum and tensor product operation. It is then easy to see that these operations descend to isomorphism classes and hence we get an abelian semi ring $(\Vect(X),\oplus,\otimes)$ (abelian semigroup with associative and distributive multiplication). In fact we even get a contravariant functor from $\Top$ to the category of abelian semirings.
\end{definition}
\begin{lemma}\label{lem:kTheo-sectionExtension}
	Let $X$ be a compact Hausdorff space, $Y\subseteq X$ a closed subspace and $E$ a vector bundle over $X$. Then any section $s:Y\to \restr{E}{Y}$ can be extended to a global section of $X$.
\end{lemma}
\begin{proof}
	For a detailed proof see Chatper 1.4 of \cite{atiyah1989k}. The general argument however is easy to follow: By Tietze's extension theorem we can extend the section to arround every $x\in X$. By compactness we can pick finitely many of those extensions. Now using a partition of unity with respect to their domains we can build the sum of all the extensions to get a well defined global one. 
\end{proof}
\begin{corollary}\label{cor:kTheo-isomorphismExtension}
	Let $Y$ be a closed subspace of a compact Hausdorff space $X$, and let $E,F$ be bundles over $X$. If $f:\restr{E}{Y}\to \restr{F}{Y}$ is an isomorphism, then there is an open neighbourhood $U\supset Y$ and an extension $f:  \restr{E}{U}\to \restr{F}{U}$ which is an isomorphism.
\end{corollary}
\begin{proof}
	By corollary \ref{cor:vb-sectionsIntoHom} we can view $f$ as a section into the bundle $\Hom(\restr{E}{Y},\restr{F}{Y})$, which can be extended to a global one in $\Hom(E,F)$. Let $U$ be the open set for which the section maps to isomorphisms. Since $U$ contains $Y$ this yields the require extension.
\end{proof}
\begin{theorem}[Homotopy Invariance of the Pullback]\label{thm:kTheo-homotopyInvariance}
	Let $X,Y$ be compact Hausdorff spaces, $I=[0,1]$, and 
	\begin{equation*}
		f: Y\times I\to X
	\end{equation*} be a homotopy. For a fixed $t\in I$, we define the function $f_t:Y\to X$, $f_t(y)=f(t,y)$. Let $\pi: E\to X$ be a vector bundle. Then 
	\begin{equation*}
		f_0^*(E)\cong f_1^*(E)\,.
	\end{equation*}
\end{theorem}
\begin{proof}
	Consider the following two vector bundles over $Y\times I$:
	\begin{equation*}
		f^*E \quad \text{and} \quad \pi_1^*f_t^*E \,,
	\end{equation*}where $\pi_1:Y\times I\to Y$ denotes the projection and $t\in I$. 
Since the restriction is by definition the pullback along the inclusion 
$	i:Y\times \{t\}\to Y\times I\,.$ The equality of the two maps:
\begin{equation*}
	f\circ i = f_t \circ \pi_1 \circ i :Y\times\{t\} \to X\,
\end{equation*} together with the functoriality gives us an isomorphism 
\begin{equation*}
\Phi:	\restr{f^*E}{Y\times\{t\}} \to \restr{\pi_1^*f_t^*E}{Y\times\{t\}}\,.
\end{equation*}
By Corollary \ref{cor:kTheo-isomorphismExtension} we can extend the isomorphism  for some open neighbourhood of $Y\times \{t\}$ and by the product topology we can deduce that there exists a $\delta_t$ such that $U$ contains the thickened line $Y\times (t-\delta_t,t+\delta_t) $. The isomorphism class of $\restr{f^*E}{Y\times \{t\}}$ as a bundle over $Y$ is locally constant in $t$: Let $0<\epsilon<\delta$ and $r\in I$. Then define the inclusions
	\begin{equation*}
			\begin{array}{rl}
				i_t: Y&\to Y\times \{t\}\subseteq Y\times I \\
				y &\mapsto (y,t)
			\end{array}
		\quad\quad \text{and}\quad \quad
			\begin{array}{rl}
				i_{t+\epsilon}:Y&\to Y\times (t-\delta_t,t+\delta_t)\\
				y &\mapsto (y,t+\epsilon)
			\end{array}
	\end{equation*} Then we have the identities $f_t=f\circ i_t$ and $\pi_1\circ i_{t+\epsilon}=\id_Y$ and we can calculate:
	\begin{equation*}
		f^*_{t+\epsilon}E=i_{t+\epsilon}^*f^*E=i_{t+\epsilon}^*\pi_1^*f_t^*E=(f_t\circ \pi_1\circ i_{t+\epsilon})^*E=f_t^*E\,.
	\end{equation*} Finally, since $I$ is connected, any locally constant map from $I$ is globally constant and therefore $f_0^*E\cong f_1^*E$.
\end{proof}

\begin{corollary}
		\label{lem:kTheo-contractibleBundlesAreTivial}
	\begin{enumerate}
		\item If $f:X\to Y$ is a homotopy equivalence, $f^*:\vect(Y)\to \vect(X)$ is a semiring isomorphism.
		\item If $X$ is contractible, every bundle over $X$ is trivial and $\vect(X)\cong \N_0$
	\end{enumerate}
\end{corollary}

\begin{definition}[Trivialization of a Vector Bundle]
		\label{def:kTheo-trivialization}
	Let $Y$ be a closed subspace of $X$, $E$ a vector bundle over $X$ and $\alpha:\restr{E}{Y}\to Y\times V$ be an isomorphism. Then we call $\alpha$ \textbf{trivialisation of $E$ over $Y$}.
\end{definition} 



\begin{lemma}\label{lem:kTheo-trivialisationDefinesABundle}
	A trivialization $\alpha$ of a bundle $E$ over $Y\subseteq X$ defines a bundle $E\big/ \alpha$ over $X\big/ Y$. The isomorphism class of $E\big/ \alpha$ depends only on the homotopy class of $\alpha$. 
\end{lemma}
\begin{proof}
	We define the bundle $E/ \alpha $ as follows: Let $\pi_2: Y \times V \to V$ denote the projection and define an equivalence relation on $\restr{E}{Y}$ by 
	\begin{equation*}
		e\sim e' ~ \Leftrightarrow \pi_2 \alpha (e) =\pi_2 \alpha (e') \, .
	\end{equation*} We extend this relation by the identity on $\restr{E}{X\setminus Y}$ and we define $E\big/\alpha$ to denote the quotient space of $E$ defined by that relation.
	To see why this is a bundle we need to check the local triviality only around the base point $Y \big/ Y$ of $X \big/ Y$. This follows from the fact that by Corollary \ref{cor:kTheo-isomorphismExtension} we can extend $\alpha$ to an isomorphism $\Tilde{\alpha}:\restr{E}{U}\to U \times V$. Then $\Tilde{\alpha}$ induces an isomorphism
	\begin{equation*}
		\restr{E \big/ \alpha}{U\slash Y} \cong  (U \big/ Y) \times V
	\end{equation*}
	proving the local triviality.
	Now for the homotopy invariance of said bundle suppose that $\alpha_0$ and $\alpha_1$ are homotopic trivializations of $E$ over $Y$. This means that we have a trivialization $\beta$ of $E \times I$ over $Y \times I\subset X \times I$ that induces the $\alpha_i$ at the endpoints. Let $f: (X \big/ Y)\times I\to (X\times I) \big/ (Y\times I)$ be the natural map. Then $f^*((E \times I) \big/  \beta)$ is a bundle on $(X \big/ Y)\times I$ whose restriction to $(X \big/  Y)\times \{i\}$ is $E \big/  \alpha_i$ ($i=0,1$). Hence by the homotopy invariance of the pullback (Theorem \ref{thm:kTheo-homotopyInvariance}) we have 
	\begin{equation*}
		E \big/ \alpha_0 \cong E \big/ \alpha _1\,.
	\end{equation*}
\end{proof}
\begin{lemma}
	\label{lem:kTheo-collapsingOfContractibleSubspace}
	Let $Y\subseteq X$ be a closed contractible subspace of a compact Hausforff space $X$. Then 
	\begin{equation*}
		f:X\to X\big/ Y \quad \text{induces a semiring isomorphism}\quad f^*: \Vect(X\big/Y) \to \Vect(X)\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Since $Y$ is contractible, $\restr{E}{Y}$ is trivial and thereby a trivialization $\alpha:\restr{E}{Y}\to Y\times V$ exists. Two such trivializations only differ by the automorphism  of $Y\times V$ which in turn corresponds to a map $Y\to \GL(V)\cong \GL_{\rank(E)}(\C)$. Now since $Y$ is contracible those two maps factor (up to homotopy) over any one point $y\in Y$ and are thereby homotopically equivalent to the constant map. Finally since $\GL_{\rank(E)}(\C)$ is connected all constant maps are homotopically equivalent. Thereby $\alpha$ is unique up to homotopy. By Lemma \ref{lem:kTheo-trivialisationDefinesABundle} the isomorphism class of $E\big/ \alpha$ only depends on the homotopy class of $\alpha$ we have a well defined map 
	\begin{equation*}
		\Vect(X)\to \Vect(X\big/ Y)\,,
	\end{equation*}which in turn is an inverse of $f^*$ concluding the proof.
\end{proof}
\begin{definition}[Gluing and Clutching of Vector Bundles]
	\label{def:kTheo-gluingAndClutching}
	Let 
	\begin{align*}
		X=X_1 \cup X_2\, , \quad A=X_1 \cap X_2 \, ,
	\end{align*}
	and all be compact Hausdorff spaces. 
	 Assume that $E_i$ is a vector bundle over $X_i$ and $\phi :\restr{E_1}{A} \to \restr{E_2}{A}$ is an isomorphism. Then we define a vector bundle $E_1 \cap_{\phi} E_2$ on $X$ as follows:
	 We topologize the space $E_1 \cup_{\phi} E_2$ as the quotient of $E_1 + E_2 \slash \sim$, where $\sim$ is the identification of points with their image under $\phi$.
	 Identifying $X$ with $X_1 +X_2 $ in the quotient space we get the natural projection $p:E_1 \cap_{\phi} E_2 \to X$ and the fibers have a natural vector space structure. It remains to show that this is locally trivial. This is obvious outside of $A$ so let $a\in A$ and $V_1$ be a closed neighbourhood of $a$ in $X_1$ such that $\restr{E_1}{V_1}$ is trivial giving us the isomorphisms
	 	\begin{align*}
	 	\theta_1: \restr{E_1}{V_1} &\to V_1 \times \C^n \, \\
	 	\theta_1^A: \restr{E_1}{V_1\cap A} &\to (V_1\cap A) \times \C^n \,
	 \end{align*}
	 	Finally, define
	 \begin{equation*}
	 	\theta^A_2: \restr{E_2}{V_1\cap A}\to (V_1\cap A) \times \C^n \,
	 \end{equation*} by composition $ \theta_1^A \circ \phi^{-1}$. Let $V_2$ be a neighborhood of $a$ in $X_2$ such that 
	 \begin{align*}
	 	\theta_2:\restr{E_2}{V_2}\to V_2 \times \C^n
	 \end{align*} is an extension of $\theta_2^A$. This extension exists again by the extending the section into the homomorphism bundle corresponding to the isomorphism $\theta_2^A$. Finally $\theta_1$ and $\theta_2$  defines a well defined isomorphism 
	 	\begin{equation*}
	 	\theta_1 \cap_{\phi} \theta _2: E_1 \cap_{\phi} E_2 \to (V_1 \cap V_2)\times \C^n\, .
	 \end{equation*} This proves the local triviality.
	 
	 We call such a triple $(E_1,E_2,\phi)$ a \textbf{clutching datum} and $\phi$ the \textbf{transition funciton}.
\end{definition}

\begin{corollary}[Elementary Properties of the Gluing and Clutching Construction]\label{cor:kTheo-elementaryPropertiesGluingAndClutching}
	Elementary properties of the gluing and clutching construction are:
	\begin{enumerate}
		\item If $E$ is a bundle over $X$, then the identity defines ans isomorphism and 
		\begin{equation*}
			E_1 \cup_{\id_A}E_2 \cong E.
		\end{equation*}
		\item If $\beta_i:E_i \to E_i'$ are isomorphisms on $X_i$ and $\phi' \beta_1=\beta_2\phi$, then  
		\begin{equation*}
			E_1\cup_{\phi}E_2 \cong E'_1\cup_{\phi'}E'_2 \, .
		\end{equation*}
		\item If $(E_i,\phi)$ and $(E'_i,\phi')$ are two "clutching data" on $X_i$, then 
		\begin{align*}
			(E_1 \cup _{\phi}E_2)\oplus (E'_1 \cup _{\phi'}E'_2)&\cong E_1\oplus E'_1 \bigcup_{\phi \oplus \phi'}E_2\oplus E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)\otimes (E'_1 \cup_{\phi'}E'_2)&\cong E_1\otimes E'_1 \bigcup_{\phi \otimes \phi'}E_2\otimes E'_2 \, ,\\
			(E_1 \cup _{\phi}E_2)^* &\cong E^*_1\bigcup_{(\phi^*)^{-1}}E^*_2 \, .
		\end{align*} $(-)^*$ denotes the dual here.
	\end{enumerate}
\end{corollary}

\begin{lemma}
	\label{lem:kTheo-homotopyInvarianceClutchingFunction}
	The isomorphism class of $E_1\cup_\phi E_2$ depends only on the homotopy class of the isomorphism $\phi:\restr{E_1}{A}\to \restr{E_2}{A}$\,.
	\end{lemma}
\begin{proof}
	A homotopy of isomorphisms is a map 
\begin{equation*}
	\Phi: \restr{E_1}{A} \times I\to \restr{E_2}{A}\,,
\end{equation*} such that for all $t$ the map 
\begin{equation*}
	\Phi_t: \restr{E_1}{A} \to \restr{E_2}{A}\quad \Phi_t(\xi)\coloneq \Phi(\xi,t)
\end{equation*} is a bundle isomorphism. We can encode this $\Phi$ into a bundle isomorphism as follows: Let $\pi: X\times I\to X$ be the projection. Then elements $\xi\in \pi^*E_1$ are by definition \ref{def:vb-pullbackOfVerctorBundles} of the form $\xi_{x,t}=((x,t),\xi_x)\in (\pi^*E_1)_{x,t}$. With this notation define the bundle isomorphism: 
\begin{equation*}
	\Psi: \restr{\pi^*(E_1)}{A\times I} \to   \restr{\pi^*(E_2)}{A\times I}\,,\quad 	\Psi(\xi_{x,t})\coloneq ((x,t),\Phi_t( \xi_x))
\end{equation*}
This is a bundle isomorphism, since $\Phi_t$ is one. Now denote the inclusion 
\begin{equation*}
	i_t: X \to X \times I\,,\quad i_t(x)\coloneq (x,t)\,,
\end{equation*}and notice that the pullback along $i_t$ is the same as restricting to $X\times\{t\}$. 

With those definitions it is clear by the construction of the gluing, that 
\begin{equation}\label{eq:kTheo-gluingHomotopyInvariant}
	E_1\,\cup_{\Phi_t}\,E_2\,\cong\, i_t^*\left(\pi^*(E_1)\, \cup_{\Psi}\, \pi^*(E_2)\right)\,.
\end{equation} The isomorphism can be seen by part (2) of Corollary  \ref{cor:kTheo-elementaryPropertiesGluingAndClutching}. as follows: 
First notice the isomorphism $\restr{\pi^*(E_i)}{A\times \{t\}}\cong E_i$ for $i=1,2$ and any $t$. By construction of the gluing this leads to the isomorphism 
\begin{equation*}
		E_1\,\cup_{\Phi_t}\,E_2\,\cong\,	\restr{\pi^*(E_1)}{A\times \{t\}} \, \cup_{(\Phi_t,\id)} \, \restr{\pi^*(E_2)}{A\times \{t\}}\,.
\end{equation*} Now again by construction we notice, that gluing restricted spaces is the same as gluing the whole space and then restricting. Furthermore the map $(\Phi_t,\id)$ is the restriction of $\Psi$ to $X\times\{t\}$. Therefore: 
\begin{equation*}
	\restr{\pi^*(E_1)}{A\times \{t\}} \, \cup_{(\Phi_t,\id)} \, \restr{\pi^*(E_2)}{A\times \{t\}}\,\cong \, 	\restr{\left(\pi^*(E_1) \cup_{\Psi} \pi^*(E_2)\right)}{X\times\{t\}}
\end{equation*} Finally, pulling back via $i_t$ is the same as restricting $X\times i$ to $X\times \{t\}$ and identifying $X$ with $X\times \{t\}$. This concludes the chain of arguments:
\begin{equation*}
	\restr{\left(\pi^*(E_1) \cup_{\Psi} \pi^*(E_2)\right)}{X\times\{t\}} \, \cong \,
	i_t^*\left(\pi^*(E_1)\, \cup_{\Psi}\, \pi^*(E_2)\right)\,.
\end{equation*}
Since the pullback is homotopy invariant we can conclude by the isomorphism (\ref{eq:kTheo-gluingHomotopyInvariant}):
\begin{equation*}
	E_1\,\cup_{\Phi_0}\,E_2\,\cong\,E_1\,\cup_{\Phi_1}\,E_2\,.
\end{equation*}
\end{proof}
\begin{definition}\label{def:kTheo-homotopyClassesOfMaps}
	Let $X,Y$ be topological spaces. By $[X,Y]$ we denote the set of homotopy classes of maps from $X$ to $Y$.
\end{definition}

\begin{lemma}\label{lem:kTheo-gluingAndHomotopyClassesToGLn}
	Let $X=X_1\cup X_2$ with $X_1$ and $X_2$ closed contractible spaces. Then there is a canonical bijection 
	\begin{equation*}
		\Vect_n(X) \to [A,\GL_n(\C)]\,.
	\end{equation*}
\end{lemma}
\begin{proof}
	Let $[E]$ be any vector bundle over $X$ of rank $n$. Then $E$ can be represented by a vector bundle obtained by gluing that is trivial on both $X_i$: Since $X_i$ is contractible we can choose two trivializations
	\begin{equation*}
		\alpha_1:\restr{E}{X_1}\to X_1\times \C^n,\quad \text{and}\quad 	\alpha_2:\restr{E}{X_2}\to X_2\times \C^n\,.
	\end{equation*} With part (2) of Corollary \ref{cor:kTheo-elementaryPropertiesGluingAndClutching} we can conclude:
	\begin{equation*}
		E\,\cong\,\restr{E}{X_1}\cup_{\id_A} \restr{E}{X_2}\, \cong\, 
		X\times \C^n \cup_{\restr{\alpha_2\circ \alpha_1^{-1}}{A}} X\times \C^n
	\end{equation*} Since a map between the two trivial bundles $A\times \C^n\to A\times \C^n$ can be identified with a map $A\to \GL_n(\C)$ we get an assignment from bundles over $X$ together with the trivializations to maps $A\to \GL_n(\C)$. The next step is to show that two isomorphic vector bundles give homotopic maps: Assume that we are given two vector bundles $E,F$ over $X$ that are isomorphic via $T:E\to F$ and without loss of generality we assume that they are given as the gluing of two trivial bundles: 
	\begin{equation*}
		E: X_1\times \C^n \,\cup_{\phi}\, X_2\times \C^n\,,\quad\text{and}\quad  
		F: X_1\times \C^n \,\cup_{\psi}\, X_2\times \C^n\,.
	\end{equation*}
	We now want to define a homotopy between $psi$ and $\phi$. For this we start by splitting the isomorphism $T$ into its two parts over $X_1$ and $X_2$:
	\begin{equation*}
		\restr{T}{X_i}: X_i\times \C^n \to X_i\times \C^n\,.
	\end{equation*} These maps are given by maps 
	\begin{equation*}
		T_i: X_i\to \GL_n(\C)\,,
	\end{equation*} where the gluing compatibility of the $\restr{T}{X_i}$ yield for any $a\in A$:
	\begin{equation*}
		\psi(a)T_1(a)=T_2(a)\phi(a)\,,\quad \text{ equivalently }\quad \psi(a)=T_2(a)\phi(a)T_1^{-1}(a)\,.
	\end{equation*} By the contractibility of $X_i$ we have for any fixed $x_i\in X_i$ a homotopy $h^i_t:X_i\to X_i$ such that $h^i_0=c_{x_i}$ is the collapsing map whilst $h^i_1=\id_{X_i}$. Furthermore, since $\GL_n(\C)$ is connected we have a path $\gamma^i: I\to \GL_n(\C)$ that connects $T_i(x_i)=\gamma^i(1)$ to $\Idm_n=\gamma^i(0)$. With this we can define the homotopies
	\begin{align*}
		T_{i,t}: X_i&\to \GL_n(\C)\\
		x &\mapsto \begin{cases}
			T_i\circ h^i_{2t}(x)& \text{ if }t\leq \frac{1}{2}\,,\\
			\gamma^i(2(t-\frac{1}{2}))& \text{ if }t > \frac{1}{2}\,,
		\end{cases}
	\end{align*} that satisfy $T_{i,1}=T_i$ and $T_{i,0}(x)=\Idm_n$ for all $x$. Finally we define the homotopy $H_t:A \to \GL_n(\C)$ by
	\begin{equation*}
		H_t(a)\coloneq T_{2,t}(a)\phi(a)T_{1,t}^{-1}(a)\,
	\end{equation*} to get a homotopy from $\phi=H_0$ to $\psi=H_1$. Hence, we get a well defined map from isomorphism classes of bundles over $X$ of rank $n$ to the homotopy classes of maps from $A$ to $\GL_n(\C)$ that is the inverse of the gluing construction and hence is bijective.
\end{proof}
\begin{definition}[Metrics on Bundles]\label{def:kTheo-metricsOnBundles}
	A \textbf{metric} on a complex bundle $E$ is a section $h:X\to \Herm(E)$ such that $h(x)$ is positive definite for all $x$. $\Herm(E)$ denotes the bundle of hermitian forms (conjugate-symmetric sesquilinear) in each fiber. A bundle together with a metric is a \textbf{Hermitian bundle}.
\end{definition}
\begin{definition}[Projection Operator]\label{def:kTheo-projectionOperator}
	A \textbf{projection operator} $P: E\to E$ is a homomorphism such that $P^2=P$. 
\end{definition}
\begin{lemma}\label{lem:kTheo-projectionsGiveSumdecomposition}
	Any projection operator $P:E\to E$ determines a direct sum decomposition $E=PE\oplus (1-P)E$.
\end{lemma}
\begin{proof}
	Observe that fibrewise $1-P$ maps each $\xi$ to the kernel of $P$. Furthermore $\im(P)+\im(1-P)=E$. 
	That is because $v=P(v)+(v-P(v))$. Hence, $\rank(P_x)+\rank(1-P)=\dim(E_x)$. Furthermore, $\rank(P_x)$ and $\rank(1-P)$ are locally constant. To see this first notice that they are lower semi continuous, meaning that for $y$ from a small enough neighbourhood of $x$ $\rank(P_x)\leq \rank(P_y)$. To see this, pick a $\rank(P_x)_x\times \rank(P_x)_x$ minor with nonzero determinant. The latter is an open condition and hence stable under small pertubation. But now we have the sum of two lower semi continuous functions that give something locally constant and hence, each summand is locally constant. Finally the above shows that $\ker(P)=(1-P)(E)$ we have the decomposition.
\end{proof}
\begin{lemma}[Metric Induced Complement] \label{lem:kTheo-metricInducedComplement}
	Suppose that $F$ is a sub-bundle of $E$. Then a metric provides a canonical complementary sub-bundle.
\end{lemma}
\begin{proof}
	For each $x$ we have the orthonormal projection $P_x:E_x\to F_x$ inducing a map $P:E \to F$. This is continuous: 
	Assume, that $F$ is trivial with the sections $f_1, \dots ,f_n$ such that $(f_1,\dots,f_n)$ give fibrewise an orthonormal basis. Then, for $v\in F_x$ we have 
	\begin{equation*}
		P_x(v)=\sum_i h_x(v,f_i(x))f_i(x)\,.
	\end{equation*}
	now $h$ is continuous and thereby $P$. Furthermore $P^2=P$ and hence we get a canonical sum decomposition.
\end{proof}
